PROJECT IN ONE PAGE
----------------------------------------------------------------------------------------------------
* NXROBO Spark / Spark-H: wheeled mobile manipulator for the 10-minute fully
  autonomous sorting challenge (自动任务赛 / «Полный автоном»).
* Arena 3x3 m, four corner zones A-D, 10 cubes.  The tier-2 Cake MUST be delivered
  to Zone A first; the other 9 cubes are sorted by the classification board.
* Hardware: Intel NUC (Ubuntu 20.04, ROS Noetic), RealSense D435, YDLidar G6,
  differential base, uArm Swift Pro 4-DOF + vacuum pump.
* Software: match_logic.py (ROS-free rules/IK/vision logic) + runtask.py (FSM) +
  detect_ros.py (two YOLO models -> /objects) + grasp_generic.py (field grasp test).
* Current state: floor cubes and the tier-2 Cake are grasped in field tests; the
  10-minute FSM has NOT yet run end-to-end because the physical arena/map is missing.
* Tests: 141/141 PASS (ROS-free unittest suite, importing the production modules).

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Pure match logic for the Spark autonomous competition (ROS-free).

WHY THIS MODULE EXISTS
----------------------
Everything that decides *what the robot does* (which cube to pick, which zone
it belongs to, which tier to stack on, when to give up) lives here, with no
ROS, no hardware and no global state.  ``runtask.py`` only wires these
functions to topics and actuators, and ``codes/tests/`` imports THIS module.

That is the whole point: the unit tests must exercise the code that actually
runs on the robot.  Before this module existed, the tests re-implemented the
logic inside the test files, so they passed while production code was broken.

Deployment: this file must sit next to ``runtask.py`` in
``auto_match/nodes/`` (``upload.ps1`` copies it there) and next to
``arm_ik.py`` in ``~/codes/`` (the compatibility shim imports it).

Conventions used everywhere below
---------------------------------
* Item classes: 1 = Teddy bear, 2 = Cake/Wine, 3 = Clock, 4 = Banana.
  YOLO also emits "<item>_site" aliases (10/20/30/40); ``norm_id`` folds them.
* Detections are tuples ``(raw_id, score, cx, cy)`` in image pixels.
* Cube lists are the historical format ``[[canonical_id, [x, y, 0]], ...]``.
* Zones: Collection_A (red square, cake only) and Collection_B/C/D.
* Tiers: 1 = floor, 2 = on a cube, 3 = on two cubes (``MAX_TIER``).
"""

import math
import os
import time
from collections import Counter

# --------------------------------------------------------------------------
# Class identifiers and normalization
# --------------------------------------------------------------------------

TEDDY_ID = 1
CAKE_ID = 2
CLOCK_ID = 3
BANANA_ID = 4

CANONICAL_IDS = (TEDDY_ID, CAKE_ID, CLOCK_ID, BANANA_ID)
# Items that appear on the classification board (the cake never does).
BOARD_ITEMS = (TEDDY_ID, CLOCK_ID, BANANA_ID)

ID_MAP = {
    1: 1, 10: 1,
    2: 2, 20: 2,
    3: 3, 30: 3,
    4: 4, 40: 4,
}


def norm_id(raw_id):
    """Map a raw YOLO class (including ``*_site`` aliases) to a canonical id."""
    try:
        return ID_MAP.get(int(raw_id), int(raw_id))
    except Exception:
        return raw_id


def is_cake(item_id):
    """True when the item is the competition cake (rule-critical item)."""
    return norm_id(item_id) == CAKE_ID


# --------------------------------------------------------------------------
# Zones, tiers and scoring
# --------------------------------------------------------------------------

ZONES_BCD = ("Collection_B", "Collection_C", "Collection_D")
# Collection_A is the red 0.45 x 0.45 m acceptance square reserved for the cake.
RED_ZONES = ("Collection_A",)
MAX_TIER = 3
TOTAL_CUBES = 10
# Drop routine per tier index (0 -> floor, 1 -> 2nd tier, 2 -> 3rd tier).
DROP_ROUTINES = ("drop", "drop1", "drop2")

# Used when the classification board cannot be read (SI-14 fallback).
DEFAULT_MAPPING = {
    CAKE_ID: "Collection_A",
    TEDDY_ID: "Collection_B",
    CLOCK_ID: "Collection_C",
    BANANA_ID: "Collection_D",
}


def score_cube(is_red_zone, tier):
    """Competition points for one cube placed at ``tier`` (1..3)."""
    if tier < 1 or tier > MAX_TIER:
        raise ValueError(f"Invalid tier {tier}, must be 1, 2 or 3")
    if is_red_zone:
        return tier + (2 ** (tier - 1))
    return 2 ** (tier - 1)


def match_total(deliveries, started_from_base=True):
    """Score a full delivery sequence; 0 when the cake-first rule is violated.

    ``deliveries``: iterable of dicts with keys cube_type, zone, is_red, tier.
    """
    deliveries = list(deliveries)
    if not deliveries:
        return 1 if started_from_base else 0
    first = deliveries[0]
    if first.get("cube_type") != "cake" or first.get("zone") != "Collection_A":
        return 0
    total = 1 if started_from_base else 0
    for delivery in deliveries:
        total += score_cube(delivery["is_red"], delivery["tier"])
    return total


def estimate_score(drop_times, red_zones=RED_ZONES, started_from_base=True):
    """Estimate the match score from per-zone drop counts only.

    Used for the end-of-run log line.  It assumes the cake was delivered first
    and that each zone was filled bottom-up, which is exactly how the FSM
    drops cubes, so it is an upper bound when the cake rule was violated.
    """
    total = 1 if started_from_base else 0
    for zone, count in drop_times.items():
        for tier in range(1, min(int(count), MAX_TIER) + 1):
            total += score_cube(zone in red_zones, tier)
    return total


def plan_drop(zone_drop_count):
    """Name of the arm routine for the next cube in a zone, or None if full."""
    if zone_drop_count >= MAX_TIER:
        return None
    return DROP_ROUTINES[zone_drop_count]


def choose_drop_zone(item_id, mapping, drop_times, zones=ZONES_BCD):
    """Zone for ``item_id``: board mapping first, then the first non-full zone.

    Returns None only when every zone already holds ``MAX_TIER`` cubes, which
    must never happen with 10 cubes and 4 zones but is handled instead of
    dropping a cube into thin air.
    """
    cid = norm_id(item_id)
    primary = mapping.get(cid) or mapping.get(item_id)
    candidates = []
    if primary:
        candidates.append(primary)
    candidates.extend(zone for zone in zones if zone not in candidates)
    for zone in candidates:
        if drop_times.get(zone, 0) < MAX_TIER:
            return zone
    return None

# Classification board -> zone mapping

def expand_aliases(mapping):
    """Return a copy of ``mapping`` that also answers alias ids (10/20/30/40)."""
    out = dict(mapping)
    for alias, cid in ID_MAP.items():
        if cid in out and alias not in out:
            out[alias] = out[cid]
    return out


def normalize_mapping(mapping=None):
    """Guarantee that every canonical item has a zone and the cake is in A."""
    out = dict(DEFAULT_MAPPING if mapping is None else mapping)
    for cid, zone in DEFAULT_MAPPING.items():
        out.setdefault(cid, zone)
    out[CAKE_ID] = "Collection_A"
    return expand_aliases(out)


def board_items(slot_items):
    """Unique canonical board items in left-to-right order, cake excluded."""
    items = []
    for raw in slot_items:
        if raw is None:
            continue
        cid = norm_id(raw)
        if cid in BOARD_ITEMS and cid not in items:
            items.append(cid)
    return items


def board_is_confident(slot_items):
    """True when the board yielded three distinct, usable items."""
    return len(board_items(slot_items)) >= len(ZONES_BCD)


def decode_classification_board(slot_items, fallback=None):
    """Build ``{item_id: zone}`` from the board slots (left to right).

    On a partial/duplicated/unreadable board the fallback mapping is returned
    unchanged, so the mission can keep scoring instead of aborting (SI-14).
    """
    mapping = normalize_mapping(fallback)
    items = board_items(slot_items)
    if len(items) < len(ZONES_BCD):
        return mapping
    mapping[CAKE_ID] = "Collection_A"
    for cid, zone in zip(items[:len(ZONES_BCD)], ZONES_BCD):
        mapping[cid] = zone
    return expand_aliases(mapping)


# --------------------------------------------------------------------------
# uArm Swift Pro kinematics (the only implementation in the project)
# --------------------------------------------------------------------------
# The maths mirrors swiftpro_rviz_node.cpp / the vendor examples.  If the
# firmware's IK cannot reach a pose it silently refuses the command and the arm
# simply does not move, so every target is checked before it is published.
# Coordinate convention (from every Spark grasp example):
#     arm_x = k1 * pixel_row + b1
#     arm_y = k2 * pixel_col + b2
# i.e. the image row drives x and the image column drives y.  Do not "fix" this
# by swapping: it is how the calibration files were generated.

MATH_TRANS = 57.2958
MATH_L1 = 106.6
MATH_L2 = 13.2
MATH_LOWER_ARM = 142.07
MATH_UPPER_ARM = 158.81
MATH_UPPER_LOWER = MATH_UPPER_ARM / MATH_LOWER_ARM
MATH_Z_OFFSET = 74.55
MATH_X_OFFSET = 56.55


# Camera intrinsics constants for D435 RealSense
CAM_FX = 605.6510620117188
CAM_FY = 605.8469848632812
CAM_CX = 315.03875732421875
CAM_CY = 249.8477020263672


def matrix_columns(M):
    """Columns of a 4x3 matrix: works for numpy arrays and plain lists.

    ``pixel_to_xy_at_z`` is called both with the np.load()-ed calibration and
    with hand-written lists from the unit tests, so no slicing syntax that only
    numpy understands.
    """
    return ([M[i][0] for i in range(4)],
            [M[i][1] for i in range(4)],
            [M[i][2] for i in range(4)])


def pixel_to_xy_at_z(M, u, v, z,
                     fx=CAM_FX, fy=CAM_FY, cx=CAM_CX, cy=CAM_CY):
    """Closed-form inverse of the 3D projective map at a fixed height z (mm).

    Given the 4x3 DLT camera-to-arm matrix M, pixel (u, v) and plane height z,
    returns the exact (x, y) coordinates in millimetres in the arm frame.
    """
    up = (u - cx) / fx
    vp = (v - cy) / fy
    r0, r1, r2 = matrix_columns(M)
    C0 = r0[2] * z + r0[3]
    C1 = r1[2] * z + r1[3]
    C2 = r2[2] * z + r2[3]
    a11 = r0[0] - up * r2[0]
    a12 = r0[1] - up * r2[1]
    a21 = r1[0] - vp * r2[0]
    a22 = r1[1] - vp * r2[1]
    b1 = up * C2 - C0
    b2 = vp * C2 - C1
    det = a11 * a22 - a12 * a21
    if abs(det) < 1e-12:
        return None
    return float((b1 * a22 - a12 * b2) / det), float((a11 * b2 - b1 * a21) / det)


def load_cake_calibration(path=None):
    """Load the dedicated Layer 2 (Cake) calibration parameters (k1, b1, k2, b2)."""
    if path is None:
        candidates = [
            os.path.expanduser("~/thefile2.txt"),
            os.path.join(os.path.dirname(__file__), "thefile2.txt"),
        ]
        for c in candidates:
            if os.path.exists(c):
                path = c
                break
    if path and os.path.exists(path):
        try:
            with open(path, "r", encoding="utf-8") as f:
                vals = [float(v) for v in f.read().split()]
                if len(vals) >= 4:
                    return (vals[0], vals[1]), (vals[2], vals[3])
        except Exception:
            pass
    # Built-in fallback parameters from shipped thefile2.txt:
    # -0.8030661953564442 395.53156305089396 -0.6955960479077558 247.96237774624151
    return (-0.8030661953564442, 395.53156305089396), (-0.6955960479077558, 247.96237774624151)


def pixel_to_cake_xy(M, u, v, fx=CAM_FX, fy=CAM_FY, cx=CAM_CX, cy=CAM_CY,
                     calib_file=None, cup_offset_x=125.0, clamp=True):
    """Dedicated approximation for the competition Cake (Tier 2 / 20 cm height).

    Uses the dedicated Layer 2 calibration (thefile2.txt) created specifically
    for the elevated Cake.
    cup_offset_x = +125.0 mm places the suction cup dead-center on the Cake cube
    (340-348 mm from arm base, compensating for detector finding vertical face).

    ``clamp=True`` (the default) keeps the result inside the arm's physical
    workspace (120..348 mm) and is what the final grasp plan needs: a Cake seen
    on its top face looks "too far" to the affine mapping, so the clamp stops a
    perfectly reachable cake from being declared out of reach.

    ``clamp=False`` returns the raw affine estimate.  That is required by the
    base-approach step: a Cake that is really standing further away must keep
    its large ``x`` so the robot drives to it instead of trying to pick air.
    """
    x_kb, y_kb = load_cake_calibration(calib_file)
    cake_x = float(x_kb[0] * v + x_kb[1] + cup_offset_x)
    cake_y = float(y_kb[0] * u + y_kb[1])
    if not clamp:
        return cake_x, cake_y
    # Clamped at hardware limit (348.0 mm) so IK never enters singularity
    cake_x = max(120.0, min(348.0, cake_x))
    cake_y = max(-220.0, min(220.0, cake_y))
    return cake_x, cake_y


def pixel_to_arm(pixel_row, pixel_col, x_kb, y_kb):
    """Return (x, y) in millimetres from an image pixel.

    ``x_kb = (k1, b1)`` and ``y_kb = (k2, b2)`` as stored in ``~/thefile.txt``.
    """
    return x_kb[0] * pixel_row + x_kb[1], y_kb[0] * pixel_col + y_kb[1]


def ik_reachable(x, y, z):
    """True if the arm can reach (x, y, z) in millimetres."""
    try:
        z = z + MATH_Z_OFFSET
        z_in = (z - MATH_L1) / MATH_LOWER_ARM
        if x < 0.1:
            x = 0.1
        if y == 0:
            rot = 90.0
        elif y < 0:
            rot = -math.atan(x / y) * MATH_TRANS
        else:
            rot = 180.0 - math.atan(x / y) * MATH_TRANS

        sin_rot = math.sin(rot / MATH_TRANS)
        if sin_rot == 0:
            return False

        x_in = (x / sin_rot - MATH_L2 - MATH_X_OFFSET) / MATH_LOWER_ARM
        if x_in == 0:
            return False

        sqrt_zx = math.sqrt(z_in * z_in + x_in * x_in)
        if sqrt_zx == 0:
            return False

        upper = MATH_UPPER_LOWER
        right_all = (sqrt_zx * sqrt_zx + upper * upper - 1) / (2 * upper * sqrt_zx)
        left_all = (sqrt_zx * sqrt_zx + 1 - upper * upper) / (2 * sqrt_zx)
        if abs(right_all) > 1 or abs(left_all) > 1:
            return False

        phi = math.atan2(z_in, x_in) * MATH_TRANS
        angle_right = math.acos(right_all) * MATH_TRANS - phi
        angle_left = math.acos(left_all) * MATH_TRANS + phi

        # Firmware/hardware mechanical limits (uArm Swift Pro):
        # theta1 (lower arm): [0.0, 135.6] deg
        # theta2 (upper arm / elbow): [0.0, 100.7] deg (cannot be negative)
        # combined span: [10.0, 151.0] deg
        return (0.0 <= angle_left <= 135.6 and
                0.0 <= angle_right <= 100.7 and
                10.0 <= (angle_left + angle_right) <= 151.0)
    except Exception:
        return False


def pixel_reachable(pixel_row, pixel_col, z_list, x_kb, y_kb):
    """True if the pixel is reachable at every height in ``z_list``."""
    arm_x, arm_y = pixel_to_arm(pixel_row, pixel_col, x_kb, y_kb)
    return all(ik_reachable(arm_x, arm_y, z) for z in z_list)


# --------------------------------------------------------------------------
# Grasp geometry and reach envelopes (field test + FSM wiring)
# --------------------------------------------------------------------------
# All heights are uArm wrist coordinates in millimetres.
PRESS_MM = 3.0                      # suction-cup compression on contact
FLOOR_TOUCH_Z = -55.0               # wrist height at contact with a floor cube
DEFAULT_CAKE_TOUCH_Z = 35.0         # contact height on the raised tier-2 cake
APPROACH_FLOOR_Z = 70.0             # safe clearance height above the floor
LIFT_FLOOR_Z = 60.0                 # wrist height of the raised cube
APPROACH_CAKE_Z = 75.0              # safe clearance height above the tier
LIFT_CAKE_Z = 80.0                  # wrist height of the raised cake

# Standard floor cube (uArm physical reach extends safely to 300 mm):
FLOOR_IN_REACH = dict(x_lo=140.0, x_hi=290.0, y_lim=100.0, dist_max=295.0,
                      dist_target=240.0)
# Tier-2 cake (reaching over the front row).  dist_target=305 keeps the
# chassis >6 cm away from first-row cubes so they are not knocked over.
CAKE_IN_REACH = dict(dist_target=305.0)


def grasp_heights(mode, z_est=None, cake_touch_z=DEFAULT_CAKE_TOUCH_Z):
    """Return the (approach, touch, lift) wrist heights for one target."""
    elevated = (mode == "cake" and (z_est is None or z_est > -70.0))
    if elevated:
        return (APPROACH_CAKE_Z, float(cake_touch_z) - PRESS_MM, LIFT_CAKE_Z)
    return (APPROACH_FLOOR_Z, FLOOR_TOUCH_Z - PRESS_MM, LIFT_FLOOR_Z)


def grasp_reach_state(mode, x, y, z_est=None, far=False):
    """Is the target inside the comfortable arm envelope AND the firmware IK?

    The IK check matters: the firmware refuses unreachable poses silently, so
    "in reach" based on distance alone would make the robot lift, hold and
    report DONE without ever touching the cube.

    ``far=True`` marks a detection taken outside the calibrated near field (the
    cube is beyond the arm's workspace).  Its (x, y) is only a coarse approach
    estimate, so it is NEVER in reach: the base has to drive first and the next
    scan - taken from the calibrated band - refines the target.  This is what
    makes the robot approach a cube instead of failing with NO_TARGET.
    """
    dist = math.hypot(x, y)
    if far:
        return False, dist
    _, z_touch, z_lift = grasp_heights(mode, z_est, DEFAULT_CAKE_TOUCH_Z)
    ik_ok = (ik_reachable(x, y, z_touch)
             and ik_reachable(x, y, z_lift))
    if mode == "cake":
        # Lateral/horizontal limits completely removed per operator directive;
        # height validation only:
        if z_est is not None and not (-140.0 <= z_est <= 130.0):
            return False, dist
        if dist > (CAKE_IN_REACH["dist_target"] + 50.0):
            return False, dist
        return ik_ok, dist
    env = FLOOR_IN_REACH
    in_reach = (env["x_lo"] <= x <= env["x_hi"] and abs(y) <= env["y_lim"]
                and dist <= env["dist_max"] and ik_ok)
    return in_reach, dist


def approach_target_distance(mode, x, y):
    """Return ``(current distance, desired distance)`` for a base approach."""
    if mode == "cake":
        return math.hypot(x, y), CAKE_IN_REACH["dist_target"]
    return math.hypot(x, y), FLOOR_IN_REACH["dist_target"]


def approach_cost(target):
    """How expensive it is to bring ``target`` into reach (lower is better).

    A target that is already graspable wins outright (negative cost).  For the
    rest, the cost mixes the heading error (30 deg ~ 1.0) with the distance
    error (300 mm ~ 1.0).  Without the heading term a low-confidence detection
    at the frame corner could make the robot swing ~60 deg away from the cube
    that is standing right ahead - and with the camera then pointing at the
    empty field the next scan found nothing at all.

    ``target`` is a scan dict as built by the field test:
    ``{mode, class, pixel, xy, z_est, far, ...}``.
    """
    mode = str(target.get("mode", "floor"))
    x = float(target["xy"][0])
    y = float(target["xy"][1])
    pixel = target.get("pixel") or [320.0, 0.0]
    centre = abs(float(pixel[0]) - 320.0) / 6400.0     # tie-break: keep it central
    in_reach, dist = grasp_reach_state(mode, x, y, z_est=target.get("z_est"),
                                       far=target.get("far", False))
    if in_reach:
        return -1.0 + centre
    heading = abs(math.atan2(y, x)) / math.radians(30.0)
    desired = approach_target_distance(mode, x, y)[1]
    return heading + abs(dist - desired) / 300.0 + centre


def choose_scan_target(candidates):
    """Pick one scan target: Cake priority first, then the cheapest approach.

    The Cake is the only rule-critical item, so a Cake candidate always beats a
    standard cube (operator rule).  Within the winning class a graspable target
    is taken immediately; otherwise the target with the lowest
    :func:`approach_cost` drives the base.
    """
    pool = [candidate for candidate in candidates if candidate]
    if not pool:
        return None
    cakes = [candidate for candidate in pool
             if candidate.get("is_cake") or is_cake(candidate.get("class", 0))]
    return min(cakes if cakes else pool, key=approach_cost)




# --------------------------------------------------------------------------
# Hand-eye calibration guards
# --------------------------------------------------------------------------
# A stale hand-eye matrix is the most expensive silent failure in this
# project: a cube pixel is converted to an arm target that is 5-7 cm off, the
# arm descends into empty space, and nothing reports an error.  These helpers
# catch the cheap cases at startup (missing/corrupted/degenerate matrix, wrong
# units) and give the operator one number to look at after the camera moves.

CALIB_IMAGE_SIZE = (640, 480)      # D435 color stream
CALIB_SCALE_RANGE = (0.3, 3.0)     # px per mm at the grasp workspace
CALIB_MAX_ROUNDTRIP_MM = 25.0      # project -> inverse must come back
CALIB_WORKSPACE_PROBE = (270.0, 0.0, -50.0)   # centre of the grasp workspace
CALIB_ROUNDTRIP_Z = (-100.0, -50.0, 0.0)


def project_arm(M, x, y, z, fx=CAM_FX, fy=CAM_FY, cx=CAM_CX, cy=CAM_CY):
    """Project an arm-frame wrist point (mm) into the color image.

    Returns ``(u, v)`` in pixels, or ``None`` when the point is behind the
    camera.  This is the forward direction of the same 4x3 matrix that
    ``pixel_to_xy_at_z`` inverts.
    """
    try:
        cam = [x * M[0][j] + y * M[1][j] + z * M[2][j] + M[3][j]
               for j in range(3)]
        if cam[2] <= 1e-9:
            return None
        return (fx * cam[0] / cam[2] + cx, fy * cam[1] / cam[2] + cy)
    except Exception:
        return None


def pad_projection_error(M, pose, observed_uv):
    """Pixel distance between the projected wrist pose and the seen pad.

    ``pose`` is the commanded (x, y, z) in mm; ``observed_uv`` is the orange
    pad centroid detected in the frame.  Returns ``None`` if the pose is not
    projectable.
    """
    projected = project_arm(M, pose[0], pose[1], pose[2])
    if projected is None:
        return None
    return math.hypot(projected[0] - observed_uv[0],
                      projected[1] - observed_uv[1])


def check_transform(M, fx=CAM_FX, fy=CAM_FY, cx=CAM_CX, cy=CAM_CY):
    """Numeric sanity check of a 4x3 camera<-arm matrix.

    Returns ``(ok, problems, stats)``.  ``problems`` is a list of human
    readable strings (empty when ok); ``stats`` carries the numbers worth
    logging: px/mm at the workspace and the project->inverse round-trip error.

    This cannot detect a camera that was physically moved after calibration -
    only the active pad check can (``calib3d.py --check`` / the startup check).
    It does catch the other ways a calibration file goes bad: truncated or
    non-numeric file, NaN/inf, a degenerate fit whose inverse returns garbage,
    and a workspace that projects outside the image.
    """
    problems = []
    stats = {}
    try:
        if len(M) != 4 or any(len(row) != 3 for row in M):
            return False, ["matrix is not 4x3"], {}
        rows = [[float(value) for value in row] for row in M]
    except Exception as exc:
        return False, ["matrix is not a 4x3 array of numbers (%s)" % exc], {}

    if not all(math.isfinite(value) for row in rows for value in row):
        return False, ["matrix contains NaN/inf"], {}

    # Volume of the linear part: a collapsed fit (all points on one plane)
    # passes every other check but inverts to garbage.
    a, b, c = rows[0][:3]
    d, e, f = rows[1][:3]
    g, h, i = rows[2][:3]
    det = a * (e * i - f * h) - b * (d * i - f * g) + c * (d * h - e * g)
    if abs(det) < 1e-15:
        problems.append("linear part is degenerate (det=%.3g)" % det)

    centre = project_arm(rows, *CALIB_WORKSPACE_PROBE, fx=fx, fy=fy, cx=cx, cy=cy)
    if centre is None:
        problems.append("workspace centre projects behind the camera")
    else:
        width, height = CALIB_IMAGE_SIZE
        stats["workspace_uv"] = (round(centre[0], 1), round(centre[1], 1))
        if not (-0.5 * width <= centre[0] <= 1.5 * width
                and -0.5 * height <= centre[1] <= 1.5 * height):
            problems.append(
                "workspace centre projects far outside the image at (%.0f, %.0f)"
                % centre)

    nearby = project_arm(rows, CALIB_WORKSPACE_PROBE[0] + 10.0,
                         CALIB_WORKSPACE_PROBE[1], CALIB_WORKSPACE_PROBE[2],
                         fx=fx, fy=fy, cx=cx, cy=cy)
    if centre is not None and nearby is not None:
        scale = math.hypot(nearby[0] - centre[0], nearby[1] - centre[1]) / 10.0
        stats["px_per_mm"] = round(scale, 3)
        low, high = CALIB_SCALE_RANGE
        if not low <= scale <= high:
            problems.append(
                "scale is %.3f px/mm, outside %.1f-%.1f (wrong units?)"
                % (scale, low, high))

    worst = 0.0
    for z in CALIB_ROUNDTRIP_Z:
        projected = project_arm(rows, CALIB_WORKSPACE_PROBE[0],
                                CALIB_WORKSPACE_PROBE[1], z,
                                fx=fx, fy=fy, cx=cx, cy=cy)
        if projected is None:
            problems.append("z=%.0f projects behind the camera" % z)
            continue
        back = pixel_to_xy_at_z(rows, projected[0], projected[1], z,
                                fx=fx, fy=fy, cx=cx, cy=cy)
        if back is None:
            problems.append("z=%.0f inverse is degenerate" % z)
            continue
        error = math.hypot(back[0] - CALIB_WORKSPACE_PROBE[0],
                           back[1] - CALIB_WORKSPACE_PROBE[1])
        worst = max(worst, error)
    stats["roundtrip_mm"] = round(worst, 2)
    if worst > CALIB_MAX_ROUNDTRIP_MM:
        problems.append(
            "project->inverse round-trip error is %.0f mm (limit %.0f)"
            % (worst, CALIB_MAX_ROUNDTRIP_MM))

    return (not problems), problems, stats

#: Anti-glare gate for the raw Cake class (id 2 / plain "cake").
#: Measured on the real field: false cake hypotheses on banana cubes reached
#: score 0.852, so this class stays strict.  The real Cake item on this field
#: is a printed "cake" cube and the detector reports it as ``cake_site``

CAKE_CONF_THRESHOLD = 0.90

# --------------------------------------------------------------------------
# Scan window: calibrated near field vs. far field (base approach)
# --------------------------------------------------------------------------
# The hand-eye matrix is fitted around the grasp workspace only, so a pixel in
# the NEAR field yields millimetre coordinates.  A cube that stands further
# away appears ABOVE that band, near the horizon.  Such a pixel is still a
# perfectly valid target - the base just has to drive to it.  Dropping those
# detections (as the field test used to) made every out-of-reach cube end in
# NO_TARGET, i.e. exit code 3, with the robot never moving.
NEAR_FIELD_U = (80.0, 560.0)
NEAR_FIELD_V = (130.0, 450.0)
#: Loose sanity window for the far field: only the image edge / off-camera
#: noise is rejected, so a real cube can still drive the base.
SCAN_WINDOW_U = (20.0, 620.0)
SCAN_WINDOW_V = (25.0, 470.0)
#: Plane used to invert a far floor-cube pixel.  The depth-based reconstruction
#: extrapolates badly outside the calibrated volume (measured +35 mm at 0.65 m),
#: while the grasp-plane inversion degrades gracefully; the next scan after the
#: approach step refines it.  Same convention as the competition grasp
#: (z=-100 plane + 20 mm cup correction).
FAR_FLOOR_PLANE_Z = -100.0
FAR_FLOOR_CUP_OFFSET_X = 20.0


def in_near_field(u, v):
    """True when the pixel lies in the calibrated, grasp-ready near field."""
    return (NEAR_FIELD_U[0] <= u <= NEAR_FIELD_U[1]
            and NEAR_FIELD_V[0] <= v <= NEAR_FIELD_V[1])


def in_scan_window(u, v):
    """True when the pixel is a plausible on-field detection worth chasing."""
    return (SCAN_WINDOW_U[0] <= u <= SCAN_WINDOW_U[1]
            and SCAN_WINDOW_V[0] <= v <= SCAN_WINDOW_V[1])


def far_target_xy(M, u, v, is_cake, cup_offset_x=125.0,
                  plane_z=FAR_FLOOR_PLANE_Z,
                  cup_correction_x=FAR_FLOOR_CUP_OFFSET_X):
    """Approximate (x, y) in millimetres for a target beyond the near field.

    A far target only needs enough geometry for ONE base-approach step: the
    heading and the distance.  The near-field depth reconstruction and the
    clamped cake affine are both invalid at that range, so:

    * a floor cube is inverted onto the known grasp plane (``plane_z``) and
      shifted by the cup correction, exactly like the competition grasp;
    * a Cake uses the dedicated affine WITHOUT the reach clamp, so a really
      distant cake keeps a large ``x`` and drives the base instead of being
      reported as already in reach.

    Returns ``None`` when the inverse mapping is degenerate.
    """
    if is_cake:
        return pixel_to_cake_xy(M, u, v, cup_offset_x=cup_offset_x, clamp=False)
    xy = pixel_to_xy_at_z(M, u, v, plane_z)
    if xy is None:
        return None
    return float(xy[0]) + cup_correction_x, float(xy[1])



def cluster_detections(frames, conf_threshold=CONF_THRESHOLD,
                       cluster_dist_px=CLUSTER_DIST_PX, min_hits=None,
                       cake_conf_threshold=CAKE_CONF_THRESHOLD,
                       cake_site_conf_threshold=CAKE_SITE_CONF_THRESHOLD):
    """Temporal voting + spatial clustering over a window of detection frames.

    ``frames``: sequence of frames, each an iterable of ``(raw_id, score, cx, cy)``.
    Returns ``[[canonical_id, [x, y, 0]], ...]`` — one entry per physical cube,
    with the majority class and the averaged centre.

    Cake detections additionally have to pass a per-class anti-glare gate:
      * raw id 2  (``cake``)      -> ``cake_conf_threshold``
      * raw id 20 (``cake_site``) -> ``cake_site_conf_threshold``
    The Cake class has priority in the FSM, so a cheap false positive would
    otherwise send the robot after a non-existent cake.  The two raw classes
    have different false-positive profiles, hence two separate thresholds.

    Because detections from *all* frames are poured into the same spatial
    clusters, this also collapses a cube reported twice in one frame (the
    detector runs two YOLO models), which single-frame code cannot do.
    """
    frames = [list(frame) for frame in frames if frame is not None]
    if not frames:
        return []

    if min_hits is None:
        if len(frames) < 2:
            min_hits = 1
        else:
            min_hits = max(2, int(round(len(frames) * MIN_HIT_RATIO)))

    cake_conf_threshold = float(cake_conf_threshold)
    cake_site_conf_threshold = float(cake_site_conf_threshold)
    clusters = []
    for frame in frames:
        for detection in frame:
            try:
                raw_id, score, cx, cy = detection[:4]
            except Exception:
                continue
            cid = norm_id(raw_id)
            if cid not in CANONICAL_IDS:
                continue
            if score is None:
                continue
            if cid == CAKE_ID:
                try:
                    is_site = int(raw_id) != CAKE_ID
                except (TypeError, ValueError):
                    is_site = False
                cake_min = (cake_site_conf_threshold if is_site
                            else cake_conf_threshold)
                min_score = max(float(conf_threshold), cake_min)
            else:
                min_score = float(conf_threshold)
            if float(score) < min_score:
                continue

            match = None
            for cluster in clusters:
                mean_x = sum(cluster["xs"]) / len(cluster["xs"])
                mean_y = sum(cluster["ys"]) / len(cluster["ys"])
                if math.hypot(cx - mean_x, cy - mean_y) < cluster_dist_px:
                    match = cluster
                    break
            if match is None:
                match = {"ids": [], "xs": [], "ys": []}
                clusters.append(match)
            match["ids"].append(cid)
            match["xs"].append(float(cx))
            match["ys"].append(float(cy))

    cubes = []
    for cluster in clusters:
        if len(cluster["ids"]) < min_hits:
            # Operator requirement: Cake priority rule. If a cluster contains Cake,
            # it must not be discarded even if detected in fewer frames than other
            # cubes.  Only detections above the per-class cake gates get here.
            if any(is_cake(c) for c in cluster["ids"]):
                pass
            else:
                continue
        # Cake priority rule: only for elevated top cubes (y < 215 px), cake detections take precedence.
        # Ground and pedestal clusters (y >= 215 px) use majority voting so banana pedestal isn't mislabeled.
        mean_y = sum(cluster["ys"]) / len(cluster["ys"])
        if mean_y < 215.0 and any(is_cake(c) for c in cluster["ids"]):
            cid = CAKE_ID
        else:
            cid = Counter(cluster["ids"]).most_common(1)[0][0]
        x = int(round(sum(cluster["xs"]) / len(cluster["xs"])))
        y = int(round(sum(cluster["ys"]) / len(cluster["ys"])))
        cubes.append([cid, [x, y, 0]])
    return cubes


def dedupe_boxes(boxes, dist_px=DEDUPE_DIST_PX):
    """Drop cross-model duplicates: same class, centres closer than ``dist_px``.

    ``boxes`` are dicts with ``name``, ``score``, ``cx``, ``cy`` (extra keys are
    preserved).  The highest-confidence box of each coincident group survives.
    """
    kept = []
    for box in sorted(boxes, key=lambda b: -float(b.get("score", 0.0))):
        duplicate = False
        for other in kept:
            if other.get("name") != box.get("name"):
                continue
            if math.hypot(box["cx"] - other["cx"], box["cy"] - other["cy"]) < dist_px:
                duplicate = True
                break
        if not duplicate:
            kept.append(box)
    return kept


def select_grasp_candidate(cubes, preferred_id=None, center_x=320.0, y_ref=260.0):
    """Pick the cube closest to the optical centre line (ahead of the robot).

    ``preferred_id`` keeps the originally targeted class after a re-scan, so a
    YOLO class flip cannot send a cube to the wrong zone mid-approach.
    """
    if not cubes:
        return None
    pool = [c for c in cubes if len(c) >= 2]
    if not pool:
        return None
    if preferred_id is not None:
        same_class = [c for c in pool if c[0] == preferred_id]
        if same_class:
            pool = same_class
    else:
        cake_candidates = [c for c in pool if is_cake(c[0])]
        if cake_candidates:
            pool = cake_candidates

    # If any two candidates are vertical duplicates of the same physical cube
    # (top face vs front/bottom face, separated by <= 90 px vertically and <= 45 px horizontally),
    # eliminate the bottom/front face (larger v) in favor of the top face (smaller v).
    filtered = []
    for c in pool:
        u, v = c[1][0], c[1][1]
        has_top_face = any(
            other != c and
            abs(other[1][0] - u) <= 45.0 and
            0 < (v - other[1][1]) <= 90.0
            for other in pool
        )
        if not has_top_face:
            filtered.append(c)

    candidates = filtered if filtered else pool
    return min(candidates, key=lambda c: abs(c[1][0] - center_x) + (y_ref - c[1][1]) * 1.2)

def choose_station(drop_zone, stations, preference=STATION_PREFERENCE):
    """Next station to harvest: nearest to ``drop_zone``, else first with picks."""
    for name in preference.get(drop_zone, ()):
        if stations.get(name, 0) > 0:
            return name
    for name in STATION_ORDER:
        if stations.get(name, 0) > 0:
            return name
    for name, left in stations.items():
        if left > 0:
            return name
    return None


def approach_waypoint(station, drop_zone):
    """Intermediate waypoint needed to reach ``station`` from ``drop_zone``."""
    return APPROACH_WAYPOINTS.get(drop_zone, {}).get(station)


def exit_maneuver(station, drop_zone):
    """``("back", metres)`` / ``("arrive", name)`` before driving to the zone."""
    return EXIT_MANEUVERS.get(drop_zone, {}).get(station)


def retire_station(stations, name):
    """Remove ``name`` from the work plan; True when it was still scheduled."""
    return stations.pop(name, None) is not None

START_COMMANDS = frozenset({"start", "1", "go", "on", "true", "begin", "run"})
STOP_COMMANDS = frozenset({"stop", "0", "halt", "off", "false", "abort", "kill"})
MATCH_DURATION_S = 600.0
DEFAULT_TIME_BUDGET_S = 540.0   # keep 60 s of the match for shutdown / scoring
TIME_WARN_S = 60.0


def parse_task_command(data):
    """Map a ``/task_start_flag`` payload to ``"start"`` / ``"stop"`` / ``None``.

    An unknown or empty payload returns None, which the caller must treat as
    "ignore".  This is deliberate: the previous code started the mission on ANY
    message, so the documented emergency-abort command could launch a run.
    """
    command = (data or "").strip().lower()
    if command in START_COMMANDS:
        return "start"
    if command in STOP_COMMANDS:
        return "stop"
    return None


class MatchClock(object):
    """Monotonic match timer with an injectable clock, so it is testable."""

    def __init__(self, limit_s=DEFAULT_TIME_BUDGET_S, warn_s=TIME_WARN_S,
                 time_fn=None):
        self.limit_s = float(limit_s)
        self.warn_s = float(warn_s)
        self._time_fn = time_fn or time.monotonic
        self._start = None

    def start(self):
        self._start = self._time_fn()
        return self

    def elapsed(self):
        if self._start is None:
            return 0.0
        return max(0.0, self._time_fn() - self._start)

    def remaining(self):
        return max(0.0, self.limit_s - self.elapsed())

    def expired(self):
        return self.remaining() <= 0.0

    def warn(self):
        return (not self.expired()) and self.remaining() <= self.warn_s

class CamAction:
    def __init__(self):
        # Load the competition item configuration.  A missing file is reported
        # but no longer fatal: the FSM works on canonical ids 1..4, and the old
        # sys.exit() killed the node halfway through startup.
        self.search_id = []
        try:
            package_path = rospkg.RosPack().get_path('auto_match')
            items_path = os.path.join(package_path, 'config', 'items_config.yaml')
            with open(items_path, "r", encoding="utf8") as f:
                items_content = yaml.load(f.read(), Loader=yaml.FullLoader)
            objects = items_content["objects"]
            self.search_id = [items_content["items"][objects[name]] for name in
                              ("objects_a", "objects_b", "objects_c", "objects_d")]
            rospy.loginfo(f">> [VISION] items_config.yaml loaded, item ids: {self.search_id}")
        except Exception as e:
            rospy.logwarn(f">> [VISION] items_config.yaml unavailable ({e}); "
                          f"working with canonical ids 1..4")

        # Rolling buffer of (monotonic timestamp, frame) for temporal voting.
        # A frame is a list of (raw_id, score, cx, cy) tuples.
        self._frames = deque(maxlen=max(2, GRASP_FRAMES))
        self._lock = threading.Lock()
        self.sub = rospy.Subscriber(
            "/objects", Detection2DArray, self._objects_cb, queue_size=5)

        # NOTE: the registered point cloud is ~10 MB/frame, so we do NOT keep
        # a 30 Hz subscriber (that ate a whole CPU core).  point3d() pulls one
        # frame only when a grasp actually needs it.
        self.transform3d = self._load_transform()

    def _load_transform(self):
        '''Load, sanity-check and describe the camera<-arm matrix.

        Sets ``self.calib_ok``.  When it is False the matrix cannot be trusted
        at all, so AutoAction refuses to start the match instead of missing
        every cube silently.  A matrix that is numerically fine but stale
        (the camera was moved) cannot be detected here - that is what the
        active pad check is for.
        '''
        path = os.path.expanduser("~/transform_3d.dat")
        source = path
        matrix = None
        try:
            matrix = np.load(path, allow_pickle=True)
        except Exception:
            matrix = None
        if matrix is None or getattr(matrix, "shape", None) != (4, 3):
            source = "built-in TRANSFORM_3D"
            matrix = TRANSFORM_3D
            rospy.logwarn(">> [CALIB] %s missing or not 4x3: using %s. "
                          "Run 'python3 -u ~/codes/calib3d.py'." % (path, source))

        # Fingerprint of the file on disk: the metadata sidecar must describe
        # THIS matrix, otherwise its check frame is meaningless.
        self.calib_md5 = None
        if source == path:
            try:
                with open(path, "rb") as handle:
                    self.calib_md5 = hashlib.md5(handle.read()).hexdigest()
            except Exception:
                self.calib_md5 = None

        ok, problems, stats = check_transform(matrix)
        self.calib_problems = list(problems)
        self.calib_ok = bool(ok)
        if ok:
            rospy.loginfo(">> [CALIB] matrix from %s: %.2f px/mm at the "
                          "workspace, round-trip %.1f mm"
                          % (source, stats.get("px_per_mm", 0.0),
                             stats.get("roundtrip_mm", 0.0)))
        else:
            for problem in problems:
                rospy.logerr(f">> [CALIB] {problem}")
            rospy.logerr(">> [CALIB] camera->arm matrix is unusable; run "
                         "'python3 -u ~/codes/calib3d.py'")

        meta = self._read_calibration_meta()
        self.calib_meta = meta
        if meta:
            rospy.loginfo(">> [CALIB] fitted %s from %s pose(s), leave-one-out "
                          "max %.1f px / mean %.1f px"
                          % (meta.get("created", "?"), meta.get("poses", "?"),
                             float(meta.get("loo_max_px", 0.0)),
                             float(meta.get("loo_mean_px", 0.0))))
        elif source == path:
            rospy.logwarn(">> [CALIB] no calibration metadata next to %s: "
                          "age and residuals unknown. Re-run calib3d.py after "
                          "moving the camera." % path)
        return matrix

    @staticmethod
    def _read_calibration_meta():
        '''Metadata written by calib3d.py next to the matrix, or None.'''
        path = os.path.expanduser("~/transform_3d.meta.json")
        try:
            with open(path, "r", encoding="utf8") as handle:
                return json.load(handle)
        except Exception:
            return None

    def _objects_cb(self, msg):
        '''Buffer one detection frame (possibly empty) for the voting window.'''
        frame = []
        for obj in msg.detections:
            try:
                result = obj.results[0]
                frame.append((result.id,
                              float(getattr(result, 'score', 1.0) or 0.0),
                              float(obj.bbox.center.x),
                              float(obj.bbox.center.y)))
            except Exception:
                continue
        with self._lock:
            self._frames.append((time.monotonic(), frame))

    def point3d(self, u, v, win=6):
        '''
        Median 3D point (metres, camera frame) around pixel (u, v).
        Pulls a single registered point cloud frame on demand; returns None
        when no valid depth is available (glossy/occluded surfaces).
        '''
        try:
            cloud = rospy.wait_for_message(
                "/camera/depth_registered/points", PointCloud2, timeout=3.0)
        except Exception:
            return None
        try:
            w, h = cloud.width, cloud.height
            u = int(round(u)); v = int(round(v))
            u0, u1 = max(0, u - win), min(w - 1, u + win)
            v0, v1 = max(0, v - win), min(h - 1, v + win)
            fo = {f.name: f.offset for f in cloud.fields}
            if not all(k in fo for k in ("x", "y", "z")):
                return None
            step = cloud.point_step
            row = cloud.row_step
            data = cloud.data
            vals = []
            for yy in range(v0, v1 + 1):
                base = yy * row
                for xx in range(u0, u1 + 1):
                    off = base + xx * step
                    x = struct.unpack_from("<f", data, off + fo["x"])[0]
                    y = struct.unpack_from("<f", data, off + fo["y"])[0]
                    z = struct.unpack_from("<f", data, off + fo["z"])[0]
                    if math.isfinite(x) and math.isfinite(y) and math.isfinite(z):
                        vals.append((x, y, z))
            if not vals:
                return None
            return [float(q) for q in np.median(np.array(vals), axis=0)]
        except Exception as e:
            rospy.logwarn_throttle(5.0, f">> [DEPTH] point3d failed: {e}")
            return None

    def point_to_arm(self, p):
        '''Convert a camera-frame point (metres) to uArm wrist coordinates (mm).'''
        try:
            B = self.transform3d
            M = B[:3, :].T
            b = B[3, :]
            arm = np.linalg.solve(M, np.array(p, dtype=float) - b)
            return float(arm[0]), float(arm[1]), float(arm[2])
        except Exception as e:
            rospy.logwarn_throttle(5.0, f">> [DEPTH] point_to_arm failed: {e}")
            return None

    def project_wrist(self, x, y, z):
        '''Project uArm wrist coordinates (mm) into the color image pixel.'''
        try:
            cam = np.array([x, y, z, 1.0], dtype=float) @ self.transform3d
            if cam[2] <= 1e-6:
                return None
            u = CAM_FX * cam[0] / cam[2] + CAM_CX
            v = CAM_FY * cam[1] / cam[2] + CAM_CY
            return float(u), float(v)
        except Exception:
            return None

    def pixel_to_arm_at_z(self, u, v, z, x0=200.0, y0=0.0):
        '''
        Wrist (x, y) whose suction-pad projection lands on pixel (u, v) at a
        known surface height z.  Uses the calibrated 3D transform and camera
        intrinsics, so it needs no depth and cannot be fooled by the arm
        occluding the cube.
        '''
        x, y = float(x0), float(y0)
        for _ in range(30):
            p = self.project_wrist(x, y, z)
            if p is None:
                return None
            ex, ey = u - p[0], v - p[1]
            if math.hypot(ex, ey) < 0.5:
                break
            px = self.project_wrist(x + 1.0, y, z)
            py = self.project_wrist(x, y + 1.0, z)
            if px is None or py is None:
                return None
            J = np.array([[px[0] - p[0], py[0] - p[0]],
                          [px[1] - p[1], py[1] - p[1]]], dtype=float)
            try:
                d = np.linalg.solve(J, np.array([ex, ey], dtype=float))
            except Exception:
                return None
            d = np.clip(d, -60.0, 60.0)
            x += float(d[0])
            y += float(d[1])
            x = min(max(x, 50.0), 350.0)
            y = min(max(y, -210.0), 210.0)
        return x, y

    def detector(self, num_samples=None, timeout=3.0, conf_threshold=None,
                 cake_conf_threshold=None, cake_site_conf_threshold=None):
        '''
        Majority-voted, spatially clustered cube list.

        Waits until ``num_samples`` detection frames have arrived *after this
        call* (so a re-scan after a base rotation never uses pre-rotation
        frames), then lets match_logic.cluster_detections() drop one-frame
        ghosts, merge the two YOLO models reporting the same cube, and vote on
        the class.

        Cake detections have to pass the per-class anti-glare gates
        ``cake_conf_threshold`` (raw id 2) and ``cake_site_conf_threshold``
        (raw id 20, the printed cake cube): the cake has FSM priority, so a
        low-confidence glare must never hijack the grasp.

        @return: list of [[canonical_id, [x, y, 0]], ...]
        '''
        wanted = max(1, int(GRASP_FRAMES if num_samples is None else num_samples))
        started = time.monotonic()
        deadline = started + max(0.0, float(timeout))
        frames = []

        while time.monotonic() < deadline:
            with self._lock:
                frames = [frame for stamp, frame in list(self._frames)
                          if stamp >= started]
            if len(frames) >= wanted:
                break
            rospy.sleep(0.05)

        if not frames:
            try:
                msg = rospy.wait_for_message(
                    "/objects", Detection2DArray, timeout=timeout)
            except Exception as e:
                rospy.logerr(f">> [DETECTOR] No message on /objects ({e}). "
                             f"Check detect_ros.py and the camera driver "
                             f"(dashboard option 1).")
                return []
            self._objects_cb(msg)
            with self._lock:
                frames = [frame for stamp, frame in list(self._frames)
                          if stamp >= started]

        frames = frames[-wanted:]
        thresh = CONF_THRESHOLD if conf_threshold is None else float(conf_threshold)
        cake_thresh = (match_logic.CAKE_CONF_THRESHOLD
                       if cake_conf_threshold is None
                       else float(cake_conf_threshold))
        cake_site_thresh = (match_logic.CAKE_SITE_CONF_THRESHOLD
                            if cake_site_conf_threshold is None
                            else float(cake_site_conf_threshold))
        cubes = match_logic.cluster_detections(
            frames, conf_threshold=thresh,
            cluster_dist_px=CLUSTER_DIST_PX,
            cake_conf_threshold=cake_thresh,
            cake_site_conf_threshold=cake_site_thresh)
        rospy.loginfo_throttle(
            1.0, f">> [VISION] {len(cubes)} cube(s) from {len(frames)} frame(s): {cubes}")
        return cubes

class ArmAction:

        self.cmd_vel_pub = rospy.Publisher("/cmd_vel", Twist, queue_size=10)
        rospy.sleep(0.2)

    def is_in_arm_limit(self, cx, cy, z):
        '''
        Check expanded physical limits of uArm Swift Pro:
        - Extension X: 50.0 .. 350.0 mm
        - Azimuth Y: -210.0 .. 210.0 mm
        - Reach radius R: up to 355.0 mm
        - Analytical inverse kinematics: ik_reachable(cx, cy, z)
        '''
        if not (50.0 <= cx <= 350.0):
            return False
        if not (-210.0 <= cy <= 210.0):
            return False
        if math.hypot(cx, cy) > 355.0:
            return False
        return ik_reachable(cx, cy, z)

    def target_from_pixel(self, u, v, kb_x, kb_y, expected_z, z_lo, z_hi):
        '''
        Convert a detected object pixel into a uArm wrist target (mm).

        1. Primary: registered RealSense 3D point + calibrated camera->arm
           transform.  The result is accepted only if its height is plausible
           for the layer being grasped; otherwise the depth hit the arm,
           the floor, or another occluder.
        2. Fallback: project the pixel onto the known layer height using the
           same 3D transform and camera intrinsics.  Needs no depth, so an
           occluding arm cannot make the arm drive upwards.
        3. Last resort: legacy 2D calibration.
        '''
        p = self.cam.point3d(u, v)
        tgt = self.cam.point_to_arm(p) if p is not None else None
        if tgt is not None and z_lo <= tgt[2] <= z_hi:
            return tgt[0], tgt[1], tgt[2], "3d"

        x0 = kb_x[0] * v + kb_x[1]
        y0 = kb_y[0] * u + kb_y[1]
        xy = self.cam.pixel_to_arm_at_z(u, v, expected_z, x0=x0, y0=y0)
        if xy is not None:
            return xy[0], xy[1], expected_z, "proj"
        return x0, y0, expected_z, "2d"

    def contact_heights(self, z_surface):
        '''
        Hover / touch / lift wrist heights around a measured object surface.

        The depth camera sees the TOP of the suction pad; the sealing rim is
        PAD_THICKNESS_MM lower.  So the wrist must stop PAD_THICKNESS_MM ABOVE
        the measured surface for the rim to meet it.  SAFE_MIN_Z prevents the
        arm from pushing into the floor (which makes the steppers skip).
        '''
        z_hover = z_surface + 45.0
        z_touch = max(z_surface + PAD_THICKNESS_MM, SAFE_MIN_Z)
        z_lift = min(z_surface + 100.0, 165.0)
        return z_hover, z_touch, z_lift

    def move_z_steps(self, x, y, z_from, z_to, steps=1, delay=0.8):
        '''
        Move vertically to target height. If steps <= 1, commands a single
        smooth G0 motion. Multiple steps are kept as an optional fallback.
        '''
        steps = max(1, int(steps))
        if steps <= 1:
            self.interface.set_pose(x, y, z_to, delay=delay)
            return
        for i in range(1, steps + 1):
            z = z_from + (z_to - z_from) * (float(i) / steps)
            self.interface.set_pose(x, y, z, delay=delay)

    def turn_base(self, angle_rad):
        '''
        Turn robot base by angle_rad (radians) in place.
        angle_rad > 0: CCW (left)
        angle_rad < 0: CW (right)
        '''
        if abs(angle_rad) < 0.025:  # Less than ~1.5 deg, no turn needed
            return

        # Fast responsive turn speed (0.75 rad/s)
        ang_speed = 0.75
        ang_vel = ang_speed if angle_rad > 0 else -ang_speed
        turn_time = max(0.08, min(abs(angle_rad) / ang_speed, 2.0))

        rospy.loginfo(f">> [VISUAL-SERVO] Fast base turn: {math.degrees(angle_rad):.1f} deg, duration {turn_time:.2f}s, vel={ang_vel:.2f} rad/s")
        tw = Twist()
        tw.angular.z = ang_vel

        t_end = rospy.Time.now() + rospy.Duration.from_sec(turn_time)
        rate = rospy.Rate(30)
        while rospy.Time.now() < t_end and not rospy.is_shutdown():
            self.cmd_vel_pub.publish(tw)
            rate.sleep()

        # Crisp active braking
        for _ in range(2):
            self.cmd_vel_pub.publish(Twist())
            rospy.sleep(0.02)
        rospy.sleep(0.10)

    def drive_base(self, dist_m):
        '''
        Move robot base forward or backward by dist_m meters.
        dist_m > 0: forward
        dist_m < 0: backward
        '''
        if abs(dist_m) < 0.03:  # Less than 3 cm, no drive needed
            return

        # Safe displacement clamp (max 25 cm per step)
        dist_m = max(-0.25, min(0.25, dist_m))
        lin_speed = 0.18  # Crisp 18 cm/s linear velocity
        lin_vel = lin_speed if dist_m > 0 else -lin_speed
        move_time = abs(dist_m) / lin_speed

        rospy.loginfo(f">> [VISUAL-SERVO] Base distance correction: {dist_m*100:.1f} cm, duration {move_time:.2f}s, vel={lin_vel:.2f} m/s")
        tw = Twist()
        tw.linear.x = lin_vel

        t_end = rospy.Time.now() + rospy.Duration.from_sec(move_time)
        rate = rospy.Rate(30)
        while rospy.Time.now() < t_end and not rospy.is_shutdown():
            self.cmd_vel_pub.publish(tw)
            rate.sleep()

        for _ in range(2):
            self.cmd_vel_pub.publish(Twist())
            rospy.sleep(0.02)
        rospy.sleep(0.10)

    # Grasp floor object (Layer 1)
    def grasp(self):
        '''
        Intelligent, high-precision grasp of Layer 1 cube:
        1. If cube is ALREADY in arm reach workspace (X: 170..315 mm, Y: -180..180 mm):
           Base DOES NOT move, robot does not center, but grasps cube immediately in place!
        2. If cube is out of workspace reach:
           - In-place base rotation towards cube (horizontal centering)
           - Forward drive ONLY if cube is too far away (> 295 mm)
           - CRITICAL: 0.20s pause and RESCAN after motion to recalculate exact cube coords!
        3. Grasp and lift:
           - Heights are computed from the cube's measured 3D position
             (works for floor cubes and raised tiers)
           - Enable pump and hold vacuum (0.40s) to build strong negative pressure
           - Confident vertical lift upwards (+100 mm)
        '''
        # 1. Primary cube detection
        cube_list = self.cam.detector(num_samples=6, timeout=3.0, conf_threshold=0.04)
        rospy.loginfo(f">> [GRASP] Cube detection: {cube_list}")

        if not cube_list:
            rospy.logwarn(">> [GRASP] Aborted: No target cubes detected in camera view.")
            self.grasp_status_pub.publish(String("1"))
            return 0

        target_cube = match_logic.select_grasp_candidate(cube_list)
        target_id = target_cube[0]
        if match_logic.is_cake(target_id):
            rospy.loginfo(f">> [GRASP] Target ID {target_id} is CAKE -> routing to grasp1 (Layer 2)!")
            return self.grasp1()
        initial_target_id = target_id
        cu = target_cube[1][0]
        cv = target_cube[1][1]

        # Geometry for floor cube: top=-50 mm, cube height=100 mm -> center=-100 mm
        z_center = -100.0
        z_touch = -50.0
        z_above = -15.0
        z_lift = 60.0

        xy = match_logic.pixel_to_xy_at_z(self.cam.transform3d, cu, cv, z_center)
        if xy is None:
            rospy.logerr(">> [GRASP] Degenerate projective mapping.")
            self.grasp_status_pub.publish(String("1"))
            return 0
        x = xy[0] + 20.0  # +20 mm cup center correction
        y = xy[1]

        # 2. Check: is cube in comfortable reach of the arm?
        # Comfortable sweet-spot envelope: arm is not strained, lands dead center on cube
        TARGET_X = 245.0
        dist_to_cube = math.hypot(x, y)
        in_reach = (160.0 <= x <= 265.0) and (abs(y) <= 85.0) and (dist_to_cube <= 270.0)
        rospy.loginfo(f">> [GRASP] Target ID {target_id}: px=({cu:.0f}, {cv:.0f}) -> x={x:.1f}, y={y:.1f}, dist={dist_to_cube:.1f} | in_reach={in_reach}")

        approach_step = 0
        while approach_step < 3 and not in_reach:
            approach_step += 1
            rospy.loginfo(f">> [GRASP] Cube outside sweet spot (x={x:.1f}, y={y:.1f}, dist={dist_to_cube:.1f}). Approach step #{approach_step}...")

            # (a) Turn base towards cube center
            angle_rad = math.atan2(y, x)
            if abs(angle_rad) > math.radians(2.0):
                rospy.loginfo(f">> [GRASP] Turning base towards cube: {math.degrees(angle_rad):.1f} deg...")
                self.turn_base(angle_rad)
                rospy.sleep(0.3)

            # (b) Drive forward to bring cube into sweet spot X = 245 mm
            drive_m = (dist_to_cube - TARGET_X) / 1000.0
            if abs(drive_m) >= 0.02:
                step_m = max(-0.25, min(0.25, drive_m))
                rospy.loginfo(f">> [GRASP] Driving {step_m*100:.1f} cm to position cube at 24.5 cm...")
                self.drive_base(step_m)
                rospy.sleep(0.4)

            # (c) THEN: re-run cube position scanning algorithm!
            rospy.loginfo(f">> [GRASP] Re-scanning cube position after approach step #{approach_step}...")
            rescan_list = self.cam.detector(timeout=3.0)
            if rescan_list:
                target_cube = match_logic.select_grasp_candidate(
                    rescan_list, preferred_id=initial_target_id)
                target_id = target_cube[0]
                cu = target_cube[1][0]
                cv = target_cube[1][1]
                xy = match_logic.pixel_to_xy_at_z(self.cam.transform3d, cu, cv, z_center)
                if xy is not None:
                    x = xy[0] + 20.0
                    y = xy[1]
                    dist_to_cube = math.hypot(x, y)
                    in_reach = (160.0 <= x <= 265.0) and (abs(y) <= 85.0) and (dist_to_cube <= 270.0)
                    rospy.loginfo(f">> [GRASP] Re-scanned target: px=({cu:.0f}, {cv:.0f}) -> x={x:.1f}, y={y:.1f}, dist={dist_to_cube:.1f} | in_reach={in_reach}")
            else:
                break

        # 4. Execute physical grasp
        x = max(120.0, min(320.0, x))
        y = max(-190.0, min(190.0, y))
        rospy.loginfo(f">> [GRASP] Executing grasp for ID {target_id}: x={x:.1f}, y={y:.1f}, z_touch={z_touch:.1f}, z_lift={z_lift:.1f}")

        self.interface.set_status(True)
        rospy.sleep(0.20)

        # Move above cube at safe height
        self.interface.set_pose(x, y, z_above, delay=0.9)

        # Enable vacuum pump early for fast suction
        self.interface.set_pump(True)
        rospy.sleep(0.10)

        # Single smooth descent straight down to the cube surface
        self.interface.set_pose(x, y, z_touch, delay=0.8)

        # Vacuum dwell time for reliable suction seal
        rospy.sleep(0.40)

        # Single smooth vertical lift
        self.interface.set_pose(x, y, z_lift, delay=0.8)
        print("Cube lifted successfully")
        rospy.loginfo(f">> [GRASP] Cube ID {target_id} lifted successfully to height z={z_lift}")

        self.grasp_status_pub.publish(String("0"))
        return target_id

                f"{target_id}. Rule-critical: verify CAKE_SORTING_STATION "

        # Geometry for Layer 2 Cake: sits on Layer 1 (top at -50 mm).
        # Cake top surface is at z_touch ~ 40.0 mm (firm suction contact with 20 cm tier)
        z_center = 0.0
        z_touch = 40.0
        z_safe = 75.0    # Clears any Layer 1 front-row obstacles safely!
        z_lift = 85.0    # Vertical separation lift (+45 mm off pedestal, theta2=9.5 deg)
        z_retract = 120.0
        x_retract = 270.0

        # Dedicated approximation for Cake on Tier 2 (thefile2.txt +3cm forward):
        cake_xy = match_logic.pixel_to_cake_xy(self.cam.transform3d, u, v, cup_offset_x=30.0)
        if cake_xy is None:
            rospy.logerr(">> [GRASP1] Degenerate cake approximation.")
            self.grasp_status_pub.publish(String("1"))
            return 0
        x, y = cake_xy

        # CRITICAL: For Cake positioned behind front-row obstacles,
        # base motion is strictly inhibited to avoid colliding with front-row cubes.
        # The manipulator reaches over the obstacle row from the stationary position.
        in_reach = True
        dist_to_cake = math.hypot(x, y)
        rospy.loginfo(f">> [GRASP1] Target ID {target_id}: u={u:.0f}, v={v:.0f} -> x={x:.1f}, y={y:.1f}, dist={dist_to_cake:.1f} | obstacle-guard in_reach=True")

        # Safe reach limits expanded specifically for Cake up to 350.0 mm
        x = max(120.0, min(350.0, x))
        y = max(-190.0, min(190.0, y))
        rospy.loginfo(f">> [GRASP1] Executing grasp for Cake ID {target_id}: x={x:.1f}, y={y:.1f}, z_touch={z_touch:.1f}, z_lift={z_lift:.1f}")

        self.interface.set_status(True)
        rospy.sleep(0.20)

        # Move above cake at safe height (z_safe=75 mm clears any ground cube in front)
        self.interface.set_pose(130.0, 0.0, z_safe, delay=0.5)
        self.interface.set_pose(x, y, z_safe, delay=0.9)

        # Activate vacuum suction before descent so it draws immediately upon contact
        self.interface.set_pump(True)
        rospy.sleep(0.30)
        self.interface.set_pose(x, y, z_touch, delay=0.9)
        rospy.sleep(0.60)

        print("Layer 2 Cake lifted successfully")
        # Stage 1: Vertical separation off pedestal
        self.interface.set_pose(x, y, z_lift, delay=0.8)
        rospy.sleep(0.50)
        # Stage 2: Pull cake inward to secure position and reduce torque
        self.interface.set_pose(x_retract, y, z_retract, delay=0.8)
        rospy.sleep(0.50)

        rospy.loginfo(f">> [GRASP1] Layer 2 Cake ID {target_id} lifted and secured successfully to height z={z_retract}")

        self.grasp_status_pub.publish(String("0"))
        return target_id
    
    
    def drop(self, check=False):
        '''
        放置方块, 可以先判断是否有方块, 从而调整放置高度
        @param check: 是否判断有无方块, 默认判断
        @return item_id: 执行结果
        '''
        x = 235
        y = 0
        z = -55

        # Lower cube to target drop point
        self.interface.set_pose(x, y, z, delay=0.40)

        # Disable vacuum pump and wait for vacuum relief
        self.interface.set_pump(0)
        rospy.sleep(0.25)

        # SAFE VERTICAL RETRACT: lift suction cup straight up +80 mm,
        # to prevent friction or lateral motion from disturbing the placed cube:
        lift_z = min(z + 80, 160)
        self.interface.set_pose(x, y, lift_z, delay=0.35)

        # Retract to service pose (50, 180, 160):
        self.arm_grasp_ready(block=False)

        self.grasp_status_pub.publish(String("0"))
        return True

    def _tower_xy(self, x_kb, y_kb, fallback=(235.0, 0.0)):
        '''
        (x, y) of the cube already standing in the drop zone, used to align the
        stacking tower.  Falls back to the calibrated default point when the
        camera sees nothing (SI-12: never index an empty detection list).
        '''
        cube = match_logic.select_grasp_candidate(self.cam.detector(timeout=3.0))
        if cube is None:
            rospy.logwarn(">> [DROP] No cube visible in the zone; "
                          "stacking on the default point.")
            return fallback
        u, v = cube[1][0], cube[1][1]
        # row (v) drives arm x, column (u) drives arm y - see arm_ik.py.
        return match_logic.pixel_to_arm(v, u, x_kb, y_kb)

    def arm_grasp_laser(self, block=False):
        self.interface.set_pose(160, 0, 20, delay=0.35 if block else 0.20)

    #: Pose used by the active calibration check: 270 mm in front, level,
    #: 80 mm above the top of a floor cube, so a cube standing anywhere in the
    #: workspace cannot be hit while the arm travels there.  Kept inside the
    #: z range covered by the fit (>= -60 mm) so the check is not extrapolated.
    CALIB_CHECK_POSE = (270.0, 0.0, 30.0)

    def measure_pad(self, pose=None):
        '''
        Move to ``pose`` and return the median orange-pad pixel, or None.

        A partially occluded pad still produces a centroid, just a biased one,
        so only full-size blobs count and a few frames are medianed instead of
        trusting the first reading.
        '''
        pose = self.CALIB_CHECK_POSE if pose is None else pose
        try:
            from cv_bridge import CvBridge
            from pad_detect import grab, pad_pixel
        except Exception as exc:
            rospy.logerr(">> [CALIB] cannot load the pad detector: %s" % exc)
            return None

        bridge = CvBridge()
        # NOTE: no M17 here.  Re-attaching the motors before every move made
        # the uArm miss steps (~15 mm low), which looked exactly like a moved
        # camera.  ArmAction.__init__ attaches them once; leave them attached.
        self.interface.set_pose(float(pose[0]), float(pose[1]), float(pose[2]),
                                delay=1.4)
        rospy.sleep(0.3)
        picks = []
        for _ in range(5):
            try:
                found = pad_pixel(grab(bridge))
            except Exception:
                found = None
            if found and found[2] >= PAD_MIN_AREA:
                picks.append((found[0], found[1]))
            if len(picks) >= 3:
                break
            rospy.sleep(0.1)
        if not picks:
            return None
        return (float(np.median([pick[0] for pick in picks])),
                float(np.median([pick[1] for pick in picks])))

    def verify_calibration(self, pose=None, tol_px=6.0):
        '''
        Check that the camera and the arm still agree with the calibration.

        Moves to a known pose, finds the pad and compares the measured pixel
        with the pixel recorded during calibration (metadata sidecar).  That
        reference is a real measurement, not a model prediction, so the check
        is not confused by the ~2-3 px residual of the fit itself; a camera
        that was moved shows up as a large jump.

        The uArm steppers occasionally miss steps on one fast move, which
        lands the pad ~15 mm low, so a failed measurement is retried once.  A
        camera that really moved fails both attempts.

        @return: (ok, detail).  A pad that is not visible returns ok=True with
                 an explanatory detail: a lighting problem must not block the
                 match, it only means the check could not run.
        '''
        pose = self.CALIB_CHECK_POSE if pose is None else pose
        if self.cam is None:
            return True, "no camera, check skipped"

        best = None
        for attempt in (1, 2):
            uv = self.measure_pad(pose)
            if uv is None:
                if attempt == 1:
                    rospy.logwarn(">> [CALIB] pad not visible at the check "
                                  "pose, retrying once")
                    self.arm_home(block=True)
                    rospy.sleep(1.2)
                    continue
                return True, ("pose=(%.0f, %.0f, %.0f), but the pad is not "
                              "visible/full-size - check skipped" % pose)
            error, detail = self._pad_error(pose, uv)
            if error is None:
                return True, detail
            if best is None or error < best[0]:
                best = (error, detail)
            if error <= tol_px:
                break
            if attempt == 1:
                rospy.logwarn(">> [CALIB] pad check %.1f px off, retrying once"
                              % error)
                self.arm_home(block=True)
                rospy.sleep(1.2)
        return (best[0] <= tol_px), best[1]

    def _pad_error(self, pose, uv):
        '''Pixel distance between ``uv`` and the calibration reference.'''
        detail = "pose=(%.0f, %.0f, %.0f), pad at (%.1f, %.1f)" % (
            pose[0], pose[1], pose[2], uv[0], uv[1])
        reference = self._calibration_reference(pose)
        if reference is not None:
            error = math.hypot(uv[0] - reference[0], uv[1] - reference[1])
            detail += ", calibration frame (%.1f, %.1f)" % reference
            source = "pad moved"
        else:
            error = pad_projection_error(self.cam.transform3d, pose, uv)
            if error is None:
                return None, detail + " - no metadata and pose not projectable"
            source = "projection differs"
        detail += ": %s by %.1f px" % (source, error)
        return error, detail

    def _calibration_reference(self, pose):
        '''Pad pixel recorded by calib3d.py for ``pose``, if it matches.'''
        meta = getattr(self.cam, "calib_meta", None)
        if not meta:
            return None
        stored = meta.get("file_md5")
        current = getattr(self.cam, "calib_md5", None)
        if stored and current and stored != current:
            rospy.logwarn(">> [CALIB] metadata belongs to a different matrix "
                          "(stale sidecar); ignoring the stored check frame")
            return None
        check = meta.get("check") or {}
        saved_pose = check.get("pose")
        uv = check.get("uv")
        if not saved_pose or not uv or len(saved_pose) != 3 or len(uv) != 2:
            return None
        if max(abs(float(a) - float(b)) for a, b in zip(saved_pose, pose)) > 1.0:
            return None
        return (float(uv[0]), float(uv[1]))

            self.distance_srv = rospy.ServiceProxy(
                'get_distance', common.srv.GetFrontBackDistance)
        except Exception:
            rospy.logwarn("/get_distance service unavailable, skipping")
            self.distance_srv = None

        # 创建导航地点的话题发布者
        self.goto_local_pub = rospy.Publisher(
            "mark_nav", String, queue_size=1)


    def goto_local(self, name):
        '''
        根据目标点名称,发布目标位置到MoveBase服务器,根据返回状态进行判断
        @return: True 为成功到达, False 为失败
        '''

        # 发布目标位置
        self.goto_local_pub.publish("go "+name)
        rospy.loginfo(f"已发送go指令至: {name}")

        # 设定1分钟的时间限制，进行阻塞等待
        try:
            ret_status = rospy.wait_for_message(
                'move_base/result', MoveBaseActionResult, rospy.Duration(60)).status.status
        except Exception:
            rospy.logwarn("nav timeout!!!")
            ret_status = GoalStatus.ABORTED

        # 如果一分钟之内没有到达，放弃目标
        if ret_status != GoalStatus.SUCCEEDED:
            rospy.Publisher("move_base/cancel", GoalID, queue_size=1).publish(
                GoalID(stamp=rospy.Time.from_sec(0.0), id=""))
            try:
                rospy.wait_for_message(
                    'move_base/result', MoveBaseActionResult, rospy.Duration(3))
            except Exception:
                rospy.logwarn("move_base result timeout. this is abnormal.")
            rospy.loginfo("==========Timed out achieving goal==========")
            return False
        else:
            rospy.loginfo("==========Goal succeeded==========")
            return True

        )
        return True


class AutoAction:
    def __init__(self):
        rospy.init_node('spark_auto_match_node', anonymous=True)

        print("========ready to task===== ")

        # Build the three subsystems one by one and record failures instead of
        # swallowing them: a half-built node used to crash later with a bare
        # AttributeError, which is impossible to diagnose on the field.
        self.cam = None
        self.arm = None
        self.robot = None
        self.init_errors = []

        try:
            self.cam = CamAction()
            print("========实例化Cam===== ")
        except Exception as e:
            self.init_errors.append(f"camera: {e}")
            rospy.logerr(f">> [INIT] CamAction failed: {e}")

        try:
            if self.cam is None:
                raise RuntimeError("camera subsystem unavailable")
            self.arm = ArmAction(cam=self.cam)
            print("========实例化Arm===== ")
        except Exception as e:
            self.init_errors.append(f"arm: {e}")
            rospy.logerr(f">> [INIT] ArmAction failed: {e}")

        try:
            self.robot = RobotMoveAction()
            print("========实例化Robot===== ")
        except Exception as e:
            self.init_errors.append(f"robot: {e}")
            rospy.logerr(f">> [INIT] RobotMoveAction failed: {e}")

        # A matrix that cannot even be read/parsed is a hard stop: every grasp
        # would miss by tens of millimetres with no error anywhere.
        if self.cam is not None and not getattr(self.cam, "calib_ok", True):
            self.init_errors.append("calibration: camera->arm matrix failed "
                                    "the sanity check (run calib3d.py)")

        # Active check: runs ONLY ONCE after the arm driver starts.
        # Once verified, /tmp/arm_calib_done persists so subsequent launches
        # do not move the manipulator unnecessarily before every run.
        calib_flag = "/tmp/arm_calib_done"
        if (self.arm is not None and not self.init_errors
                and os.environ.get("SPARK_CALIB_CHECK", "1") != "0"
                and not os.path.exists(calib_flag)):
            try:
                calib_ok, calib_detail = self.arm.verify_calibration()
            except Exception as e:
                calib_ok, calib_detail = True, "check crashed: %s" % e
            if calib_ok:
                rospy.loginfo(f">> [CALIB] active pad check: {calib_detail}")
                try:
                    with open(calib_flag, "w") as f:
                        f.write(time.strftime("%Y-%m-%dT%H:%M:%S\n"))
                except Exception:
                    pass
            else:
                rospy.logerr(f">> [CALIB] active pad check FAILED: {calib_detail}")
                self.init_errors.append(
                    f"calibration: pad check failed ({calib_detail})")
            try:
                self.arm.arm_home(block=True)
            except Exception as e:
                rospy.logwarn(f">> [CALIB] could not home the arm: {e}")
        elif os.path.exists(calib_flag):
            rospy.loginfo(">> [CALIB] Manipulator already calibrated once after driver startup, skipping pad check.")
        elif self.arm is not None and not self.init_errors:
            rospy.logwarn(">> [CALIB] active pad check disabled by "
                          "SPARK_CALIB_CHECK=0: a moved camera will not be "
                          "detected at startup")

        #: True only when the whole chain is up; the FSM refuses to start otherwise.
        self.ready = not self.init_errors

        # Item ID mapping (rule-compliant normalization lives in match_logic).
        self.BANANA_ID = match_logic.BANANA_ID
        self.TEDDY_ID = match_logic.TEDDY_ID
        self.CLOCK_ID = match_logic.CLOCK_ID
        self.CAKE_ID = CAKE_ID
        self.CUP_ID = match_logic.CAKE_ID  # alias for backwards compatibility

        # Zone mapping; always defined so no code path can hit AttributeError.
        self.mapping = expand_aliases(dict(DEFAULT_MAPPING))

        # 订阅任务控制指令的话题
        self.task_cmd_sub = rospy.Subscriber("/task_start_flag", String, self.task_cmd_cb)
        self.task_run_th = threading.Thread(target=lambda: "pass")
        self.stop_flag = False  # 任务的启停标志

        # Subscribe to manual arm control (/grasp)
        self.grasp_sub = rospy.Subscriber("/grasp", String, self.grasp_cb)

        rospy.loginfo("==============================================")
        if self.ready:
            rospy.loginfo("  spark_auto_match_node is ready!             ")
            rospy.loginfo("  /task_start_flag start|stop   /grasp 1|0     ")
        else:
            rospy.logerr("  spark_auto_match_node is NOT ready:          ")
            for problem in self.init_errors:
                rospy.logerr(f"    - {problem}")
            rospy.logerr("  Fix the subsystems above; the match will not start.")
        rospy.loginfo("==============================================")

    # ------------------------------------------------------------------
    # Mission helpers
    # ------------------------------------------------------------------
    def _abort_requested(self, clock=None):
        """True when the operator stopped the run or the match budget is gone."""
        if self.stop_flag:
            rospy.logwarn(">> [TASK] stop requested by the operator")
            return True
        if clock is not None:
            if clock.warn():
                rospy.logwarn(f">> [TASK] only {clock.remaining():.0f} s of the "
                              f"time budget left")
            if clock.expired():
                rospy.logerr(f">> [TASK] time budget exhausted after "
                             f"{clock.elapsed():.0f} s; stopping before overtime")
                return True
        return False

    def _navigate(self, name):
        """goto_local with one retry and a diagnostic hint when nav is absent."""
        if self.robot.goto_local(name):
            return True
        rospy.logwarn(f">> [TASK] navigation to {name} failed, retrying once")
        if self.robot.goto_local(name):
            return True
        rospy.logerr(f">> [TASK] navigation to {name} failed twice. Is the "
                     f"navigation stack running? start_system.sh only launches "
                     f"the drivers; /move_base and mark_nav must come from the "
                     f"arena navigation launch file (Phase 2).")
        return False

    def _execute_exit_maneuver(self, station, drop_zone):
        """Leave the station without dragging the cube through the centre board."""
        maneuver = match_logic.exit_maneuver(station, drop_zone)
        if not maneuver:
            return
        kind, value = maneuver
        if kind == "back":
            self.robot.step_back(value)
            rospy.sleep(0.6)
        elif kind == "arrive":
            self.robot.arriveto_local(value)
            rospy.sleep(0.6)

    def _deliver(self, item_id, station, drop_times, force_zone=None):
        """Drive to the mapped zone and stack the held cube; returns the zone."""
        zone = force_zone
        if zone is None or drop_times.get(zone, 0) >= MAX_TIER:
            zone = match_logic.choose_drop_zone(item_id, self.mapping, drop_times)
        if zone is None:
            rospy.logerr(">> [DROP] every zone already holds a 3-cube tower; "
                         "releasing the cube here so the arm is free")
            self.arm.drop()
            return None

        self._execute_exit_maneuver(station, zone)
        self.arm.arm_grasp_ready()

        if not self._navigate(zone):
            rospy.logerr(f">> [DROP] releasing the cube at the current position "
                         f"instead of {zone}")
            self.arm.drop()
            return None

        rospy.sleep(SETTLE_S)
        routine = match_logic.plan_drop(drop_times.get(zone, 0)) or "drop"
        getattr(self.arm, routine)()
        drop_times[zone] = drop_times.get(zone, 0) + 1
        rospy.loginfo(f">> [DROP] item {item_id} -> {zone}, "
                      f"tier {drop_times[zone]} via {routine}()")
        return zone

    def _finish(self, clock, drop_times, total_drop_times, cake_delivered):
        """Park the arm and print a match summary that can be scored by hand."""
        self.arm.arm_home()
        estimate = match_logic.estimate_score(
            drop_times, red_zones=match_logic.RED_ZONES, started_from_base=True)
        rospy.logwarn("***** task finished *****")
        rospy.logwarn(f"  cubes delivered: {total_drop_times} / {TOTAL_CUBES}")
        rospy.logwarn(f"  tiers per zone:  {drop_times}")
        rospy.logwarn(f"  estimated score: {estimate}")
        if not cake_delivered:
            rospy.logwarn("  NOTE: the cake-first rule was not satisfied, so the "
                          "real score is lower than this estimate.")
        rospy.logwarn(f"  elapsed: {clock.elapsed():.0f} s of a "
                      f"{clock.limit_s:.0f} s budget")
        rospy.logwarn("Re-send 'start' on /task_start_flag to run again, or Ctrl+C.")

    # 接收到启动自动赛信号，开始执行任务
    def run_task(self):
        """Full 10-cube mission: cake first, then board-driven sorting."""
        if not self.ready:
            rospy.logerr(">> [TASK] refusing to start: " + "; ".join(self.init_errors))
            return

        clock = MatchClock(limit_s=TASK_TIME_BUDGET_S,
                           warn_s=match_logic.TIME_WARN_S).start()
        self.stop_flag = False
        self.mapping = expand_aliases(dict(DEFAULT_MAPPING))

        stations = {name: STATION_PICKS for name in match_logic.STATION_ORDER}
        drop_times = {zone: 0 for zone in ("Collection_A",) + ZONES_BCD}
        empty_attempts = {name: 0 for name in match_logic.STATION_ORDER}
        total_drop_times = 0
        cake_delivered = False
        last_zone = "Collection_A"

        rospy.loginfo("start task now.")
        self.arm.arm_grasp_ready()
        # ==== 离开起始区,避免在膨胀区域中，导致导航失败 =====
        self.robot.step_go(0.4)
        if self._abort_requested():
            return
        # Motors stay continuously attached from driver bringup (M17); avoid motor reset slip

        # ===== classification board (SI-14: fall back instead of aborting) =====
        if self._navigate("Classification_area"):
            rospy.sleep(0.2)
            self.mapping = self.detect_classification_board()
        else:
            rospy.logerr(">> [TASK] board area unreachable: running with the "
                         "DEFAULT mapping (cake->A, bear->B, clock->C, banana->D)")

        # ===== cake rush: the first cube of the match MUST be the cake =====
        if not self._abort_requested(clock):
            rospy.loginfo(f">> [RULES] first delivery from {CAKE_SORTING_STATION} "
                          f"goes to Collection_A")
            if self._navigate(CAKE_SORTING_STATION):
                rospy.sleep(SETTLE_S)
                first_id = self.arm.grasp1()
                if first_id == 0:
                    rospy.logerr(">> [RULES] no cube seen at the cake station: "
                                 "the cake-first rule cannot be guaranteed")
                else:
                    cake_delivered = match_logic.is_cake(first_id)
                    if not cake_delivered:
                        rospy.logerr(f">> [RULES] first grasp was item {first_id}, "
                                     f"not the cake: cake points are forfeited")
                    if self._deliver(first_id, CAKE_SORTING_STATION, drop_times,
                                     force_zone="Collection_A"):
                        total_drop_times += 1
                        last_zone = "Collection_A"
                        stations[CAKE_SORTING_STATION] = max(
                            0, stations.get(CAKE_SORTING_STATION, 1) - 1)
                        if stations.get(CAKE_SORTING_STATION, 0) <= 0:
                            match_logic.retire_station(stations,
                                                       CAKE_SORTING_STATION)
            else:
                rospy.logerr(f">> [RULES] cake station {CAKE_SORTING_STATION} "
                             f"unreachable; continuing with the sorting loop")

        # ===== sorting loop =====
        while not self._abort_requested(clock):
            if total_drop_times >= TOTAL_CUBES:
                break
            if not stations:
                rospy.logwarn(">> [TASK] every station is retired; finishing early")
                break

            station = match_logic.choose_station(last_zone, stations)
            if station is None:
                break
            rospy.loginfo(f">> [TASK] cycle {total_drop_times + 1}/{TOTAL_CUBES}: "
                          f"station {station} "
                          f"({stations.get(station, 0)} pick(s) left)")

            waypoint = match_logic.approach_waypoint(station, last_zone)
            if waypoint:
                self.robot.arriveto_local(waypoint)

            if not self._navigate(station):
                match_logic.retire_station(stations, station)
                rospy.logerr(f">> [TASK] {station} unreachable; retired")
                continue

            rospy.sleep(SETTLE_S)
            # Floor routine everywhere in the loop: the tier-2 cake is handled
            # once, up front, by the cake rush.
            item_id = self.arm.grasp()
            if item_id == 0:
                empty_attempts[station] = empty_attempts.get(station, 0) + 1
                rospy.logwarn(f">> [TASK] nothing grasped at {station} "
                              f"({empty_attempts[station]}/{MAX_EMPTY_ATTEMPTS})")
                if empty_attempts[station] >= MAX_EMPTY_ATTEMPTS:
                    match_logic.retire_station(stations, station)
                    rospy.logerr(f">> [TASK] {station} retired after "
                                 f"{MAX_EMPTY_ATTEMPTS} empty scans")
                continue

            if self._abort_requested(clock):
                rospy.logwarn(">> [TASK] abort while holding a cube: parking it here")
                self.arm.drop()
                break

            delivered_zone = self._deliver(item_id, station, drop_times)
            if delivered_zone:
                last_zone = delivered_zone
            total_drop_times += 1
            empty_attempts[station] = 0
            stations[station] = max(0, stations.get(station, 1) - 1)
            if stations.get(station, 0) <= 0:
                match_logic.retire_station(stations, station)

        self._finish(clock, drop_times, total_drop_times, cake_delivered)

    def detect_classification_board(self):
        '''
        Read the three-slot classification board and build the zone mapping.

        Slots are voted over up to BOARD_FRAMES frames (left to right in image
        order).  The vote result goes through
        match_logic.decode_classification_board(), which keeps the DEFAULT
        mapping when the board was not read cleanly (SI-14: never leave the
        mission without a mapping).
        '''
        rospy.loginfo(">> [BOARD] reading the classification board ...")
        slot_votes = {0: [], 1: [], 2: []}
        collected = 0
        deadline = rospy.Time.now() + rospy.Duration(6.0)
        slots = len(ZONES_BCD)

        while (not rospy.is_shutdown() and collected < BOARD_FRAMES
               and rospy.Time.now() < deadline):
            try:
                obj_array = rospy.wait_for_message(
                    "/objects", Detection2DArray, timeout=0.5)
            except Exception:
                rospy.logwarn_throttle(2.0, ">> [BOARD] no /objects message, waiting ...")
                continue

            objs = []
            for obj in obj_array.detections:
                try:
                    result = obj.results[0]
                    item_id = norm_id(result.id)
                    score = float(getattr(result, 'score', 1.0) or 0.0)
                except Exception:
                    continue
                # The board carries only the bear/clock/banana cards.
                if item_id not in match_logic.BOARD_ITEMS or score < CONF_THRESHOLD:
                    continue
                objs.append((item_id, obj.bbox.center.x))

            if len(objs) < slots:
                continue

            objs.sort(key=lambda pair: pair[1])
            for slot in range(slots):
                slot_votes[slot].append(objs[slot][0])
            collected += 1

        winners = []
        for slot in range(slots):
            votes = slot_votes[slot]
            winners.append(Counter(votes).most_common(1)[0][0] if votes else None)

        mapping = decode_classification_board(winners)
        if match_logic.board_is_confident(winners):
            rospy.loginfo(f">> [BOARD] {collected} frame(s) voted; mapping = {mapping}")
        else:
            rospy.logerr(f">> [BOARD] board not read cleanly (winners={winners}, "
                         f"{collected} frame(s)); using DEFAULT mapping {DEFAULT_MAPPING}")
        return mapping

    def task_cmd_cb(self, msg):
        '''Start/stop the mission from /task_start_flag, honouring the payload.'''
        command = parse_task_command(getattr(msg, 'data', ''))
        if command == "start":
            if not self.ready:
                rospy.logerr(">> [TASK] cannot start: " + "; ".join(self.init_errors))
                return
            if self.task_run_th.is_alive():
                rospy.logwarn(">> [TASK] a mission is already running; send 'stop' first")
                return
            self.stop_flag = False
            self.task_run_th = threading.Thread(
                target=self.run_task, args=(), daemon=True)
            self.task_run_th.start()
            rospy.loginfo(">> [TASK] mission started by /task_start_flag")
        elif command == "stop":
            if self.task_run_th.is_alive():
                self.stop_flag = True
                rospy.logwarn(">> [TASK] stop requested; the mission exits at its "
                              "next checkpoint (the thread is not joined because it "
                              "may be mid-motion)")
            else:
                rospy.loginfo(">> [TASK] nothing to stop")
        else:
            rospy.logwarn(f">> [TASK] ignoring unknown /task_start_flag payload "
                          f"'{getattr(msg, 'data', '')}' (use 'start' or 'stop')")

    def grasp_cb(self, msg):
        rospy.loginfo(f">> [GRASP_CB] Received message on /grasp: '{msg.data}'")
        if not self.ready:
            rospy.logerr(">> [GRASP_CB] manual grasp unavailable: "
                         + "; ".join(self.init_errors))
            return
        if self.task_run_th.is_alive():
            rospy.logwarn("Match task thread already running; /grasp command ignored")
            return
        if msg.data == "1":
            rospy.loginfo(">> [GRASP_CB] Starting cube grasp (arm.grasp)...")
            self.arm.grasp()
        elif msg.data == "0":
            rospy.loginfo(">> [GRASP_CB] Releasing cube (arm.drop)...")
            self.arm.drop()
            self.arm.arm_grasp_ready()
        else:
            rospy.logwarn(f"Unknown grasp command: '{msg.data}'")

    except rospy.ROSInterruptException:

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Cube grasp -> raise -> 5 s hold -> put back -> park, field test (NXROBO Spark).

FIELD-TEST SEQUENCE (exactly the operator-requested algorithm)
---------------------------------------------------------------
1. Lower the manipulator: safe transit above the detected item at the
   clearance height.
2. Suction: vacuum ON, then a strictly vertical descent to the item surface
   and a dwell at the contact height to build vacuum.
3. Raise the cube STRICTLY VERTICALLY (X and Y never change while holding it).
4. Hold the cube in the raised position for ``hold_seconds`` (5 s by default).
5. Put the cube back at the SAME X/Y (strictly vertical descent) and release
   the pump only after the cube is resting on the ground.
6. Retract the arm vertically and park it in its ORIGINAL pose.

There is deliberately NO camera snapshot after the lift and NO success check
after the put-back.

TARGET ACQUISITION: A CUBE OUT OF THE ARM'S REACH
-------------------------------------------------
A cube that is visible but outside the manipulator's workspace is NOT an
error.  The scan reports it as a far-field target (``far=True``), the base
aligns its heading and drives one bounded step towards it, and the next scan -
now taken from the calibrated near field - produces the exact grasp
coordinates.  ``MAX_APPROACH_ATTEMPTS`` bounds the loop; a target that still
cannot be reached ends with ``OUT_OF_REACH`` (exit code 6).  Before this was
implemented, any cube beyond ~35 cm was filtered out of the scan, so the
robot answered ``NO_TARGET`` (exit code 3) without moving.

The script installs its own SIGINT/SIGTERM handler (``disable_signals=True``):
rospy's default handler calls ``signal_shutdown()`` first (after that every
publish is silently dropped) and then swallows ``KeyboardInterrupt``, so the
abort path below would never run.  With the custom handler Ctrl-C raises
inside the sequence while the node is still alive: the abort path lowers the
cube back to the pickup point, releases the pump only at contact and then
parks the arm.  That is what prevents the "robot dropped the cube instead of
returning it" failures (a false camera check after the lift used to abort the
put-back and switch the pump off in mid-air).

INTEGRATION INTO THE MAIN PROGRAM
---------------------------------
The reusable API is free of camera/field-test scaffolding:

    plan = plan_grasp(target, cake_touch_z)      # wrist heights from a detection
    execute_sequence(arm, plan,
                     hold_seconds=5.0,
                     return_cube=False)           # steps 1-4 only
    # returns with the cube attached, vacuum still ON, wrist at plan.z_lift;
    # the main program now carries the cube to its destination zone.

Steps 5-6 (put the cube back at the pickup point + park the arm) are marked
with a banner in ``execute_sequence`` and are deleted for the final
integration (or skipped with ``return_cube=False``).

Run on the robot through ``run_grasp_test.sh``:

    python3 -u ~/codes/grasp_generic.py               # full test incl. put-back
    python3 -u ~/codes/grasp_generic.py --no-place    # steps 1-4 only (preview)
    python3 -u ~/codes/grasp_generic.py --scan-only   # perception/assessment only
"""

def _sleep(seconds, raw=False):
    """Wait ``seconds``: ROS time normally, wall time in the abort path.

    ``raw=True`` uses plain ``time.sleep`` so the abort path keeps working if
    rospy is already shutting down (``rospy.sleep`` raises immediately then).
    """
    if seconds is None or seconds <= 0:
        return
    if raw:
        time.sleep(seconds)
        return
    rospy.sleep(seconds)
    if rospy.is_shutdown():
        # Nobody should shut rospy down behind our back, but if it happens the
        # publishers are already dead - fail now so the caller can react.
        raise KeyboardInterrupt("ROS shutdown requested")


def _move(arm, x, y, z, settle, raw=False):
    """Command the wrist to (x, y, z) mm and wait ``settle`` seconds."""
    arm.interface.set_pose(float(x), float(y), float(z), delay=0.0)
    _sleep(settle, raw=raw)

def grab_depth():
    msg = rospy.wait_for_message("/camera/aligned_depth_to_color/image_raw",
                                  Image, timeout=8)
    raw = np.frombuffer(msg.data, dtype=np.uint16)
    return raw.reshape(msg.height, msg.width).astype(np.float64) / 1000.0


def depth_at(depth, u, v, half=2):
    u, v = int(round(u)), int(round(v))
    patch = depth[max(0, v - half):v + half + 1, max(0, u - half):u + half + 1]
    valid = patch[patch > 0]
    return float(np.median(valid)) if len(valid) else None


def unproject(u, v, d):
    return np.array([(u - match_logic.CAM_CX) / match_logic.CAM_FX * d,
                     (v - match_logic.CAM_CY) / match_logic.CAM_FY * d, d])


def solve_arm_point(M, cam_point, scale):
    a = np.array([[M[0][j], M[1][j], M[2][j]] for j in range(3)])
    b = np.array([cam_point[j] / scale - M[3][j] for j in range(3)])
    return np.linalg.solve(a, b)

def compute_target_coords(cam, u, v, is_cake, scale, cake_offset=CAKE_CUP_OFFSET_X):
    """Compute the 3D arm workspace coordinates (x, y) from an image pixel."""
    xy = None
    z_est = None
    try:
        depth = grab_depth()
        d = depth_at(depth, u, v)
        if d is not None:
            point = solve_arm_point(cam.transform3d, unproject(u, v, d), scale)
            cand_z = float(point[2])
            # For an elevated cake, a depth reading that hit the floor/background
            # (z < -30 mm) is rejected so the dedicated cake affine calibration
            # is used instead.
            if is_cake and cand_z < -30.0:
                log("[TARGET] Cake depth reading hit floor/background (z=%.1f) "
                    "-> using cake affine" % cand_z)
            else:
                xy = (float(point[0]), float(point[1]))
                z_est = cand_z
    except Exception as exc:
        log("[WARN] depth reconstruction exception: %s" % exc)

    if xy is None:
        if is_cake:
            xy = match_logic.pixel_to_cake_xy(cam.transform3d, u, v,
                                              cup_offset_x=cake_offset)
        else:
            dlt_xy = match_logic.pixel_to_xy_at_z(cam.transform3d, u, v, -100.0)
            if dlt_xy is not None:
                xy = (float(dlt_xy[0]) + 20.0, float(dlt_xy[1]))

    return xy, z_est


def _make_scan_target(cid, u, v, xy, z_est, is_cake, far):
    """Build the scan-target dict that ``_prepare_target`` consumes."""
    return {
        "mode": "cake" if is_cake else "floor",
        "is_cake": bool(is_cake),
        "class": int(cid),
        "pixel": [round(u, 1), round(v, 1)],
        "xy": [round(xy[0], 1), round(xy[1], 1)],
        "z_est": round(z_est, 1) if z_est is not None else None,
        "far": bool(far),
    }


def _scan_target_coords(cam, u, v, is_cake, scale, cake_offset, far):
    """``(x, y)`` and the height estimate of one detection, near or far field."""
    if far:
        # Outside the calibrated band the depth reconstruction extrapolates
        # badly; the grasp-plane inversion is the only usable estimate.  The
        # next scan, taken after the approach step, refines it.
        xy = match_logic.far_target_xy(cam.transform3d, u, v, is_cake,
                                       cup_offset_x=cake_offset)
        if xy is None:
            log("[SCAN] far-field reconstruction failed at pixel (%.0f, %.0f)"
                % (u, v))
        return xy, None
    return compute_target_coords(cam, u, v, is_cake, scale, cake_offset)


def _scan_candidates(cam, cubes, far, scale, cake_offset):
    """Turn raw ``(class_id, u, v)`` detections into target dicts."""
    candidates = []
    for cid, u, v in cubes:
        is_cake = match_logic.is_cake(cid)
        xy, z_est = _scan_target_coords(cam, u, v, is_cake, scale,
                                        cake_offset, far)
        if xy is None:
            continue
        if not far:
            if is_cake and z_est is not None and not (-130.0 <= z_est <= 130.0):
                log("[FILTER] Cake candidate %d at (%.1f, %.1f) has "
                    "out-of-bounds z_est=%.1f -> skipped" % (cid, u, v, z_est))
                continue
            if not is_cake and z_est is not None and not (-250.0 <= z_est <= 110.0):
                log("[FILTER] Floor candidate %d at (%.1f, %.1f) has "
                    "out-of-bounds z_est=%.1f -> skipped" % (cid, u, v, z_est))
                continue
        candidates.append(_make_scan_target(cid, u, v, xy, z_est, is_cake, far))
    return candidates


def scan_scene(cam, arm, scale, fast=False, tag="", cake_offset=CAKE_CUP_OFFSET_X,
               conf_threshold=DEFAULT_DETECT_CONF,
               cake_conf_threshold=DEFAULT_CAKE_CONF,
               cake_site_conf_threshold=DEFAULT_CAKE_SITE_CONF):
    """Detect cubes, separating Cake vs Floor and near-field vs far-field ones.

    If fast=False: park the arm out of the camera view and take the samples.
    If fast=True: the arm is already parked; take stable samples only.

    Two anti-glare gates protect the priority Cake class:
    ``cake_conf_threshold`` for raw id 2 and ``cake_site_conf_threshold`` for
    raw id 20 (the printed cake cube, which is what the detector reports for
    the real Cake on the field).  Boxes below the gate are ignored completely.

    A detection ABOVE the calibrated near field (``v < 130 px``) is a cube that
    is still too far away for the arm.  It used to be discarded as "far
    wall/ceiling noise", which turned every out-of-reach cube into a NO_TARGET
    failure (exit code 3) with no base motion.  It is now returned as a target
    with ``far=True``: ``grasp_reach_state`` reports it out of reach and the
    caller drives the base one step, re-scans from the calibrated band and
    only then plans the grasp.

    Which candidate wins is decided by ``match_logic.choose_scan_target``:
    Cake priority first (operator rule), then the target that is already
    graspable or cheapest to reach.  A low-confidence detection at the frame
    corner therefore no longer outranks the cube standing right ahead.
    """
    if not fast:
        arm.interface.set_pose(PARK_POSE[0], PARK_POSE[1], PARK_POSE[2], delay=0.35)
        rospy.sleep(0.15)

    num_samples = 5
    raw_detections = cam.detector(num_samples=num_samples, timeout=4.0,
                                  conf_threshold=conf_threshold,
                                  cake_conf_threshold=cake_conf_threshold,
                                  cake_site_conf_threshold=cake_site_conf_threshold) or []
    log("[SCAN%s] Detections (samples=%d, conf>=%.2f, cake_conf>=%.2f, "
        "cake_site_conf>=%.2f): %s"
        % (tag, num_samples, conf_threshold, cake_conf_threshold,
           cake_site_conf_threshold, raw_detections))

    near_cubes = []
    far_cubes = []
    dismissed = []
    for item in raw_detections:
        try:
            cid = int(item[0])
            u, v = float(item[1][0]), float(item[1][1])
        except Exception:
            continue
        if match_logic.in_near_field(u, v):
            near_cubes.append((cid, u, v))
        elif match_logic.in_scan_window(u, v):
            far_cubes.append((cid, u, v))
        else:
            dismissed.append((cid, round(u), round(v)))

    if dismissed:
        log("[SCAN%s] %d detection(s) outside the scan window ignored: %s"
            % (tag, len(dismissed), dismissed))
    if far_cubes:
        log("[SCAN%s] %d target(s) beyond the calibrated near field "
            "(out of arm reach): %s -> base approach required"
            % (tag, len(far_cubes), far_cubes))

    candidates = (_scan_candidates(cam, near_cubes, False, scale, cake_offset)
                  + _scan_candidates(cam, far_cubes, True, scale, cake_offset))
    target = match_logic.choose_scan_target(candidates)
    if target is not None:
        log("[SCAN%s] Chosen target: mode=%s class=%d pixel=%s far=%s"
            % (tag, target["mode"], target["class"], target["pixel"],
               target["far"]))
        return target

    if not near_cubes and not far_cubes:
        log("[SCAN%s] No valid cubes detected in the camera window" % tag)
    return None

# Single-command base approach with vibration damping

def drive_approach_step(arm, target):
    """Align the heading and approach the target in one clean command sequence."""
    mode = target["mode"]
    x, y = target["xy"]
    far = bool(target.get("far"))

    # 1. Heading alignment (in-place turn)
    angle_rad = math.atan2(y, x)
    if abs(angle_rad) > math.radians(2.5):
        log("[APPROACH] Aligning heading: %.1f deg" % math.degrees(angle_rad))
        arm.turn_base(angle_rad)
        rospy.sleep(0.35)

    # 2. Linear displacement to the target sweet spot
    dist, desired = target_distance(mode, x, y)
    drive_m = (dist - desired) / 1000.0

    if abs(drive_m) >= 0.02:
        target_dist = math.hypot(x, y)
        forward = drive_m > 0
        blocked = bumper("front" if forward else "back")

        if blocked and target_dist < 180.0:
            log("[APPROACH] Bumper triggered and obstacle within %.0f mm - "
                "refusing forward drive to protect the front-row cubes!" % target_dist)
            return False

        # A far-field target is only an estimate: take a shorter step so the
        # re-scan can correct the remaining distance instead of overshooting.
        step_limit = FAR_DRIVE_STEP_M if far else MAX_DRIVE_STEP_M
        step_m = max(-step_limit, min(step_limit, drive_m))
        log("[APPROACH] Single drive: %+.1f cm (target dist: %.0f mm, desired: "
            "%.0f mm%s)" % (step_m * 100.0, dist, desired,
                            ", far-field estimate" if far else ""))
        arm.drive_base(step_m)

    # 3. Vibration damping dwell: let the suspension and the camera mast settle
    log("[APPROACH] Base motion complete. Settling the chassis for 0.85 s ...")
    rospy.sleep(0.85)
    return True

def plan_grasp(target, cake_touch_z=DEFAULT_CAKE_TOUCH_Z):
    """Turn a scan target into the wrist heights of one grasp.

    Raises :class:`GraspError` (``result="UNREACHABLE"``) when the touch or
    lift pose is outside the firmware IK workspace: the arm would silently
    refuse to move and the sequence would hold/return nothing.
    """
    mode = str(target.get("mode", "floor"))
    x = float(target["xy"][0])
    y = float(target["xy"][1])
    z_est = target.get("z_est")
    z_approach, z_touch, z_lift = _grasp_heights(mode, z_est, cake_touch_z)

    for pose_name, z in (("touch", z_touch), ("lift", z_lift)):
        if not match_logic.ik_reachable(x, y, z):
            raise GraspError(
                "grasp plan is outside the arm IK workspace (%s pose: x=%.1f, "
                "y=%.1f, z=%.1f mm)" % (pose_name, x, y, z),
                result="UNREACHABLE", code=7)

    pixel = target.get("pixel") or [0.0, 0.0]
    return GraspPlan(
        mode=mode,
        x=x,
        y=y,
        z_approach=z_approach,
        z_touch=z_touch,
        z_lift=z_lift,
        class_id=int(target.get("class", 0)),
        pixel=(float(pixel[0]), float(pixel[1])),
    )


def execute_sequence(arm, plan, hold_seconds=HOLD_SECONDS, return_cube=True,
                     park_pose=PARK_POSE,
                     transit_xy=(TRANSIT_X, TRANSIT_Y_FACTOR)):
    """Run the operator-requested grasp sequence (see the module docstring).

    ``return_cube=False`` stops after the hold: the cube is attached, the
    pump stays ON and the wrist stays at ``plan.z_lift``.  This is the mode
    the main program uses before carrying the cube to its destination zone.

    ``park_pose`` and ``transit_xy`` are the field-test landmarks (initial arm
    pose and safe lateral transit point); the main program may pass its own
    carry/park pose here.

    The function is exception-safe: it never releases the cube in the air.
    """
    phase = "transit"
    try:
        # -------- Step 1: lower the manipulator -------------------------
        log("[1/6] Lowering the manipulator above the cube (z=%.1f mm)"
            % plan.z_approach)
        _move(arm, transit_xy[0], plan.y * transit_xy[1], plan.z_approach,
              TRANSIT_TIME)
        _move(arm, plan.x, plan.y, plan.z_approach, APPROACH_TIME)
        phase = "above"

        # -------- Step 2: suction --------------------------------------
        phase = "pump_on"
        log("[2/6] Vacuum ON; descending strictly vertically to the cube "
            "(z=%.1f mm)" % plan.z_touch)
        arm.interface.set_pump(True)
        _sleep(PUMP_SETTLE_TIME)
        phase = "descent"
        _move(arm, plan.x, plan.y, plan.z_touch, DESCENT_TIME)
        _sleep(SEAL_DWELL_TIME)

        # -------- Step 3: strictly vertical lift ------------------------
        phase = "lift"
        log("[3/6] Raising the cube strictly vertically (z=%.1f mm)"
            % plan.z_lift)
        _move(arm, plan.x, plan.y, plan.z_lift, LIFT_TIME)
        _sleep(LIFT_SETTLE_TIME)

        # -------- Step 4: hold in the raised position -------------------
        phase = "hold"
        log("[4/6] Holding the cube in the raised position for %.1f s"
            % hold_seconds)
        _sleep(max(0.0, float(hold_seconds)))
        log("      hold finished")

        if not return_cube:
            log("[INTEGRATION] Cube stays attached (pump ON, wrist z=%.1f); "
                "the caller now carries it to the destination zone." % plan.z_lift)
            return

        # ==================================================================
        # STEPS 5-6: PUT THE CUBE BACK AND PARK THE ARM.
        # For the final integration this block is deleted (or skipped with
        # return_cube=False): the main program carries the cube to the zone.
        # ШАГИ 5-6: ВЕРНУТЬ КУБ НА МЕСТО И УБРАТЬ МАНИПУЛЯТОР - при интеграции
        # этот блок просто вырезается.
        # ==================================================================
        phase = "return_descent"
        log("[5/6] Lowering the cube strictly vertically back to the pickup point")
        _move(arm, plan.x, plan.y, plan.z_touch, DESCENT_TIME)
        _sleep(PLACE_REST_DWELL_TIME)
        phase = "release"
        log("      cube is resting on the ground: vacuum OFF")
        arm.interface.set_pump(False)
        phase = "released"
        _sleep(RELEASE_DWELL_TIME)

        phase = "retract"
        _move(arm, plan.x, plan.y, plan.z_approach, RETRACT_TIME)
        phase = "transit_back"
        _move(arm, transit_xy[0], 0.0, plan.z_approach, PARK_TIME)

        # -------- Step 6: park in the initial position ------------------
        phase = "park"
        log("[6/6] Returning the manipulator to its initial position %s"
            % (tuple(park_pose),))
        _move(arm, park_pose[0], park_pose[1], park_pose[2], PARK_TIME)
        phase = "parked"
        log("[DONE] Cube returned to its original position; arm parked.")
    except BaseException as exc:
        log("[ABORT] Sequence interrupted in phase '%s': %s: %s"
            % (phase, type(exc).__name__, exc))
        _recover(arm, plan, phase, park_pose=park_pose)
        raise


def _emergency_put_back(arm, plan):
    """Best-effort abort path: lower the cube to the pickup point and release.

    Returns ``(released, retracted)``: ``released`` means the vacuum was
    switched off while the cup was at the contact height (cube is on the
    ground), ``retracted`` means the arm was then raised to the clearance
    height.  Raw ``time.sleep`` waits are used so the recovery still works
    while rospy is shutting down after Ctrl-C.
    """
    released = False
    retracted = False
    try:
        log("[SAFE] Cube may be attached - lowering it back to the pickup point")
        _move(arm, plan.x, plan.y, plan.z_touch, DESCENT_TIME, raw=True)
        _sleep(PLACE_REST_DWELL_TIME, raw=True)
        arm.interface.set_pump(False)
        released = True
        log("[SAFE] Vacuum OFF: the cube is resting on the ground")
        _sleep(RELEASE_DWELL_TIME, raw=True)
        _move(arm, plan.x, plan.y, plan.z_approach, RETRACT_TIME, raw=True)
        retracted = True
    except BaseException as exc:
        log("[SAFE] emergency return incomplete: %s: %s"
            % (type(exc).__name__, exc))
    return released, retracted


def _recover(arm, plan, phase, park_pose=PARK_POSE):
    """Abort handling: never release the cube in the air, then park the arm."""
    released = False
    retracted = False

    if phase in _HOLDING_PHASES:
        released, retracted = _emergency_put_back(arm, plan)
    elif phase in _RETRACT_PHASES:
        # The cube was already released on the ground; only retract vertically.
        released = True
        try:
            # Repeat the release BEFORE moving away: if the original pump-off
            # command had been lost, retracting would carry the cube into the
            # air and the repeated release would then drop it.
            arm.interface.set_pump(False)
        except BaseException:
            pass
        try:
            _move(arm, plan.x, plan.y, plan.z_approach, RETRACT_TIME, raw=True)
            retracted = True
        except BaseException as exc:
            log("[SAFE] vertical retract failed: %s" % exc)

    if phase in _HOLDING_PHASES and not retracted:
        if released:
            log("[SAFE] Cube is on the floor, but the arm could not retract; "
                "leaving the arm in place.")
        else:
            log("[SAFE] Pump LEFT ON and the arm LEFT IN PLACE: the cube may "
                "still be attached. Release it manually or retry the script.")
        return
    if phase in _RETRACT_PHASES and not retracted:
        log("[SAFE] Cube is on the floor, but the arm could not retract; "
            "leaving the arm in place.")
        return

    try:
        log("[SAFE] Parking the manipulator ...")
        _move(arm, park_pose[0], park_pose[1], park_pose[2], PARK_TIME, raw=True)
    except BaseException as exc:
        log("[SAFE] could not park the manipulator: %s" % exc)

def _wait_for_arm_driver(arm, timeout=3.0):
    """Wait until the uArm driver subscribes to the position topic."""
    deadline = time.time() + timeout
    while (arm.interface.arm_position_pub.get_num_connections() == 0
           and time.time() < deadline):
        rospy.sleep(0.1)
    if arm.interface.arm_position_pub.get_num_connections() == 0:
        raise GraspError(
            "uArm SwiftPro driver is NOT connected "
            "(0 subscribers on position_write_topic)",
            result="ARM_NOT_CONNECTED", code=2)


def _prepare_target(cam, arm, scale, cake_offset, scan_only, entry,
                    conf_threshold, cake_conf_threshold,
                    cake_site_conf_threshold):
    """Scan the field and, if needed, drive the base until the target is in reach."""
    is_initial = True
    for attempt in range(MAX_APPROACH_ATTEMPTS):
        target = scan_scene(cam, arm, scale, fast=not is_initial,
                            tag="_init" if is_initial else "_approach_%d" % attempt,
                            cake_offset=cake_offset,
                            conf_threshold=conf_threshold,
                            cake_conf_threshold=cake_conf_threshold,
                            cake_site_conf_threshold=cake_site_conf_threshold)
        is_initial = False
        if target is None:
            if attempt == 0:
                raise GraspError(
                    "no valid target detected: the scan window is empty - put "
                    "the cube in front of the camera",
                    result="NO_TARGET", code=3)
            raise GraspError(
                "target lost after base-approach step %d (the re-scan is "
                "empty); the base may have turned away from the cube" % attempt,
                result="TARGET_LOST", code=6)

        mode = target["mode"]
        x, y = target["xy"]
        far = bool(target.get("far"))
        in_reach, metric = reach_state(mode, x, y, z_est=target.get("z_est"),
                                       far=far)
        if far:
            desc = ("5) Target beyond the near field (OUT OF ARM REACH) -> "
                    "base approach step, then re-scan")
        elif mode == "cake":
            desc = ("1) Cake IN REACH -> pick immediately without base motion"
                    if in_reach else
                    "2) Cake OUT OF REACH -> one base approach, then re-scan")
        else:
            desc = ("3) Standard cube IN REACH -> pick immediately without "
                    "base motion" if in_reach else
                    "4) Standard cube OUT OF REACH -> one base approach, "
                    "then re-scan")
        log("[DECISION] (Attempt %d/%d) %s"
            % (attempt + 1, MAX_APPROACH_ATTEMPTS, desc))
        log("[TARGET] mode=%s class=%d xy=(%.1f, %.1f) far=%s in_reach=%s "
            "dist=%.1f z_est=%s"
            % (mode, target["class"], x, y, far, in_reach, metric,
               target.get("z_est")))
        entry["target"] = target

        if in_reach or scan_only:
            return target

        if not drive_approach_step(arm, target):
            raise GraspError("base approach blocked or unsafe",
                             result="APPROACH_BLOCKED", code=6)
        log("[RETRY] Base stopped; re-scanning and re-checking the reach ...")

    last = entry.get("target") or {}
    raise GraspError(
        "target still out of reach after %d base-approach step(s) "
        "(last: mode=%s xy=%s far=%s); the cube is too far away - move it "
        "closer or check that the base can drive"
        % (MAX_APPROACH_ATTEMPTS, last.get("mode"), last.get("xy"),
           last.get("far")),
        result="OUT_OF_REACH", code=6)


def _write_result(entry):
    """Append one JSON line with the outcome (field diagnostics only)."""
    path = os.path.expanduser("~/logs/grasp_generic.jsonl")
    try:
        directory = os.path.dirname(path)
        if directory:
            os.makedirs(directory, exist_ok=True)
        with open(path, "a", encoding="utf8") as handle:
            handle.write(json.dumps(entry) + "\n")
    except Exception as exc:
        log("[WARN] cannot write the run log: %s" % exc)


def build_parser():
    parser = argparse.ArgumentParser(
        description="Grasp a cube, raise it, hold it 5 s, put it back, park the arm.")
    parser.add_argument(
        "--scan-only", action="store_true",
        help="only detect/assess the target: the base never moves and the "
             "grasp sequence is not executed (the arm is parked for the camera)")
    parser.add_argument(
        "--no-place", action="store_true",
        help="stop after the 5 s hold and keep the cube on the suction cup "
             "(pump ON) - preview of the main-program integration")
    parser.add_argument(
        "--hold", type=float, default=HOLD_SECONDS,
        help="hold time in the raised position, seconds (default %.1f)"
             % HOLD_SECONDS)
    parser.add_argument(
        "--cake-offset", type=float, default=CAKE_CUP_OFFSET_X,
        help="suction-cup forward offset for the raised cake, mm (default %.1f)"
             % CAKE_CUP_OFFSET_X)
    parser.add_argument(
        "--conf", type=float, default=DEFAULT_DETECT_CONF,
        help="confidence floor for ordinary cubes (default %.2f; the detector "
             "node itself already drops non-cake boxes below 0.25)"
             % DEFAULT_DETECT_CONF)
    parser.add_argument(
        "--cake-conf", type=float, default=DEFAULT_CAKE_CONF,
        help="anti-glare confidence gate for raw cake id 2 (default %.2f): "
             "false cake hypotheses on banana cubes reach 0.85, so keep it high"
             % DEFAULT_CAKE_CONF)
    parser.add_argument(
        "--cake-site-conf", type=float, default=DEFAULT_CAKE_SITE_CONF,
        help="anti-glare confidence gate for the cake_site id 20, i.e. the "
             "real Cake cube on the field (default %.2f; measured real cake "
             "0.44..0.87, false cake_site <= 0.53)" % DEFAULT_CAKE_SITE_CONF)
    return parser


def _install_signal_handlers():
    """Turn Ctrl-C/SIGTERM into a KeyboardInterrupt inside the sequence.

    rospy's default handler calls ``signal_shutdown()`` first - all topics are
    closed and later publishes are silently dropped - and then swallows the
    KeyboardInterrupt raised by the previous handler.  The script must instead
    catch the interruption while its publishers are still alive, so it uses
    ``init_node(disable_signals=True)`` and installs this handler.
    """

    def _raise_interrupt(signum, _frame):
        raise KeyboardInterrupt("signal %d" % signum)

    signal.signal(signal.SIGINT, _raise_interrupt)
    signal.signal(signal.SIGTERM, _raise_interrupt)


def main():
    args = build_parser().parse_args(rospy.myargv()[1:])
    rospy.init_node("grasp_generic", anonymous=True, disable_signals=True)
    _install_signal_handlers()

    entry = {
        "ts": time.strftime("%Y-%m-%dT%H:%M:%S"),
        "result": None,
        "target": None,
        "plan": None,
        "hold_seconds": float(args.hold),
    }
    result = "ERROR"
    exit_code = 1
    arm = None

    try:
        cam = CamAction()
        arm = ArmAction(cam=cam)
        _wait_for_arm_driver(arm)

        # NEVER blindly switch the pump off here: a previous interrupted run
        # may still hold a cube at lift height, and releasing it would drop
        # the cube.  The pump is controlled by steps 2 and 5 only.
        arm.interface.set_status(True)
        rospy.sleep(0.20)

        scale, cake_touch = load_memory()
        log("[CONFIG] depth scale=%s | cake touch z=%.1f mm | floor touch z=%.1f mm"
            % (scale, cake_touch, FLOOR_TOUCH_Z))

        target = _prepare_target(cam, arm, scale, args.cake_offset,
                                 args.scan_only, entry,
                                 conf_threshold=args.conf,
                                 cake_conf_threshold=args.cake_conf,
                                 cake_site_conf_threshold=args.cake_site_conf)
        if args.scan_only:
            log("[SCAN-ONLY] Assessment complete; no motion command was sent.")
            result, exit_code = "SCANNED", 0
        else:
            plan = plan_grasp(target, cake_touch)
            entry["plan"] = asdict(plan)
            log("=" * 70)
            log("GRASP -> LIFT -> HOLD %.1f s -> RETURN -> PARK" % args.hold)
            log("plan: mode=%s class=%d xy=(%.1f, %.1f) approach=%.1f "
                "touch=%.1f lift=%.1f"
                % (plan.mode, plan.class_id, plan.x, plan.y,
                   plan.z_approach, plan.z_touch, plan.z_lift))
            log("=" * 70)

            execute_sequence(arm, plan, hold_seconds=args.hold,
                             return_cube=not args.no_place)

            if args.no_place:
                log("[HOLD-ONLY] The cube stays attached to the suction cup "
                    "(pump ON); this mode is the main-program integration preview.")
                result = "HELD"
            else:
                result = "DONE"
            exit_code = 0
    except GraspError as exc:
        log("[FAIL] %s" % exc)
        result, exit_code = exc.result, exc.code
    except BaseException as exc:
        # execute_sequence() has already run its safe recovery (the cube was
        # lowered to the pickup point before the pump release).  Do NOT touch
        # the pump here - an attached cube would be dropped.
        log("[ABORT] %s: %s" % (type(exc).__name__, exc))
        result, exit_code = "ABORTED", 4
    finally:
        entry["result"] = result
        _write_result(entry)
        log("[COMPLETED] result=%s" % result)

    return exit_code


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""YOLO object detector for the Spark competition (publishes /objects).

Two YOLOv5 models are run on every frame (the generic model plus a banana
specialist).  They can both report the same physical cube, so their detections
are merged with ``match_logic.dedupe_boxes`` before publishing: one physical
cube = one entry in /objects.  Without this, the FSM could pick a duplicated
box as its target.

Test-time augmentation is OFF by default: it triples inference time on the
on-board NUC and the FSM already votes over several frames
(``runtask.py: GRASP_FRAMES``).  Enable it with ``~augment:=true`` only if the
CPU has headroom.
"""

import os
import pathlib
import platform

import cv2
import yolov5
import rospy
import rospkg
import yaml
from cv_bridge import CvBridge, CvBridgeError
from sensor_msgs.msg import Image
from vision_msgs.msg import Detection2D, Detection2DArray, ObjectHypothesisWithPose

import match_logic

# Torch checkpoints saved on Windows carry WindowsPath objects; they cannot be
# unpickled on Linux without this alias.
if platform.system() != 'Windows':
    pathlib.WindowsPath = pathlib.PosixPath

# Everything below this confidence is published; the FSM applies its own,
# stricter grasp threshold (match_logic.CONF_THRESHOLD).
PUBLISH_CONF = 0.25

# Anti-glare gates for the two raw Cake classes; two thresholds because their
# false-positive profiles differ (measured on the robot, 20 s / 394 boxes):
#   * id 2  "cake"      : false positives on banana cubes up to 0.852
#   * id 20 "cake_site" : real cake cube 0.44..0.873, false ones <= 0.527
# The Cake has FSM priority, so a cheap false positive must never be published.
# Tunable with ``~cake_conf`` / ``~cake_site_conf`` (values > 1.0 disable).
CAKE_PUBLISH_CONF = 0.90
CAKE_SITE_PUBLISH_CONF = 0.60

# Only competition classes are meaningful downstream. items_config.yaml also
# declares ``base: 99`` and YOLO happily detects the robot's own base at 0.95,
# which used to fill /objects with an id the FSM ignores anyway.
REPORTED_IDS = frozenset(match_logic.ID_MAP)


class SparkDetect:
    """Dual-model YOLOv5 detector with cross-model de-duplication."""

    def __init__(self, model_path_1, model_path_2, augment=False,
                 dedupe_px=match_logic.DEDUPE_DIST_PX):
        self.augment = bool(augment)
        self.dedupe_px = float(dedupe_px)
        rospy.loginfo(f">> [SPARK-DETECT] Loading model 1: {model_path_1}")
        self.model_1 = yolov5.load(model_path_1)
        self.model_1.conf = 0.03
        rospy.loginfo(f">> [SPARK-DETECT] Loading model 2: {model_path_2}")
        self.model_2 = yolov5.load(model_path_2)
        self.model_2.conf = 0.03
        rospy.loginfo(f">> [SPARK-DETECT] Both models loaded "
                      f"(augment={self.augment}, dedupe={self.dedupe_px:.0f}px).")

    def _collect(self, model, label, image, boxes):
        """Append one model's detections to ``boxes`` (never raises)."""
        try:
            results = model(image, augment=self.augment)
        except Exception as e:
            rospy.logwarn_throttle(5.0, f">> [SPARK-DETECT] {label} inference failed: {e}")
            return
        try:
            predictions = results.xyxy[0]
        except Exception as e:
            rospy.logwarn_throttle(5.0, f">> [SPARK-DETECT] {label} returned no "
                                        f"prediction tensor: {e}")
            return
        for prediction in predictions:
            try:
                x1, y1, x2, y2, conf, cls = [float(v) for v in prediction[:6]]
                boxes.append({
                    "name": model.model.names[int(cls)],
                    "score": conf,
                    "cx": int((x1 + x2) / 2),
                    "cy": int((y1 + y2) / 2),
                    "size_x": int(x2 - x1),
                    "size_y": int(y2 - y1),
                    "x1": int(x1), "y1": int(y1),
                    "x2": int(x2), "y2": int(y2),
                    "model": label,
                })
            except Exception as e:
                rospy.logwarn_throttle(5.0, f">> [SPARK-DETECT] bad box from "
                                            f"{label}: {e}")

    def detect(self, image):
        """Return de-duplicated boxes (dicts) for ``image``; highlights in place."""
        boxes = []
        self._collect(self.model_1, "model_1", image, boxes)
        self._collect(self.model_2, "model_2", image, boxes)
        kept = match_logic.dedupe_boxes(boxes, dist_px=self.dedupe_px)

        for box in kept:
            cv2.rectangle(image, (box["x1"], box["y1"]), (box["x2"], box["y2"]),
                          (0, 255, 0), 3)
            label = f'{box["name"]} {box["score"]:.2f}'
            cv2.putText(image, label, (box["x1"], box["y1"] - 10),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)
            cv2.circle(image, (box["cx"], box["cy"]), 5, (255, 0, 0), -1)
        return kept


class Detector:
    def __init__(self):

                      f"cake_publish_conf={self.cake_publish_conf:.2f}, "

        # Load the competition item configuration.

        except Exception:

            if os.path.exists(alt_path):
                item_file = alt_path
        with open(item_file, 'r') as handle:
            config_contents = yaml.safe_load(handle)
        self.obj_id = config_contents['items']
        self.items = list(self.obj_id.keys())
        rospy.loginfo(f">> [DETECTOR] item classes from {item_file}: {self.items}")

        # Load the models BEFORE subscribing so the first frame is not delayed
        # and a broken checkpoint fails loudly at startup.
        home_dir = os.environ.get('HOME', '/home/spark')
        model1 = os.path.join(home_dir, "auto.pt")
        model2 = os.path.join(home_dir, "banana.pt")

        self.detector = SparkDetect(model1, model2, augment=augment,

        rospy.loginfo(">> [DETECTOR] Neural network models loaded and ready.")

        self._frame_count = 0
        self.image_sub = rospy.Subscriber(
            "/camera/color/image_raw", Image, self.image_cb,
            queue_size=1, buff_size=2 ** 24,
        )
        rospy.loginfo(">> [DETECTOR] Subscribed to /camera/color/image_raw")

    def image_cb(self, data):
        obj_array = Detection2DArray()
        try:
            cv_image = self.bridge.imgmsg_to_cv2(data, "bgr8")
        except CvBridgeError as e:
            rospy.logerr(f">> [DETECTOR] Bridge error: {e}")
            return
        image = cv2.cvtColor(cv_image, cv2.COLOR_BGR2RGB)
        obj_array.header = data.header

        try:
            boxes = self.detector.detect(image)
        except Exception as e:
            rospy.logwarn_throttle(5.0, f">> [DETECTOR] inference exception: {e}")
            boxes = []

        published = 0
        for box in boxes:
            if box["name"] not in self.items:
                continue
            item_id = int(self.obj_id[box["name"]])
            if item_id not in REPORTED_IDS:
                continue
            if item_id == match_logic.CAKE_ID:
                min_conf = self.cake_publish_conf
            elif match_logic.is_cake(item_id):
                # Printed "cake_site" alias (id 20): this is what the detector
                # actually reports for the Cake cube on the field.
                min_conf = self.cake_site_publish_conf
            else:
                min_conf = self.publish_conf

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Hand-eye calibration of the camera<-arm transform from the orange pad.

Fits the 4x3 projective matrix ``M`` used by runtask.py::

    [Xc, Yc, Zc] = [x, y, z, 1] @ M          (arm mm -> camera metres)
    u = CAM_FX * Xc/Zc + CAM_CX
    v = CAM_FY * Yc/Zc + CAM_CY

Only the projection is needed for grasping (pixel_to_arm_at_z works on ratios),
so the matrix is recovered up to scale with the Direct Linear Transform from
>= 6 correspondences (commanded wrist pose -> detected pad pixel).  The depth
camera is NOT used, so the broken depth stream (SI-33) cannot pollute the fit.

Output: ``~/transform_3d.dat`` (np.save of a (4,3) array), which runtask.py
already loads automatically and prefers over the hard-coded TRANSFORM_3D.
A sidecar ``~/transform_3d.meta.json`` records when the fit was made and how
good it was, so the startup log can say how old the calibration is.

Usage:
    python3 -u ~/codes/calib3d.py            # fit and save
    python3 -u ~/codes/calib3d.py --poses ...  # custom
    python3 -u ~/codes/calib3d.py --check    # verify the saved matrix only

The ``--check`` mode moves the arm to one pose above the cubes, finds the
orange suction pad and compares its pixel with the projection of the saved
matrix.  Run it after moving the camera: a stale matrix shows up as tens of
pixels of error, a good one as ~2 px.
"""

import argparse
import hashlib
import json
import os
import sys
import time

import cv2
import numpy as np
import rospy
from cv_bridge import CvBridge

from calib_pad import pad_pixel, grab
from runtask import ArmAction, CAM_FX, CAM_FY, CAM_CX, CAM_CY

OUT_PATH = os.path.expanduser("~/transform_3d.dat")
META_PATH = os.path.expanduser("~/transform_3d.meta.json")

# Spread over the reachable + visible workspace (the pad is invisible for
# x < ~235 mm, SI-29).  Three z levels give vertical spread so the fit is
# well conditioned, unlike the single-height homography (SI-31).
DEFAULT_POSES = [
    (240, 0, -60), (240, 0, -10), (240, 0, 30),
    (270, 0, -60), (270, 0, -10),
    (300, 0, -60), (300, 0, -10),
    (270, 55, -35), (270, -55, -35),
    (300, 55, -35), (300, -55, -35),
    (270, 0, -35),
]

#: A suction pad that is partly cut off by the arm or the image border still
#: returns a centroid, but it is biased.  Reject those blobs: one weak point
#: used to drag a whole fit from 1.1 px to 3.1 px leave-one-out.
MIN_SAMPLE_AREA = 400.0


def collect(arm, bridge, poses, settle):
    samples = []
    for (x, y, z) in poses:
        arm.interface.set_pose(float(x), float(y), float(z), delay=settle)
        rospy.sleep(0.25)
        found = None
        for _ in range(3):
            found = pad_pixel(grab(bridge))
            if found:
                break
            rospy.sleep(0.12)
        if not found:
            print("[MISS] pad not visible at (%d, %d, %d)" % (x, y, z))
            continue
        u, v, area = found
        if area < MIN_SAMPLE_AREA:
            # A pad cut in half by the arm or the border still yields a
            # centroid, but it is biased; one such point dragged a whole fit
            # from 1.1 px to 3.1 px leave-one-out.
            print("[SKIP] pad at (%d, %d, %d) is tiny (area=%.0f, need %.0f): "
                  "centroid would bias the fit" % (x, y, z, area, MIN_SAMPLE_AREA))
            continue
        print("[SEEN] arm=(%d, %d, %d) -> pixel (%.1f, %.1f) area=%.0f"
              % (x, y, z, u, v, area))
        samples.append((float(x), float(y), float(z), u, v))
    return samples


def dlt(samples):
    """Solve the 4x3 projective matrix from pixel correspondences (DLT)."""
    fx, fy, cx, cy = CAM_FX, CAM_FY, CAM_CX, CAM_CY
    rows = []
    for x, y, z, u, v in samples:
        p = (x, y, z, 1.0)
        up = (u - cx) / fx
        vp = (v - cy) / fy
        rows.append([p[0], p[1], p[2], p[3], 0, 0, 0, 0,
                     -up * p[0], -up * p[1], -up * p[2], -up * p[3]])
        rows.append([0, 0, 0, 0, p[0], p[1], p[2], p[3],
                     -vp * p[0], -vp * p[1], -vp * p[2], -vp * p[3]])
    A = np.array(rows, dtype=np.float64)
    if A.shape[0] < 12:
        return None
    _, _, vh = np.linalg.svd(A)
    m = vh[-1]                 # null vector, 12 elements
    return m.reshape(3, 4).T   # -> (4, 3)


def project(M, x, y, z):
    cam = np.array([x, y, z, 1.0], dtype=np.float64) @ M
    if cam[2] <= 1e-6:
        return None
    return (CAM_FX * cam[0] / cam[2] + CAM_CX,
            CAM_FY * cam[1] / cam[2] + CAM_CY)


def evaluate(M, samples):
    errors = []
    for x, y, z, u, v in samples:
        proj = project(M, x, y, z)
        if proj is None:
            # Keep the list aligned with ``samples``: the robust trim zips them.
            errors.append(float("inf"))
            continue
        errors.append(float(np.hypot(proj[0] - u, proj[1] - v)))
    return errors


def leave_one_out(samples):
    loo = []
    for i in range(len(samples)):
        M = dlt(samples[:i] + samples[i + 1:])
        if M is None:
            continue
        x, y, z, u, v = samples[i]
        proj = project(M, x, y, z)
        if proj is not None:
            loo.append(float(np.hypot(proj[0] - u, proj[1] - v)))
    return loo


def fit(samples):
    """DLT plus the sign convention; returns (M, per-sample pixel errors)."""
    M = dlt(samples)
    if M is None:
        return None, []
    # The DLT null vector has an arbitrary sign.  A point in front of the
    # camera must have Zc > 0, so flip the whole matrix if the samples end up
    # behind it.
    zcs = [float((np.array([x, y, z, 1.0]) @ M)[2])
           for x, y, z, _, _ in samples]
    if np.median(zcs) < 0:
        M = -M
        print("[FIT ] sign flipped so samples are in front of the camera")
    return M, evaluate(M, samples)


def fit_robust(samples, keep_min=8):
    """Fit, drop gross outliers once, refit.  Returns (M, errors, samples).

    The DLT is a least-squares fit, so a single bad correspondence (pad half
    hidden behind the arm) tilts the whole matrix.  Trimming once and refitting
    keeps the honest ~1 px residual instead of the ~3 px one.
    """
    M, errors = fit(samples)
    if M is None or len(samples) < keep_min + 1:
        return M, errors, samples
    ordered = sorted(errors)
    median = ordered[len(ordered) // 2]
    limit = max(3.0, 3.0 * median)
    keep = [sample for sample, error in zip(samples, errors) if error <= limit]
    if len(keep) < keep_min or len(keep) == len(samples):
        return M, errors, samples
    M2, errors2 = fit(keep)
    if M2 is None:
        return M, errors, samples
    dropped = [sample for sample, error in zip(samples, errors) if error > limit]
    print("[FIT ] dropped %d outlier(s) above %.1f px: %s"
          % (len(dropped), limit, dropped))
    return M2, errors2, keep


def run_check(tol_px=6.0):
    """Verify the saved matrix against the real pad; 0 = pass, 2 = fail."""
    arm = ArmAction()
    ok, detail = arm.verify_calibration(tol_px=tol_px)
    print("[CHECK] %s: %s" % ("PASS" if ok else "FAIL", detail))
    if not ok:
        print("[HINT] the camera probably moved: run the full fit "
              "`python3 -u ~/codes/calib3d.py`")
    arm.arm_home(block=True)
    print("[DONE] arm home")
    return 0 if ok else 2

    # frame detects a moved camera far more sharply than comparing against the
    # model, whose own residual is ~2-3 px.
    check_pose = ArmAction.CALIB_CHECK_POSE
    pad_uv = arm.measure_pad(check_pose)
    if pad_uv:
        meta["check"] = {
            "pose": [float(value) for value in check_pose],
            "uv": [round(pad_uv[0], 2), round(pad_uv[1], 2)],
        }
        print("[CHECK] reference pad at (%.1f, %.1f) for pose %s"
              % (pad_uv[0], pad_uv[1], check_pose))
    else:
        print("[WARN] pad not visible at the check pose %s: the startup "
              "self-check will fall back to the projection" % (check_pose,))

    try:
        with open(META_PATH, "w", encoding="utf8") as handle:
            json.dump(meta, handle, indent=2, sort_keys=True)
        print("[META] %s" % META_PATH)
    except Exception as exc:
        print("[WARN] could not write %s: %s" % (META_PATH, exc))

    # Sanity anchors for the operator's measurements: floor ~ -150 mm, cube top
    # ~ -50 mm, pad ~ 180 mm above the floor at home z=35.
    for label, (px, py, pz) in (
            ("home(35)", (130.0, 0.0, 35.0)),
            ("touch(-50)", (250.0, 0.0, -50.0))):
        proj = project(M, px, py, pz)
        print("[MAP ] %s -> %s" % (label, proj if proj else "behind camera"))

    arm.arm_home()
    print("[DONE] arm home")
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
import sys

def patch_file(filepath):
    with open(filepath, "r") as f:
        content = f.read()

    old_block = """\ttry
\t{
\t\t_serial.setPort("/dev/ttyACM0");
\t\t_serial.setBaudrate(115200);
\t\tserial::Timeout to = serial::Timeout::simpleTimeout(1000);
\t\t_serial.setTimeout(to);
\t\t_serial.open();
\t\tROS_INFO_STREAM("Port has been open successfully");
\t}
\tcatch (serial::IOException& e)
\t{
\t\tROS_ERROR_STREAM("Unable to open port");
\t\treturn -1;
\t}"""

    new_block = """\tstd::string candidate_ports[] = {"/dev/ttyACM0", "/dev/ttyACM1", "/dev/ttyACM2"};
\tbool port_opened = false;
\tfor (size_t i = 0; i < 3; i++)
\t{
\t\ttry
\t\t{
\t\t\t_serial.setPort(candidate_ports[i]);
\t\t\t_serial.setBaudrate(115200);
\t\t\tserial::Timeout to = serial::Timeout::simpleTimeout(1000);
\t\t\t_serial.setTimeout(to);
\t\t\t_serial.open();
\t\t\tif (_serial.isOpen())
\t\t\t{
\t\t\t\tROS_INFO_STREAM("Port " << candidate_ports[i] << " has been open successfully");
\t\t\t\tport_opened = true;
\t\t\t\tbreak;
\t\t\t}
\t\t}
\t\tcatch (serial::IOException& e)
\t\t{
\t\t}
\t}
\tif (!port_opened)
\t{
\t\tROS_ERROR_STREAM("Unable to open any port (/dev/ttyACM0, /dev/ttyACM1, /dev/ttyACM2)");
\t\treturn -1;
\t}"""

    if old_block in content:
        content = content.replace(old_block, new_block, 1)
        with open(filepath, "w") as f:
            f.write(content)
        print("Patched:", filepath)
        return True
    else:
        print("Already patched or pattern not found in:", filepath)
        return False

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Orange suction-pad detection, shared by runtask.py and the calib tools.

WHY A SEPARATE MODULE
---------------------
The startup calibration check needs to see the pad, and so do calib_pad.py and
calib3d.py.  This code used to live in calib_pad.py, which imports runtask.py
(for ArmAction) at module level.  When runtask.py itself is the running program
it is the ``__main__`` module, so ``import calib_pad`` tried to import a SECOND
copy of runtask under a different name; that failed on the robot and the
self-check silently degraded to "pad not visible".  Keeping the detector free
of runtask breaks that cycle.

Deployment: this file must sit next to runtask.py in
``auto_match/nodes/`` and in ``~/codes/`` (upload.ps1 copies it to both).
"""

import cv2
import numpy as np
import rospy
from sensor_msgs.msg import Image


# Orange suction pad. The bear/banana plush are beige/brown, so saturation and
# value stay low there; the printed cubes are muted too.
LOWER = np.array([4, 130, 110])
UPPER = np.array([24, 255, 255])
MIN_AREA = 200          # a partially visible pad gives a biased centroid
BORDER_PX = 6           # blobs touching the image edge are clipped -> reject


def grab(bridge, timeout=3.0):
    """One color frame from the RealSense as a BGR numpy array."""
    msg = rospy.wait_for_message("/camera/color/image_raw", Image,
                                 timeout=timeout)
    return bridge.imgmsg_to_cv2(msg, "bgr8")


def pad_pixel(image, debug=False):
    """Centroid of the largest orange blob (the pad), or None.

    Rejects blobs that touch the image border: a clipped pad still yields a
    centroid, but it is shifted towards the visible part and would silently
    bend the homography.
    """
    hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)
    mask = cv2.inRange(hsv, LOWER, UPPER)
    mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN,
                            np.ones((5, 5), np.uint8), iterations=1)
    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL,
                                   cv2.CHAIN_APPROX_SIMPLE)
    if not contours:
        return None
    blob = max(contours, key=cv2.contourArea)
    area = cv2.contourArea(blob)
    if area < MIN_AREA:
        return None
    bx, by, bw, bh = cv2.boundingRect(blob)
    height, width = image.shape[:2]
    if (bx <= BORDER_PX or by <= BORDER_PX
            or bx + bw >= width - BORDER_PX or by + bh >= height - BORDER_PX):
        if debug:
            print("      pad blob clipped at the border, rejected")
        return None
    moments = cv2.moments(blob)
    if moments["m00"] == 0:
        return None
    u = moments["m10"] / moments["m00"]
    v = moments["m01"] / moments["m00"]
    if debug:
        print("      pad blob area=%.0f at (%.1f, %.1f)" % (area, u, v))
    return u, v, area

"""Guards around the camera->arm (hand-eye) calibration.

The expensive failure this suite protects against: a stale or broken
``transform_3d.dat`` makes every grasp miss by 5-7 cm while every subsystem
reports "OK".  ``check_transform`` catches the cheap breakages at startup and
``pad_projection_error`` is the number the active pad check compares.
"""

import os
import sys
import unittest

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from _bootstrap import CODES_DIR, match_logic  # noqa: E402

    import numpy as np
except ImportError:            # the robot has numpy; keep the suite importable
    np = None

CALIB_FILE = os.path.join(CODES_DIR, "transform_3d.dat")
CHECK_POSE = (270.0, 0.0, 40.0)


@unittest.skipUnless(np is not None and os.path.exists(CALIB_FILE),
                     "needs numpy and the shipped calibration file")
class TestTransformSanity(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.matrix = np.load(CALIB_FILE).tolist()

    def test_shipped_matrix_passes(self):
        ok, problems, stats = match_logic.check_transform(self.matrix)
        self.assertTrue(ok, problems)
        self.assertGreater(stats["px_per_mm"], 0.5)
        self.assertLess(stats["px_per_mm"], 2.0)
        self.assertLess(stats["roundtrip_mm"], 5.0)

    def test_wrong_shape_is_rejected(self):
        for bad in ([[1, 2, 3], [4, 5, 6], [7, 8, 9]], [[1, 2], [3, 4]],
                    [[1, 2, 3, 4]] * 4):
            ok, problems, _ = match_logic.check_transform(bad)
            self.assertFalse(ok)
            self.assertTrue(problems)

    def test_non_finite_values_are_rejected(self):
        broken = [row[:] for row in self.matrix]
        broken[2][1] = float("nan")
        self.assertFalse(match_logic.check_transform(broken)[0])

    def test_all_zero_matrix_is_rejected(self):
        self.assertFalse(match_logic.check_transform([[0.0] * 3] * 4)[0])

    def test_workspace_outside_the_image_is_rejected(self):
        moved = [row[:] for row in self.matrix]
        moved[3][0] += 5.0          # workspace now metres off to the side
        ok, problems, _ = match_logic.check_transform(moved)
        self.assertFalse(ok)
        self.assertTrue(problems)

    def test_uniform_scale_does_not_change_the_projection(self):
        """The DLT is defined up to scale; a scaled copy must still pass."""
        scaled = [[5.0 * value for value in row] for row in self.matrix]
        ok, problems, _ = match_logic.check_transform(scaled)
        self.assertTrue(ok, problems)
        base = match_logic.project_arm(self.matrix, *CHECK_POSE)
        scaled_uv = match_logic.project_arm(scaled, *CHECK_POSE)
        self.assertAlmostEqual(base[0], scaled_uv[0], places=6)
        self.assertAlmostEqual(base[1], scaled_uv[1], places=6)

class TestClassificationBoard(unittest.TestCase):
    """decode_classification_board() is what run_task() uses for the mapping."""

    def assertFullMapping(self, mapping):
        """Every canonical item has a zone and the cake is pinned to A."""
        for item in (match_logic.TEDDY_ID, match_logic.CLOCK_ID, match_logic.BANANA_ID):
            self.assertIn(item, mapping, f"item {item} has no zone")
            self.assertIn(mapping[item], ZONES)
        self.assertEqual(mapping[match_logic.CAKE_ID], "Collection_A")

    def test_standard_board(self):
        mapping = match_logic.decode_classification_board([1, 3, 4])
        self.assertFullMapping(mapping)
        self.assertEqual(mapping[1], "Collection_B")
        self.assertEqual(mapping[3], "Collection_C")
        self.assertEqual(mapping[4], "Collection_D")

    def test_permutation(self):
        mapping = match_logic.decode_classification_board([4, 1, 3])
        self.assertEqual(mapping[4], "Collection_B")
        self.assertEqual(mapping[1], "Collection_C")
        self.assertEqual(mapping[3], "Collection_D")
        self.assertEqual(mapping[match_logic.CAKE_ID], "Collection_A")

    def test_yolo_aliases(self):
        """Board read as [30, 40, 1] must normalise before it is applied."""
        mapping = match_logic.decode_classification_board([30, 40, 1])
        self.assertEqual(mapping[3], "Collection_B")
        self.assertEqual(mapping[4], "Collection_C")
        self.assertEqual(mapping[1], "Collection_D")

    def test_duplicate_or_corrupt_board_falls_back(self):
        """SI-14: a broken board must never leave the mission without a mapping."""
        for broken in ([1, 1, 99], [None, None, None], [], [1, 3]):
            mapping = match_logic.decode_classification_board(broken)
            self.assertFullMapping(mapping)
            self.assertEqual(mapping, match_logic.normalize_mapping(None))

    def test_alias_keys_are_present(self):
        mapping = match_logic.decode_classification_board([1, 3, 4])
        self.assertEqual(mapping[10], mapping[1])
        self.assertEqual(mapping[20], "Collection_A")
        self.assertEqual(mapping[30], mapping[3])
        self.assertEqual(mapping[40], mapping[4])

    def test_board_is_confident(self):
        self.assertTrue(match_logic.board_is_confident([1, 3, 4]))
        self.assertTrue(match_logic.board_is_confident([30, 40, 1]))
        self.assertFalse(match_logic.board_is_confident([1, 1, 99]))
        self.assertFalse(match_logic.board_is_confident([]))

    def test_cake_is_never_taken_from_the_board(self):

class FakeTime(object):
    """Controllable stand-in for time.monotonic."""

    def __init__(self):
        self.now = 1000.0

    def __call__(self):
        return self.now

    def advance(self, seconds):
        self.now += seconds


class TestMatchClock(unittest.TestCase):
    """The match clock is what keeps the run inside the 10-minute limit."""

    def setUp(self):
        self.clock_time = FakeTime()
        self.clock = match_logic.MatchClock(
            limit_s=540.0, warn_s=60.0, time_fn=self.clock_time)

    def test_elapsed_is_zero_before_start(self):
        self.assertEqual(self.clock.elapsed(), 0.0)
        self.assertEqual(self.clock.remaining(), 540.0)
        self.assertFalse(self.clock.expired())

    def test_start_returns_self_and_resets_elapsed(self):
        self.assertIs(self.clock.start(), self.clock)
        self.assertEqual(self.clock.elapsed(), 0.0)

    def test_elapsed_and_remaining_follow_time(self):
        self.clock.start()
        self.clock_time.advance(120.0)
        self.assertEqual(self.clock.elapsed(), 120.0)
        self.assertEqual(self.clock.remaining(), 420.0)

    def test_warning_window(self):
        self.clock.start()
        self.clock_time.advance(500.0)      # 40 s left
        self.assertTrue(self.clock.warn())
        self.assertFalse(self.clock.expired())

    def test_expiry_is_exact_at_the_limit(self):
        self.clock.start()
        self.clock_time.advance(539.9)
        self.assertFalse(self.clock.expired())
        self.clock_time.advance(0.2)
        self.assertTrue(self.clock.expired())
        self.assertEqual(self.clock.remaining(), 0.0)
        self.assertFalse(self.clock.warn(), "warn() is False once expired")

    def test_elapsed_never_goes_negative(self):
        self.clock.start()
        self.clock_time.advance(-50.0)
        self.assertEqual(self.clock.elapsed(), 0.0)

    def test_default_budget_leaves_room_in_a_ten_minute_match(self):
        default = match_logic.MatchClock()
        self.assertGreater(default.limit_s, 0)
        self.assertLess(default.limit_s, match_logic.MATCH_DURATION_S,
                        "the mission must stop before the 600 s match ends")

"""Handling of a cube that is OUTSIDE the manipulator's reach.

The failure this suite protects against: the field test used to discard every
detection above the calibrated near field (``v < 130 px``) as "far
wall/ceiling noise".  A cube standing further than ~35 cm therefore produced
``NO_TARGET`` (exit code 3) and the robot never drove towards it.

The replacement contract, pinned down here:

* a far pixel that passes the loose scan window is still a usable target, but
  it is flagged ``far`` and is NEVER considered in reach, so the base approach
  step runs;
* the dedicated Cake affine keeps its reach clamp for the final grasp plan
  (a Cake seen on its top face looks "too far"), while the far path keeps the
  unclamped ``x`` so a genuinely distant Cake also drives the base;
* near-field pixels keep working exactly as before;
* ``grasp_generic.py`` actually calls the tested helpers.

All coordinates use the shipped hand-eye matrix; the pixels are real values
measured on the field when the bug was reproduced.
"""

import io
import os
import sys
import unittest

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from _bootstrap import CODES_DIR, match_logic  # noqa: E402

try:
    import numpy as np
except ImportError:            # the robot has numpy; keep the suite importable
    np = None

CALIB_FILE = os.path.join(CODES_DIR, "transform_3d.dat")
GRASP_GENERIC = os.path.join(CODES_DIR, "grasp_generic.py")

#: Live detections of the failing run: a cube at ~0.65 m (v < 130) next to a
#: reachable cube (v = 294).  The first one used to be thrown away.
FAR_FLOOR_PIXEL = (345.0, 107.0)
NEAR_FLOOR_PIXEL = (346.0, 294.0)


def read(path):
    with io.open(path, encoding="utf-8") as handle:
        return handle.read()


class TestScanWindow(unittest.TestCase):
    """The scan window must not throw away an out-of-reach cube."""

    def test_recorded_far_cube_passes_the_scan_window(self):
        u, v = FAR_FLOOR_PIXEL
        self.assertTrue(match_logic.in_scan_window(u, v),
                        "an out-of-reach cube must reach the approach logic")
        self.assertFalse(match_logic.in_near_field(u, v),
                         "v=107 is outside the calibrated near field")

    def test_recorded_near_cube_is_in_the_near_field(self):
        u, v = NEAR_FLOOR_PIXEL
        self.assertTrue(match_logic.in_near_field(u, v))
        self.assertTrue(match_logic.in_scan_window(u, v))

    def test_the_window_contains_the_whole_near_field(self):
        """Every calibrated near-field pixel must also pass the loose window."""
        for u in (80.0, 320.0, 560.0):
            for v in (130.0, 270.0, 450.0):
                self.assertTrue(match_logic.in_scan_window(u, v),
                                "near-field pixel (%s, %s) was rejected" % (u, v))

    def test_image_edges_are_still_rejected(self):
        for u, v in ((5.0, 10.0), (630.0, 100.0), (320.0, 5.0), (320.0, 478.0)):
            self.assertFalse(match_logic.in_scan_window(u, v),
                             "edge pixel (%s, %s) must stay rejected" % (u, v))

    def test_the_window_is_a_strict_superset_of_the_near_field(self):
        near = (match_logic.NEAR_FIELD_U[0] >= match_logic.SCAN_WINDOW_U[0]
                and match_logic.NEAR_FIELD_U[1] <= match_logic.SCAN_WINDOW_U[1]
                and match_logic.NEAR_FIELD_V[0] >= match_logic.SCAN_WINDOW_V[0]
                and match_logic.NEAR_FIELD_V[1] <= match_logic.SCAN_WINDOW_V[1])
        self.assertTrue(near)


@unittest.skipUnless(np is not None and os.path.exists(CALIB_FILE),
                     "needs numpy and the shipped calibration file")
class TestFarFieldTarget(unittest.TestCase):
    """A far pixel must produce an approach point and be declared out of reach."""

    @classmethod
    def setUpClass(cls):
        cls.matrix = np.load(CALIB_FILE).tolist()

    def test_far_floor_pixel_is_a_valid_approach_point(self):
        xy = match_logic.far_target_xy(self.matrix, *FAR_FLOOR_PIXEL,
                                       is_cake=False)
        self.assertIsNotNone(xy)
        x, y = xy
        self.assertAlmostEqual(x, 410.8, delta=3.0)
        self.assertAlmostEqual(y, -19.6, delta=3.0)
        self.assertGreater(x, match_logic.FLOOR_IN_REACH["dist_max"],
                           "the far target must stay outside the grasp envelope")

    def test_far_flag_forces_a_base_approach(self):
        xy = match_logic.far_target_xy(self.matrix, *FAR_FLOOR_PIXEL,
                                       is_cake=False)
        in_reach, dist = match_logic.grasp_reach_state(
            "floor", xy[0], xy[1], far=True)
        self.assertFalse(in_reach)
        self.assertGreater(dist, match_logic.FLOOR_IN_REACH["dist_max"])

    def test_far_cake_keeps_its_distance_for_the_approach(self):
        """The reach clamp must NOT turn a distant Cake into a "reachable" one."""
        raw = match_logic.pixel_to_cake_xy(self.matrix, *FAR_FLOOR_PIXEL,
                                           clamp=False)
        clamped = match_logic.pixel_to_cake_xy(self.matrix, *FAR_FLOOR_PIXEL)
        self.assertGreater(raw[0], match_logic.CAKE_IN_REACH["dist_target"] + 50.0)
        self.assertLessEqual(clamped[0], 348.0)

def read(path):
    with io.open(path, encoding="utf-8") as handle:
        return handle.read()


def module_tree(path):
    return ast.parse(read(path), filename=path)


def defined_functions(path):
    tree = module_tree(path)
    return {node.name for node in ast.walk(tree)
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef))}


def imported_modules(path):
    names = set()
    for node in ast.walk(module_tree(path)):
        if isinstance(node, ast.Import):
            names.update(alias.name.split(".")[0] for alias in node.names)
        elif isinstance(node, ast.ImportFrom) and node.module:
            names.add(node.module.split(".")[0])
            names.update(alias.name for alias in node.names)
    return names


def describes_only_logic_of(production_function, test_paths):
    """Names a test file defines that shadow production logic."""
    findings = []
    for path in test_paths:
        if production_function in defined_functions(path):
            findings.append(os.path.basename(path))
    return findings
