#!/usr/bin/env python3
"""
Move the uArm end effector to an absolute Cartesian pose.

Prerequisite: arm driver running on the robot:
    roslaunch swiftpro pro_control_nomoveit.launch

Usage:
    source ~/spark_noetic/devel/setup.bash
    python3 ~/codes/arm_move.py --x 200 --y 0 --z 20
    python3 ~/codes/arm_move.py --home
    python3 ~/codes/arm_move.py --x 198.8 --y 0.6 --z -50

Coordinates are in the same units as ~/thefile.txt (millimetres).
Home preset: (120, 0, 35).
"""

import sys

import rospy
from swiftpro.msg import position

HOME = (120.0, 0.0, 35.0)


def parse_args(argv):
    args = {"x": None, "y": None, "z": None, "home": False}
    i = 1
    while i < len(argv):
        token = argv[i]
        if token == "--home":
            args["home"] = True
            i += 1
        elif token in ("--x", "--y", "--z") and i + 1 < len(argv):
            args[token[2:]] = float(argv[i + 1])
            i += 2
        elif ":=" in token or token.startswith("__"):
            # ROS remapping / private args: leave them for rospy
            i += 1
        else:
            raise SystemExit("unknown argument: %s" % token)
    if not args["home"] and None in (args["x"], args["y"], args["z"]):
        raise SystemExit("give --x --y --z, or --home")
    return args


def main():
    args = parse_args(sys.argv)
    if args["home"]:
        x, y, z = HOME
    else:
        x, y, z = args["x"], args["y"], args["z"]

    rospy.init_node("arm_move", anonymous=True, argv=[sys.argv[0]])
    pub = rospy.Publisher("/position_write_topic", position, queue_size=10)
    rospy.sleep(0.5)  # let the publisher connect

    pose = position()
    pose.x, pose.y, pose.z = x, y, z
    print("moving arm to x=%.1f y=%.1f z=%.1f" % (x, y, z))
    pub.publish(pose)
    rospy.sleep(0.5)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Live blue-box detector for the NXROBO Spark robot (ROS Noetic).

Subscribes to /camera/color/image_raw, applies the same HSV mask and
morphology used by spark_carry_object, and prints the largest contour's
area and centroid. Use it to check that the camera sees the box before
moving the arm.

Run on the robot (camera driver must already be running):

    source ~/spark_noetic/devel/setup.bash
    python3 ~/codes/blue_detect.py

Tune LOWER/UPPER if the box is not detected or other objects are.
The values below are the patched defaults used for the light-blue cube.
"""

import time

import cv2
import numpy as np
import rospy
from cv_bridge import CvBridge
from sensor_msgs.msg import Image

LOWER = np.array([90, 50, 50])      # H, S, V
UPPER = np.array([135, 255, 255])
MIN_AREA = 300
PRINT_HZ = 5.0


class BlueDetector:
    def __init__(self):
        self.bridge = CvBridge()
        self.kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (25, 25))
        self.last_print = 0.0
        self.latest = None

    def mask(self, bgr):
        hsv = cv2.cvtColor(bgr, cv2.COLOR_BGR2HSV)
        m = cv2.inRange(hsv, LOWER, UPPER)
        m = cv2.morphologyEx(m, cv2.MORPH_CLOSE, self.kernel)
        m = cv2.erode(m, None, iterations=4)
        m = cv2.dilate(m, None, iterations=4)
        return m

    def callback(self, msg):
        try:
            frame = self.bridge.imgmsg_to_cv2(msg, "bgr8")
        except Exception as exc:  # CvBridgeError and friends
            rospy.logwarn("image conversion failed: %s", exc)
            return

        m = self.mask(frame)
        contours, _ = cv2.findContours(m, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        contours = [c for c in contours if cv2.contourArea(c) > MIN_AREA]

        if contours:
            biggest = max(contours, key=cv2.contourArea)
            moment = cv2.moments(biggest)
            if moment["m00"] > 0:
                cx = moment["m10"] / moment["m00"]
                cy = moment["m01"] / moment["m00"]
                self.latest = (cv2.contourArea(biggest), cx, cy)
            else:
                self.latest = None
        else:
            self.latest = None

        now = time.time()
        if now - self.last_print >= 1.0 / PRINT_HZ:
            self.last_print = now
            if self.latest:
                area, cx, cy = self.latest
                print("BLUE area=%4d center=(%3d,%3d)" % (area, cx, cy))
            else:
                print("no blue box")

    def run(self):
        rospy.init_node("blue_detect", anonymous=True)
        rospy.Subscriber("/camera/color/image_raw", Image, self.callback, queue_size=1)
        rospy.loginfo("blue_detect ready; waiting for /camera/color/image_raw")
        rospy.spin()


if __name__ == "__main__":
    BlueDetector().run()

#!/usr/bin/env python3
"""
Save one camera frame plus its blue mask.

Useful when the detector reports "no blue box": copy the two PNG files to
your PC and look at them.

Run on the robot:

    source ~/spark_noetic/devel/setup.bash
    python3 ~/codes/blue_snap.py

Then from your PC:

    scp spark@10.25.141.8:/tmp/blue.png spark@10.25.141.8:/tmp/blue_mask.png .

Optional arguments:
    python3 blue_snap.py /tmp/my.png /tmp/my_mask.png
"""

import sys

import cv2
import numpy as np
import rospy
from cv_bridge import CvBridge
from sensor_msgs.msg import Image

LOWER = np.array([90, 50, 50])
UPPER = np.array([135, 255, 255])

IMAGE_PATH = sys.argv[1] if len(sys.argv) > 1 else "/tmp/blue.png"
MASK_PATH = sys.argv[2] if len(sys.argv) > 2 else "/tmp/blue_mask.png"


def main():
    rospy.init_node("blue_snap", anonymous=True, argv=[sys.argv[0]])
    msg = rospy.wait_for_message("/camera/color/image_raw", Image, timeout=5.0)
    frame = CvBridge().imgmsg_to_cv2(msg, "bgr8")

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
    mask = cv2.inRange(hsv, LOWER, UPPER)

    cv2.imwrite(IMAGE_PATH, frame)
    cv2.imwrite(MASK_PATH, mask)
    print("saved %s and %s" % (IMAGE_PATH, MASK_PATH))
    print("blue pixels: %d" % int(mask.sum() / 255))


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Replay the grasp node's detection pipeline frame by frame.

Prints, for each frame: number of contours, largest area, centroid and how
many consecutive frames satisfied the stability test. Use this when the
grasp demo keeps printing "not found".

Run on the robot:

    source ~/spark_noetic/devel/setup.bash
    python3 ~/codes/detect_pipeline_debug.py

The grasp demo (s_carry_object_cv3.py) was patched to require 10 stable
frames within 6 px. Original values were 20 frames / 2 px.
"""

import cv2
import numpy as np
import rospy
from cv_bridge import CvBridge
from sensor_msgs.msg import Image

LOWER = np.array([90, 50, 50])
UPPER = np.array([135, 255, 255])
FRAMES = 20
STABLE_PX = 6.0


def pipeline(frame, kernel):
    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
    mask = cv2.inRange(hsv, LOWER, UPPER)
    masked_hsv = cv2.bitwise_and(hsv, hsv, mask=mask)
    hue = masked_hsv[:, :, 0]
    blurred = cv2.blur(hue, (9, 9))
    _, thresh = cv2.threshold(blurred, 90, 255, cv2.THRESH_BINARY)
    thresh = cv2.morphologyEx(thresh, cv2.MORPH_CLOSE, kernel)
    thresh = cv2.erode(thresh, None, iterations=4)
    thresh = cv2.dilate(thresh, None, iterations=4)
    contours, _ = cv2.findContours(thresh, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)
    return contours


def main():
    rospy.init_node("detect_pipeline_debug", anonymous=True)
    bridge = CvBridge()
    kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (25, 25))
    frames = []

    def callback(msg):
        frames.append(bridge.imgmsg_to_cv2(msg, "bgr8"))

    subscriber = rospy.Subscriber("/camera/color/image_raw", Image, callback, queue_size=30)
    rospy.sleep(1.0)
    subscriber.unregister()

    print("captured %d frames" % len(frames))
    previous = None
    stable = 0
    for index, frame in enumerate(frames[-FRAMES:]):
        contours = pipeline(frame, kernel)
        if not contours:
            print("%2d no contours" % index)
            previous = None
            stable = 0
            continue
        biggest = max(contours, key=cv2.contourArea)
        moment = cv2.moments(biggest)
        if moment["m00"] == 0:
            continue
        cx = moment["m10"] / moment["m00"]
        cy = moment["m01"] / moment["m00"]
        if previous is not None and abs(cx - previous[0]) <= STABLE_PX and abs(cy - previous[1]) <= STABLE_PX:
            stable += 1
        else:
            stable = 0
        print("%2d n=%d area=%d centroid=(%.1f,%.1f) stable=%d"
              % (index, len(contours), cv2.contourArea(biggest), cx, cy, stable))
        previous = (cx, cy)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Detect the blue box and move the uArm suction cup to touch its top centre.

Pipeline:
  camera frame -> HSV blue mask -> largest contour centroid (xc, yc)
  -> pixel-to-arm mapping from ~/thefile.txt
  -> publish two poses on /position_write_topic (hover, then touch)
  -> return home.

Prerequisites on the robot:
  - camera driver running:
        roslaunch spark_bringup driver_bringup.launch camera_type_tel:=d435 lidar_type_tel:=ydlidar_g6
  - arm driver running:
        roslaunch swiftpro pro_control_nomoveit.launch
  - ~/thefile.txt exists (run ./onekey.sh -> 8 -> 2)

Run:
    source ~/spark_noetic/devel/setup.bash
    python3 ~/codes/touch_blue.py            # move and touch
    python3 ~/codes/touch_blue.py --dry-run  # print coordinates only

The box must be inside the arm's reach. Keep hands clear; the arm moves
directly to the computed point.
"""

import sys

import cv2
import numpy as np
import rospy
from cv_bridge import CvBridge
from sensor_msgs.msg import Image
from swiftpro.msg import position

CALIBRATION_FILE = "/home/spark/thefile.txt"

LOWER = np.array([90, 50, 50])
UPPER = np.array([135, 255, 255])
MIN_AREA = 300
FRAMES = 10          # frames averaged before moving

HOVER_Z = 20.0       # above the box
TOUCH_Z = -50.0      # down onto the box (try -40 / -60 if needed)
HOME_X, HOME_Y, HOME_Z = 120.0, 0.0, 35.0

DRY_RUN = "--dry-run" in sys.argv
ROS_ARGV = [arg for arg in sys.argv if arg != "--dry-run"]


def read_calibration(path):
    """Return (a1, b1, a3, b3): x = a1*yc + b1, y = a3*xc + b3."""
    with open(path) as handle:
        values = [float(v) for v in handle.read().split()]
    if len(values) < 4:
        raise RuntimeError("calibration file needs 4 values: %s" % path)
    return values[0], values[1], values[2], values[3]


def detect(frame, kernel):
    mask = cv2.inRange(cv2.cvtColor(frame, cv2.COLOR_BGR2HSV), LOWER, UPPER)
    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)
    mask = cv2.dilate(cv2.erode(mask, None, iterations=4), None, iterations=4)
    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    contours = [c for c in contours if cv2.contourArea(c) > MIN_AREA]
    if not contours:
        return None
    biggest = max(contours, key=cv2.contourArea)
    moment = cv2.moments(biggest)
    if moment["m00"] == 0:
        return None
    return moment["m10"] / moment["m00"], moment["m01"] / moment["m00"]


def main():
    a1, b1, a3, b3 = read_calibration(CALIBRATION_FILE)
    print("calibration: x=%.5f*yc+%.2f  y=%.5f*xc+%.2f" % (a1, b1, a3, b3))

    rospy.init_node("touch_blue", anonymous=True, argv=ROS_ARGV)
    pub = rospy.Publisher("/position_write_topic", position, queue_size=10)
    bridge = CvBridge()
    kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (25, 25))

    xs, ys = [], []
    for _ in range(FRAMES):
        msg = rospy.wait_for_message("/camera/color/image_raw", Image, timeout=5.0)
        result = detect(bridge.imgmsg_to_cv2(msg, "bgr8"), kernel)
        if result:
            xs.append(result[0])
            ys.append(result[1])

    if not xs:
        print("no blue box")
        return 1

    xc = sum(xs) / len(xs)
    yc = sum(ys) / len(ys)
    arm_x = a1 * yc + b1
    arm_y = a3 * xc + b3
    print("pixel=(%.0f,%.0f)  arm=(%.1f,%.1f)" % (xc, yc, arm_x, arm_y))

    if DRY_RUN:
        print("dry run: not moving")
        return 0

    pose = position()
    pose.x, pose.y = arm_x, arm_y

    pose.z = HOVER_Z
    pub.publish(pose)
    rospy.sleep(1.5)

    pose.z = TOUCH_Z
    pub.publish(pose)
    rospy.sleep(2.0)
    print("touched - going home")

    pose.x, pose.y, pose.z = HOME_X, HOME_Y, HOME_Z
    pub.publish(pose)
    rospy.sleep(2.0)
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""机械臂可达性判定 + 像素↔机械臂坐标换算（纯数学，无 ROS 依赖）。

被两个地方用：
* ``nodes/grasp.py`` —— 抓取前先判一次，够不到就别发（见那边的说明）
* ``test/reachable_area.py`` —— 算出"画面里哪些区域够得到"，指导摆放

数学照抄 ``swiftpro_rviz_node.cpp`` 的 ``swiftpro_ik()``：够不到时内部两个
``acos`` 的参数会超出 [-1, 1]（三角形两边夹不出这个点），那边就报
``Inverse kinematic is wrong``。实测对齐：报错那次的 (67.18, 22.28, 20) 判不可达，
日志里真正抓成功的 (126.42, -8.58) 与 (100.78, -2.77) 判可达。

**别在这里加 ROS 依赖** —— 这个模块要能在没有 roscore 的机器上直接跑。
"""

import math

# swiftpro 的几何常数（swiftpro_rviz_node.cpp 顶部）
MATH_TRANS = 57.2958
MATH_L1 = 106.6
MATH_L2 = 13.2
MATH_LOWER_ARM = 142.07
MATH_UPPER_ARM = 158.81
MATH_UPPER_LOWER = MATH_UPPER_ARM / MATH_LOWER_ARM

# 手眼标定系数（板上 /data/local/release/usr/thefile.txt）
# pos.x 由**像素行**算，pos.y 由**像素列**算 —— 原程序就是这样反的，别"顺手改正"。
DEFAULT_X_KB = (-0.8840816286146583, 448.66407041831445)
DEFAULT_Y_KB = (-0.8946292902364896, 315.7187464490772)


def pixel_to_arm(pixel_row, pixel_col, x_kb=DEFAULT_X_KB, y_kb=DEFAULT_Y_KB):
    """像素 -> 机械臂坐标（毫米），与 grasp() 里的算法逐字一致。"""
    return x_kb[0] * pixel_row + x_kb[1], y_kb[0] * pixel_col + y_kb[1]


def ik_reachable(x, y, z):
    """(x, y, z)（毫米）这个位姿机械臂够得到吗？"""
    z = z + 74.55
    z_in = (z - MATH_L1) / MATH_LOWER_ARM
    if x < 0.1:
        x = 0.1
    if y == 0:
        rot = 90.0
    elif y < 0:
        rot = -math.atan(x / y) * MATH_TRANS
    else:
        rot = 180.0 - math.atan(x / y) * MATH_TRANS

    sin_rot = math.sin(rot / MATH_TRANS)
    if sin_rot == 0:
        return False
    x_in = (x / sin_rot - MATH_L2 - 56.55) / MATH_LOWER_ARM
    if x_in == 0:
        return False
    sqrt_zx = math.sqrt(z_in * z_in + x_in * x_in)
    if sqrt_zx == 0:
        return False
    upper = MATH_UPPER_LOWER
    right_all = (sqrt_zx * sqrt_zx + upper * upper - 1) / (2 * upper * sqrt_zx)
    left_all = (sqrt_zx * sqrt_zx + 1 - upper * upper) / (2 * sqrt_zx)
    return abs(right_all) <= 1 and abs(left_all) <= 1


def pixel_reachable(pixel_row, pixel_col, z_list, x_kb=DEFAULT_X_KB, y_kb=DEFAULT_Y_KB):
    """这个像素对应的位置，在 z_list 里**每个**高度都够得到才算可达。"""
    ax, ay = pixel_to_arm(pixel_row, pixel_col, x_kb, y_kb)
    return all(ik_reachable(ax, ay, z) for z in z_list)

    # -- 每帧 ----------------------------------------------------------
    def image_cb(self, msg):
        try:
            image = self.bridge.imgmsg_to_cv2(msg, "bgr8")
        except CvBridgeError as exc:
            rospy.logerr_throttle(5.0, "[detect_rknn] 图像转换失败: %s", exc)
            return

        roi_image = self._crop(image)
        if roi_image is None:
            rospy.logerr_throttle(
                5.0, "[detect_rknn] 图像 %s 装不下兴趣区 %s，请改 ~roi",
                image.shape, self.roi)
            return

        try:
            detections = self.detector.infer(roi_image)
        except Exception as exc:            # 单帧异常不该打死节点
            rospy.logerr_throttle(5.0, "[detect_rknn] 推理失败: %s", exc)
            return

        for det in detections:
            self.location_pub.publish(self._build_location(det))

        if self.publish_debug_image:
            draw_detections(roi_image, detections)
            try:
                out = self.bridge.cv2_to_imgmsg(roi_image, "bgr8")
            except CvBridgeError as exc:
                rospy.logerr_throttle(5.0, "[detect_rknn] 调试图转换失败: %s", exc)
            else:
                out.header = msg.header
                self.image_pub.publish(out)

        self._record_fps()

    def _crop(self, image):
        """按 ``~roi`` 裁出兴趣区；画面比 ROI 小就返回 None（不猜、不补边）。"""
        window = crop_slice(image.shape, self.roi)
        if window is None:
            return None
        return image[window]

    def _build_location(self, det):
        """把 ROI 内的框中心换算回整幅图像坐标，发成原程序的 LocationCentre。

        原版是 ``int((x_min + x_max) * roi_w / 2 + roi_x)``，即
        "ROI 内中心 + ROI 原点"；换算细节见 ``nodes/roi_map.py``。
        """
        cx, cy = centre_to_source(det.x1, det.y1, det.x2, det.y2, self.roi_offset)
        loc = LocationCentre()
        loc.obj_id = int(det.class_id)
        loc.loc_centre_x = cx
        loc.loc_centre_y = cy
        return loc

    def _record_fps(self):
        now = time.time()
        self._stamps.append(now)
        if len(self._stamps) > FPS_WINDOW:
            del self._stamps[:-FPS_WINDOW]
        if len(self._stamps) < 2:
            return
        span = self._stamps[-1] - self._stamps[0]
        fps = (len(self._stamps) - 1) / span if span > 0 else 0.0
        rospy.loginfo_throttle(
            2.0, "[detect_rknn] 端到端 %.1f FPS | NPU %.1f ms | 后处理 %.1f ms",
            fps, self.detector.last_infer_ms, self.detector.last_post_ms)


def main(argv):
    rospy.init_node("detect_rknn", argv=argv)
    try:
        RknnDetectorNode()
    except Exception as exc:
        # 模型/运行时版本不匹配这类问题必须立刻暴露，而不是静默发空消息。
        rospy.logfatal("[detect_rknn] 初始化失败: %s", exc)
        return 1
    try:
        rospy.spin()
    except KeyboardInterrupt:
        print("ShutDown")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))

def resolve_calib_file():
    """决定用哪个标定文件：``~calib_file`` > 环境变量 > HOME > 本板落点。"""
    candidates = []
    explicit = rospy.get_param("~calib_file", "")
    if explicit:
        candidates.append(explicit)
    env = os.environ.get("SEMI_AUTO_MATCH_CALIB")
    if env:
        candidates.append(env)
    candidates.append(HOME_CALIB_FILE)
    candidates.append(BOARD_CALIB_FILE)
    for path in candidates:
        if path and os.path.exists(path):
            return path
    raise RuntimeError(
        "找不到手眼标定文件；找过: %s。用 ~calib_file:=<路径> 指定，"
        "或设 SEMI_AUTO_MATCH_CALIB，或先按原流程做一次标定生成它" % (candidates,))

            self.grasp(self.broccoli_x,self.broccoli_y, msg.data)
            self.broccoli_found_count = 0

        # 接受到进行放置的信号
        else:

"""通用 YOLO 检测运行时（RK3588 NPU，基于 ``rknn`` 兼容层）。

本模块与 ROS 无关，可以单独在板子上跑单测/基准：
``run python3 rknn_yolo.py <model.rknn> <bus.jpg>``。

支持的导出头
------------
* **v5 系**（yolov5 / v6 / v7）：3 个带 anchor 的尺度头，通道数 = na*(5+nc)，
  含 objectness 通道；输出可能是 raw logits（自动识别并补 sigmoid）。
* **v8 系**（yolov8 / v9 / v11）：单头 ``(1, 4+nc, N)`` 或 ``(1, N, 4+nc)``，
  无 objectness、无 anchor，xywh 已经是输入图像像素。

DFL 未解码的 6 输出头（3×box + 3×cls）不做支持，遇到会给出明确报错而不是猜。

解码的两条硬约定
----------------
1. **先过滤后解码**：阈值判断只跑 objectness（v5）或类分数（v8）这一个通道，
   只对过阈值的候选做 sigmoid 与框解码。25200 个 anchor 全量解码是纯浪费，
   而 ``conf = class_max * objectness <= objectness``，所以先按 objectness
   过滤不会漏框（数学等价，不是近似）。
2. **框一律用 xyxy 表示**，NMS 交集直接算 ``min(x2) - max(x1)``；
   不要退回 (x, y, w, h) 那种写法——那里漏写右下角的 ``+w``/``+h``
   会让 IoU 恒为 0，NMS 静默失效（见 SKILL 铁律 8）。

def _iter_data_lines(path):
    with open(path, "r") as handle:
        for lineno, raw in enumerate(handle, 1):
            line = raw.split("#", 1)[0].strip()
            if line:
                yield lineno, line


def _is_int(text):
    try:
        int(text)
    except ValueError:
        return False
    return True


def _parse_label_line(line):
    """解析一行标签，返回 ``(模型下标|None, 外部id|None, 名称)``。

    下面几种写法都支持，名称允许带空格（取剩余部分，不会把 ``wine glass``
    截成 ``glass``）：

    * ``person``           —— 只有名字
    * ``0 person``         —— 模型类别下标 + 名字
    * ``0 1 person``       —— 模型下标 + 外部 id + 名字（本包配置的格式）
    * ``0 1 wine glass``   —— 名字带空格
    """
    parts = line.split(None, 2)
    if len(parts) == 3 and _is_int(parts[0]) and _is_int(parts[1]):
        return int(parts[0]), int(parts[1]), parts[2]
    if len(parts) >= 2 and _is_int(parts[0]):
        return int(parts[0]), None, line.split(None, 1)[1]
    return None, None, line


def load_labels(path):
    """读类别名列表，一行一个（格式见 :func:`_parse_label_line`）。"""
    if not path:
        return None
    names = []
    for lineno, line in _iter_data_lines(path):
        name = _parse_label_line(line)[2]
        if not name:
            raise ValueError("%s:%d 解析不出类别名" % (path, lineno))
        names.append(name)
    if not names:
        raise ValueError("标签文件是空的: %s" % path)
    return names


def load_id_map(path):
    """读"模型类别下标 → 对外 id"的映射表。

    要求**每行都带下标与外部 id**（``下标 外部id 名字``），否则返回 ``None``
    表示这份标签文件不做映射。返回的列表下标即模型类别下标。
    """
    if not path:
        return None
    rows = []
    for _, line in _iter_data_lines(path):
        index, external, _ = _parse_label_line(line)
        if index is None or external is None:
            return None
        rows.append((index, external))
    if not rows:
        return None
    rows.sort()
    if [index for index, _ in rows] != list(range(len(rows))):
        raise ValueError("%s 的模型类别下标必须是 0..%d 连续" % (path, len(rows) - 1))
    return [external for _, external in rows]

class LetterBox(object):
    """等比缩放 + 居中填充，映射规则与 rknn_model_zoo 的 COCO_test_helper 一致。

    ``to_source`` 把解码出来的框从"填充后的输入空间"搬回"原图像素"。
    """

    def __init__(self, new_shape=(640, 640), pad_value=(0, 0, 0)):
        self.new_shape = (int(new_shape[0]), int(new_shape[1]))  # (h, w)
        self.pad_value = pad_value
        self.ratio = 1.0
        self.dw = 0.0
        self.dh = 0.0

    def apply(self, img):
        h, w = img.shape[:2]
        new_h, new_w = self.new_shape
        ratio = min(float(new_h) / h, float(new_w) / w)
        new_unpad = (int(round(w * ratio)), int(round(h * ratio)))
        dw = (new_w - new_unpad[0]) / 2.0
        dh = (new_h - new_unpad[1]) / 2.0
        if (w, h) != new_unpad:
            img = cv2.resize(img, new_unpad, interpolation=cv2.INTER_LINEAR)
        top, bottom = int(round(dh - 0.1)), int(round(dh + 0.1))
        left, right = int(round(dw - 0.1)), int(round(dw + 0.1))
        img = cv2.copyMakeBorder(img, top, bottom, left, right,
                                 cv2.BORDER_CONSTANT, value=self.pad_value)
        self.ratio, self.dw, self.dh = ratio, dw, dh
        return img

    def to_source(self, boxes):
        out = np.array(boxes, dtype=np.float32, copy=True)
        out[:, [0, 2]] = (out[:, [0, 2]] - self.dw) / self.ratio
        out[:, [1, 3]] = (out[:, [1, 3]] - self.dh) / self.ratio
        return out

# ------------------------------------------------------------------ 检测器 ----

class RknnYoloDetector(object):
    """把 ``.rknn`` 模型包成 ``infer(bgr) -> [Detection]``。

    :param model_path: ``.rknn`` 模型路径
    :param labels: 类别名列表，或标签文件路径（见 :func:`load_labels`）
    :param anchors: anchor 列表 ``[[[w,h], ...], ...]``（每个尺度一组），
                    v5 系必需；给文件路径时按每行一个数字读，形状 ``(n_scale, -1, 2)``
    :param img_size: ``(w, h)``；None 表示从模型输入属性里读
    :param id_map: 模型类别下标 → 对外 id 的映射；None 表示不映射
    :param head: ``auto`` / ``v5`` / ``v8``
    :param sigmoid: ``auto`` / ``on`` / ``off``；只作用于 v5 系的解码
    """

    def __init__(self, model_path, labels=None, anchors=None, lib_path=None,
                 img_size=None, conf_thres=0.25, nms_thres=0.45,
                 core_mask="0_1_2", head="auto", sigmoid="auto",
                 max_det=100, id_map=None, verbose=False, warmup=True):
        self.model_path = model_path
        self.conf_thres = float(conf_thres)
        self.nms_thres = float(nms_thres)
        self.head = head
        self.sigmoid = sigmoid
        self.max_det = int(max_det)
        self.verbose = bool(verbose)
        self.lib_path = lib_path

        self.labels = load_labels(labels) if isinstance(labels, str) else labels
        self.id_map = list(id_map) if id_map is not None else None
        self.anchors = self._read_anchors(anchors)

        self.last_infer_ms = 0.0
        self.last_post_ms = 0.0
        self.frames = 0
        self.head_kind = None
        self.num_classes = None

        self._rknn = RKNNLite(verbose=self.verbose, lib_path=lib_path)
        ret = self._rknn.load_rknn(model_path)
        if ret != 0:
            raise RuntimeError("load_rknn(%s) 失败: %s" % (model_path, error_name(ret)))
        self._rknn.init_runtime(core_mask=core_mask_from_string(core_mask))

        if img_size is None:
            img_size = self._detect_input_size()
        self.img_size = (int(img_size[0]), int(img_size[1]))    # (w, h)

        if warmup:
            self._warmup()

    def _warmup(self):
        """用一张空图跑一次完整前向。

        两个作用：把模型能跑通这件事在**节点启动时**就验证掉（而不是等第一帧
        图像才发现模型/运行时不对），以及把 head 类型与类别数定下来，
        好让启动日志里直接看到真实的 nc —— 类别数对不上是最常见的错配。
        """
        blank = np.zeros((self.img_size[1], self.img_size[0], 3), dtype=np.uint8)
        self.infer(blank)
        self.frames = 0
        self.last_infer_ms = 0.0
        self.last_post_ms = 0.0

    # -- 初始化辅助 ----------------------------------------------------
    @staticmethod
    def _read_anchors(anchors):
        if anchors is None:
            return None
        if isinstance(anchors, str):
            values = []
            for _, line in _iter_data_lines(anchors):
                values.append(float(line.split()[0]))
            return np.array(values, dtype=np.float32).reshape(3, -1, 2).tolist()
        return np.array(anchors, dtype=np.float32).tolist()

    def _detect_input_size(self):
        """从模型输入属性里读输入尺寸；读不到就报错让人显式给 img_size。"""
        attrs = getattr(self._rknn, "input_attrs", None)
        if not attrs:
            raise ValueError("拿不到模型输入属性，请显式指定 img_size=(w, h)")
        shape = attrs[0].shape
        if len(shape) != 4:
            raise ValueError("输入维度不是 4 维: %s，请显式指定 img_size" % (shape,))
        # NHWC: (1, H, W, C)；NCHW: (1, C, H, W)
        if attrs[0].is_nhwc or shape[-1] in (1, 3, 4):
            return (int(shape[2]), int(shape[1]))
        return (int(shape[3]), int(shape[2]))

    @property
    def runtime_versions(self):
        return (self._rknn.sdk_version, self._rknn.driver_version)

    def describe(self):
        return ("model=%s api=%s driver=%s input=%dx%d head=%s nc=%s conf=%.2f nms=%.2f"
                % (os.path.basename(self.model_path), self._rknn.sdk_version,
                   self._rknn.driver_version, self.img_size[0], self.img_size[1],
                   self.head_kind or self.head, self.num_classes,
                   self.conf_thres, self.nms_thres))

    # -- 推理 ----------------------------------------------------------
    def infer(self, bgr_image):
        """对一帧 BGR uint8 图像做检测。

        :returns: ``[Detection]``，坐标已反映射回原图像素，按分数降序。
        """
        if bgr_image is None or getattr(bgr_image, "size", 0) == 0:
            return []

        box = LetterBox((self.img_size[1], self.img_size[0]))
        padded = box.apply(bgr_image)
        # 驱动按模型元数据做 mean/std/rgb2bgr，这里只负责把 BGR 转成模型要的 RGB。
        rgb = np.ascontiguousarray(cv2.cvtColor(padded, cv2.COLOR_BGR2RGB))

        start = time.time()
        outputs = self._rknn.inference(inputs=[rgb], data_type="uint8",
                                       data_format="nhwc")
        self.last_infer_ms = (time.time() - start) * 1000.0
        self.frames += 1

        post_start = time.time()
        kind = self._resolve_head_kind(outputs)
        if kind == "v5":
            boxes, classes, scores = self._post_process_v5(outputs)
        else:
            boxes, classes, scores = self._post_process_v8(outputs)

        detections = []
        if boxes is not None and len(boxes):
            real = box.to_source(boxes)
            height, width = bgr_image.shape[:2]
            order = np.argsort(-scores)[:self.max_det]
            for idx in order:
                x1, y1, x2, y2 = real[idx]
                x1 = int(round(min(max(x1, 0), width - 1)))
                y1 = int(round(min(max(y1, 0), height - 1)))
                x2 = int(round(min(max(x2, 0), width - 1)))
                y2 = int(round(min(max(y2, 0), height - 1)))
                if x2 <= x1 or y2 <= y1:
                    continue
                raw_id = int(classes[idx])
                class_id = self.id_map[raw_id] if self.id_map else raw_id
                detections.append(Detection(class_id, self._class_name(raw_id),
                                            float(scores[idx]), x1, y1, x2, y2))
        self.last_post_ms = (time.time() - post_start) * 1000.0
        return detections

    def _class_name(self, raw_id):
        if self.labels and 0 <= raw_id < len(self.labels):
            return self.labels[raw_id]
        return str(raw_id)

    # -- head 判定 -----------------------------------------------------
    def _resolve_head_kind(self, outputs):
        if self.head in ("v5", "v8"):
            kind = self.head
        else:
            shapes = [tuple(o.shape) for o in outputs]
            if len(outputs) == 1 and len(shapes[0]) == 3:
                kind = "v8"
            elif len(outputs) >= 3 and all(len(s) == 4 for s in shapes):
                kind = "v5"
            elif len(outputs) == 6:
                raise NotImplementedError(
                    "这个模型导出的是未做 DFL 解码的 6 输出头 %s；"
                    "请在导出 .rknn 时打开 DFL 解码（rknn_model_zoo 的 "
                    "YOLOv8 导出脚本默认会解码），或改用单头模型" % (shapes,)
                )
            else:
                raise ValueError(
                    "认不出模型输出布局 %s；用 head='v5' / head='v8' 显式指定"
                    % (shapes,)
                )
        self.head_kind = kind
        return kind

    # -- v5 系解码 -----------------------------------------------------
    def _post_process_v5(self, outputs):
        if not self.anchors:
            raise ValueError("v5 系模型需要 anchors，请给 anchors= 文件路径或列表")

        all_boxes, all_classes, all_scores = [], [], []

        for level, out in enumerate(outputs):
            data = self._as_chw(out, "v5 尺度头 %d" % level)
            if level >= len(self.anchors):
                raise ValueError(
                    "模型有 %d 个输出头，但只给了 %d 组 anchors"
                    % (len(outputs), len(self.anchors))
                )
            anchors = np.asarray(self.anchors[level], dtype=np.float32)
            n_anchor = anchors.shape[0]
            if data.shape[0] % n_anchor != 0:
                raise ValueError(
                    "尺度 %d 的通道数 %d 不能被 anchor 数 %d 整除"
                    % (level, data.shape[0], n_anchor)
                )
            channels = data.shape[0] // n_anchor
            n_classes = channels - 5
            if n_classes < 1:
                raise ValueError("尺度 %d 的通道数 %d 太少，不像 YOLO 头"
                                 % (level, data.shape[0]))
            self._remember_classes(n_classes)

            grid_h, grid_w = data.shape[1], data.shape[2]
            data = data.reshape(n_anchor, channels, grid_h, grid_w)

            # sigmoid 约定**每个尺度头判定一次**（只看 objectness 通道），
            # 判定结果对这个头的所有元素一致生效 —— 否则同一个头里
            # "objectness 走原值、类别走 sigmoid"这种混搭会静默出错。
            sig = self._sigmoid_for(data)

            # 只按 objectness 过滤：conf = class_max * obj <= obj，
            # 所以不会漏掉任何最终能过阈值的候选。
            objectness = sig(data[:, 4, :, :])
            a_idx, i_idx, j_idx = np.nonzero(objectness >= self.conf_thres)
            if a_idx.size == 0:
                continue

            probs = sig(data[a_idx, :, i_idx, j_idx])          # (K, channels)
            class_probs = probs[:, 5:]
            class_max = class_probs.max(axis=1)
            class_id = class_probs.argmax(axis=1)
            conf = class_max * probs[:, 4]

            keep = conf >= self.conf_thres
            if not keep.any():
                continue
            probs, class_id, conf = probs[keep], class_id[keep], conf[keep]
            a_k, i_k, j_k = a_idx[keep], i_idx[keep], j_idx[keep]

            stride_w = float(self.img_size[0]) / grid_w
            stride_h = float(self.img_size[1]) / grid_h
            anchor_wh = anchors[a_k]

            cx = (probs[:, 0] * 2.0 - 0.5 + j_k) * stride_w
            cy = (probs[:, 1] * 2.0 - 0.5 + i_k) * stride_h
            bw = np.power(probs[:, 2] * 2.0, 2) * anchor_wh[:, 0]
            bh = np.power(probs[:, 3] * 2.0, 2) * anchor_wh[:, 1]

            all_boxes.append(np.stack([cx - bw / 2.0, cy - bh / 2.0,
                                       cx + bw / 2.0, cy + bh / 2.0], axis=1))
            all_classes.append(class_id.astype(np.int64))
            all_scores.append(conf)

        if not all_boxes:
            return None, None, None
        return self._nms_per_class(np.concatenate(all_boxes),
                                   np.concatenate(all_classes),
                                   np.concatenate(all_scores))

    # -- v8 系解码 -----------------------------------------------------
    def _post_process_v8(self, outputs):
        if len(outputs) != 1:
            raise NotImplementedError(
                "v8 系只支持单头输出，实际有 %d 个头" % len(outputs)
            )
        pred = self._as_head_matrix(outputs[0], "v8 头")
        if pred.ndim != 2:
            raise ValueError("v8 头期望 2 维 (4+nc, N)，实际 %s" % (pred.shape,))
        if pred.shape[0] > pred.shape[1]:
            pred = pred.T                       # (N, 4+nc) → (4+nc, N)
        n_classes = pred.shape[0] - 4
        if n_classes < 1:
            raise ValueError("v8 头通道数 %d 太少" % pred.shape[0])
        self._remember_classes(n_classes)

        xywh = pred[:4].T                       # (N, 4)，输入图像像素
        scores = pred[4:].T                     # (N, nc)
        if self.sigmoid == "on":
            scores = 1.0 / (1.0 + np.exp(-scores))
        elif self.sigmoid == "auto" and (scores.min() < -0.01 or scores.max() > 1.01):
            scores = 1.0 / (1.0 + np.exp(-scores))

        class_id = scores.argmax(axis=1)
        conf = scores[np.arange(scores.shape[0]), class_id]
        keep = conf >= self.conf_thres
        if not keep.any():
            return None, None, None
        xywh, class_id, conf = xywh[keep], class_id[keep], conf[keep]

        cx, cy, bw, bh = xywh[:, 0], xywh[:, 1], xywh[:, 2], xywh[:, 3]
        boxes = np.stack([cx - bw / 2.0, cy - bh / 2.0,
                          cx + bw / 2.0, cy + bh / 2.0], axis=1)
        return self._nms_per_class(boxes, class_id.astype(np.int64), conf)

    # -- 公共小工具 ----------------------------------------------------
    def _sigmoid_for(self, head):
        """决定这个尺度头要不要补 sigmoid，返回对应的逐元素函数。

        只看 objectness 通道（几万个值）而不是整个头（几十万个值）；
        阈值取 [-0.01, 1.01]：概率一定落在这个区间内，日志值几乎不会。
        """
        if self.sigmoid == "on":
            apply = True
        elif self.sigmoid == "off":
            apply = False
        else:
            obj = head[:, 4, :, :]
            apply = bool(obj.min() < -0.01 or obj.max() > 1.01)
        if apply:
            return lambda x: 1.0 / (1.0 + np.exp(-x))
        return lambda x: x

    def _remember_classes(self, n_classes):
        if self.num_classes is None:
            self.num_classes = n_classes
        elif self.num_classes != n_classes:
            raise ValueError("各输出头的类别数不一致: %d vs %d"
                             % (self.num_classes, n_classes))
        if self.labels and len(self.labels) != n_classes and self.verbose:
            print("rknn_yolo: 警告 标签数 %d 与模型类别数 %d 不一致"
                  % (len(self.labels), n_classes))

    @staticmethod
    def _as_chw(out, what):
        """去掉 batch 维：``(1, C, H, W) -> (C, H, W)``；已经是低维就原样返回。"""
        if out.ndim == 4:
            if out.shape[0] != 1:
                raise ValueError("%s 期望 batch=1，实际 %s" % (what, out.shape))
            return out[0]
        if out.ndim in (2, 3):
            return out
        raise ValueError("%s 的维度不支持: %s" % (what, out.shape))

    @staticmethod
    def _as_head_matrix(out, what):
        """v8 头是 ``(1, 4+nc, N)``：去掉 batch 维，得到 2 维矩阵。

        不能和 :meth:`_as_chw` 共用 —— 那边的 3 维表示"(C, H, W) 无 batch"，
        这边的 3 维表示"(1, C, N) 有 batch"，同一个形状两种含义。
        """
        if out.ndim == 3:
            if out.shape[0] != 1:
                raise ValueError("%s 期望 batch=1，实际 %s" % (what, out.shape))
            return out[0]
        if out.ndim == 2:
            return out
        raise ValueError("%s 期望 (1, 4+nc, N) / (4+nc, N)，实际 %s" % (what, out.shape))

    def _nms_per_class(self, boxes, classes, scores):
        keep_boxes, keep_classes, keep_scores = [], [], []
        for cls in np.unique(classes):
            idx = np.where(classes == cls)[0]
            keep = self._nms_boxes(boxes[idx], scores[idx])
            if keep.size:
                keep_boxes.append(boxes[idx][keep])
                keep_classes.append(classes[idx][keep])
                keep_scores.append(scores[idx][keep])
        if not keep_boxes:
            return None, None, None
        return (np.concatenate(keep_boxes),
                np.concatenate(keep_classes),
                np.concatenate(keep_scores))

    def _nms_boxes(self, boxes, scores):
        """标准 IoU NMS；boxes 为 xyxy。"""
        x1, y1, x2, y2 = boxes[:, 0], boxes[:, 1], boxes[:, 2], boxes[:, 3]
        areas = (x2 - x1) * (y2 - y1)
        order = scores.argsort()[::-1]
        keep = []
        while order.size > 0:
            current = order[0]
            keep.append(current)
            rest = order[1:]
            if rest.size == 0:
                break
            xx1 = np.maximum(x1[current], x1[rest])
            yy1 = np.maximum(y1[current], y1[rest])
            xx2 = np.minimum(x2[current], x2[rest])
            yy2 = np.minimum(y2[current], y2[rest])
            inter = np.maximum(0.0, xx2 - xx1) * np.maximum(0.0, yy2 - yy1)
            union = areas[current] + areas[rest] - inter
            iou = np.where(union > 0, inter / np.maximum(union, 1e-12), 0.0)
            order = rest[iou <= self.nms_thres]
        return np.asarray(keep, dtype=np.int64)

    # -- 收尾 ----------------------------------------------------------
    def release(self):
        if self._rknn is not None:
            self._rknn.release()

    def __del__(self):
        try:
            self.release()
        except Exception:
            pass

"""兴趣区（ROI）与整幅图像坐标之间的换算 —— 本赛项的坐标契约。

为什么单独一个模块
------------------
机械臂标定文件 ``thefile.txt`` 给出的是**整幅图像像素坐标**到机械臂坐标的线性映射
（``pos.x = x_kb[0] * 像素x + x_kb[1]``，y 同理）。而检测器只看画面里的一小块
兴趣区，所以每一条发出去的 ``LocationCentre`` 都必须把"ROI 内的框中心"换算回
"整幅图像坐标"。

这层换算是纯数学、与 ROS 无关，但错了不会报错、只会让机械臂少走或多走几厘米。
所以把它从节点里拆出来：只依赖标准库，可以在任何地方单独跑
（见 ``test/test_rknn_yolo_postprocess.py``），不需要 numpy / cv2 / NPU。

与原程序的对应关系
------------------
原 ``scripts/detect.py`` 写的是::

    image = image[160:480, 160:480]                     # ROI 320×320，原点 (160, 160)
    loc_x = int((x_min + x_max) * 320 / 2 + 0.25 * 640) # 0.25*640 = 160  == ROI 原点 x
    loc_y = int((y_min + y_max) * 320 / 2 + 0.33 * 480) # 0.33*480 = 158.4 != ROI 原点 y(160)

x 方向的偏移正好是 ROI 原点，y 方向则比真实裁剪原点少 1.6 px。本模块按**真实裁剪
原点**换算；要逐像素复现原程序，把偏移显式设成 ``[160, 158.4]`` 即可。
"""

__all__ = ["parse_roi", "parse_offset", "centre_to_source", "crop_slice"]


def parse_roi(value):
    """``~roi`` → ``(x, y, w, h)``，单位像素，必须是 4 个整数。"""
    try:
        roi = [int(v) for v in value]
    except (TypeError, ValueError):
        raise ValueError("roi 必须是 4 个整数 (x, y, w, h)，收到 %r" % (value,))
    if len(roi) != 4:
        raise ValueError("roi 必须是 4 个整数 (x, y, w, h)，收到 %r" % (value,))
    if roi[2] <= 0 or roi[3] <= 0:
        raise ValueError("roi 的宽高必须为正，收到 %r" % (value,))
    return tuple(roi)


def parse_offset(value, roi):
    """``~roi_offset`` → ``(dx, dy)``；留空时用 ROI 左上角。

    允许小数：原程序 y 方向的 158.4 就是这么复现的。
    """
    if value in (None, "", [], ()):
        return (float(roi[0]), float(roi[1]))
    try:
        offset = [float(v) for v in value]
    except (TypeError, ValueError):
        raise ValueError("roi_offset 必须是 2 个数 (dx, dy)，收到 %r" % (value,))
    if len(offset) != 2:
        raise ValueError("roi_offset 必须是 2 个数 (dx, dy)，收到 %r" % (value,))
    return (offset[0], offset[1])


def centre_to_source(x1, y1, x2, y2, offset):
    """ROI 内的框（xyxy，像素）→ 整幅图像里的中心点，取整方式与原程序一致。

    原程序是 ``int(中心 + 偏移)`` —— 先加再截断，不是先各自取整。
    """
    cx = int(offset[0] + (x1 + x2) / 2.0)
    cy = int(offset[1] + (y1 + y2) / 2.0)
    return cx, cy


def crop_slice(shape, roi):
    """按 ``roi`` 算出裁剪用的切片；画面装不下就返回 ``None``（不猜、不补边）。

    :param shape: ``image.shape``（至少两维，前两维是 h, w）
    """
    height, width = shape[0], shape[1]
    x, y, w, h = roi
    if width < x + w or height < y + h:
        return None
    return (slice(y, y + h), slice(x, x + w))

## 2. 推理：TensorFlow SSD → RKNN YOLOv5

**为什么必须换**：板上装不了 TensorFlow，`object_detection` / `mscoco_label_map.pbtxt` /
`frozen_inference_graph.pb` 都不在镜像里；而 RK3588 有 NPU，厂商随镜像提供了
`librknnrt.so` 与官方 `rknn_model_zoo` 的默认 YOLOv5 模型。

**怎么换**：

* 推理运行时 = `librknnrt.so` 的 ctypes 兼容层（`python/rknn/api.py`）。
  板上是 musl，PyPI 的 `rknn-toolkit-lite2` 只有 glibc 轮子，pip 装不上；
  C ABI 稳定，用 ctypes 直连是唯一可行路径。
* YOLOv5 解码 + NMS = `nodes/rknn_yolo.py`（v5 三尺度头、anchor、先过滤后解码、按类 NMS）。
* 模型 = 官方默认 `yolov5s-640-640.rknn`（COCO 80 类），
  路径 `/data/local/release/rknn-demo/model/yolov5s-640-640.rknn`，
  可用 `~model_path` 覆盖（`config/yolo/detect_rknn.yaml` 或 `model_path:=...`）。

**对外接口一字未改**，所以 `grasp.py` 与 launch 都不用跟着动：

| 方向 | 话题 | 类型 |
|---|---|---|
| 订阅 | `image` | `sensor_msgs/Image`（launch 里 remap 到相机话题） |
| 发布 | `location` | `semi_auto_match_24/LocationCentre` |
| 发布 | `debug_image` | `sensor_msgs/Image` |

## 3. 两处**必须**说清楚的语义契约

这两处错了都不会报错，只会"抓不到"或"抓偏"，所以单独写出来，并各有一条回归测试。

### 3.1 类别 id：西兰花必须还是 56

`grasp.py` 里判断西兰花是 `loc.obj_id == 56`。这个 56 是
`tensorflow_object_detector` 那份 `mscoco_label_map.pbtxt` 的 **90 类编号**
（中间有 12 个空洞）。而 YOLOv5 输出的是 **COCO-80 下标**，西兰花是 **50**。

直接发模型下标 → `grasp.py` 的西兰花分支永远不触发，且没有任何报错。

处理：`config/yolo/coco_labels.txt` 是三列格式 `模型下标 外部id 名字`，
`detect_rknn.py` 默认（`publish_tf_ids: true`）按它把 50 换算成 56 再发。
`test/test_rknn_yolo_postprocess.py::test_grasp_id_contract` 钉住这条契约。

> 换模型/换类别表时注意：映射表必须是 80 行、下标连续。

### 3.2 坐标：ROI 内坐标 → 整幅图像坐标

原程序只把 640×480 里的 `image[160:480, 160:480]`（320×320）送进模型，
再把框中心换算回**整幅图像坐标**发出去 —— 因为机械臂标定文件 `thefile.txt`
是整幅图像坐标系下的线性映射。

原版写法与本次实现的差别：

| | 原 `detect.py` | 本次实现（默认） |
|---|---|---|
| x 偏移 | `0.25 * 640 = 160` | `roi` 左上角 = 160 |
| y 偏移 | `0.33 * 480 = 158.4` | `roi` 左上角 = 160 |

**x 方向逐像素一致；y 方向默认比原版大 1.6 px。** 原版的 0.33×480 显然是想表达
裁剪原点 160，只是用了近似系数。要**逐像素复现原版**（例如沿用旧的标定文件），
把参数设成：

```yaml
roi: [160, 160, 320, 320]
roi_offset: [160, 158.4]     # 留空 [] 则用 roi 左上角 (160, 160)
```

`test_roi_mapping_matches_original` 同时钉住了"x 一致"、"默认 y 差 1.6 px"、
"显式 158.4 时 y 也一致"三件事。

### 4.1 rviz 的 Fixed Frame：别抄 SLAM 布局

**症状**：VNC 里 rviz 起来了、OpenGL 也正常（`OpenGL device: Mali-G610 (Panfrost)`），
但 3D 画面空白 / 模型发白，什么都画不出来。

**根因**：rviz 的 **Fixed Frame** 必须是一个**本 launch 真的会发布**的 TF 帧，
否则它在状态栏报 `Fixed Frame [xxx] does not exist`，然后整块 3D 视图什么都不画。

本 launch 实际存在的帧（板端 `rostopic info /tf` + `rostopic echo /tf` 实测）：

| 帧 | 谁发 | 本 launch 有没有 |
|---|---|---|
| `odom` | `spark_base_server`（发 `odom → base_footprint`） | **有** |
| `base_link` | `robot_state_publisher` | 有 |
| `camera_link` / `camera_*` | `realsense2_camera`（静态） | 有 |
| `laser` | `robot_state_publisher`（URDF 里的固定关节） | 有 |
| **`map`** | SLAM / `map_server` | **没有** —— 本 launch 不起 SLAM |

一开始我直接抄了 `spark_gmapping_d435.rviz`（它是**建图**布局，Fixed Frame = `map`），
所以上板就白屏。`auto_match.rviz`（2026 版）也是 `map` 中心，但**它的 launch 会起
SLAM 或 map_server**，所以那份没问题、抄过来就有问题。

**现在的做法**：`rviz/semi_auto.rviz` 以官方的
`spark_description/rviz/urdf.rviz` 为底 —— 那份是"看机器人"布局，Fixed Frame 就是
`odom`，且恰好配对 `joint_state_publisher` + `robot_state_publisher` 这一对节点；
再把 PointCloud2 面板换成雷达 + 两路图像（LaserScan/Image 的字段块直接从
`auto_match.rviz` 整块搬，同板同 rviz 版本，schema 一定对）：

```
Grid / RobotModel / TF(关) / LaserScan(/scan)
/ Image(/camera/color/image_raw) / Image(/debug_image)      Fixed Frame: odom
```

**视角与窗口也一起调了**（官方 `urdf.rviz` 存的是 Distance 6.64 m、窗口 1859×1056）：

| 项 | 原值 | 改成 | 为什么 |
|---|---|---|---|
| `Target Frame` | `<Fixed Frame>`（= odom 原点） | `base_link` | 相机跟着机器人走；盯 odom 原点的话，车一开走就不在画面里了 |
| `Distance` | 6.64 m | 2.5 m | 底盘只有 34 cm，6.6 m 外只剩一个小点 |
| 窗口尺寸 | 1859×1056 | 1100×650 | 板上 Xvfb 就是 1100×650；尺寸不匹配时 Qt 按旧布局摆面板，机器人会被挤到画面角落 |

生成脚本 `test/gen_rviz.py` 一次搞定，要改帧或视角直接改脚本里的 `VIEW_FIXES` 重跑，
或者在 rviz 里 `Global Options → Fixed Frame` 改完 `File → Save Config As` 覆盖。

## 5. 刻意**没有**改的行为

原程序里的这些写法看着奇怪，但它们直接影响标定基准或比赛策略，改动属于另一件事，
本次一律保留（只在代码注释里标出来）：

* `image_cb` 与 `image_cb_red` 订阅**同一个话题**、每帧各跑一遍 HSV（浪费，但改了会动时序）。
* 距离度量 `sqrt((x_mid-320)**2 + (480-y_mid)**2)` 按 640×480 写死。
* `grasp(self, y, x, layer)` 的形参顺序与实参顺序不一致（调用处传的是 `(x, y)`）——
  保持原样，改了会改变机械臂落点。
* `broccoli_found_count >= 10` / `water_found_count >= 30` 等稳定性阈值。

### 5.1 改过的一处真 bug：水源与红块**共用计数器**

原 `grasp.py` 里 `image_cb_red` 和 `image_cb` **共用** `self.water_found_count` /
`self.found_water`。两个回调订阅同一个话题、每帧各跑一次，于是互相清零：

```
image_cb     （蓝）：中心稳定 -> water_found_count += 1
image_cb_red （红）：没找到轮廓 -> water_found_count = 0     ← 把蓝攒的清了
```

**现象**：按 G 抓水源，只看到 `[WARN] finding water` 一直刷、机械臂不动 ——
`found_water` 永远到不了 True。

**实测（2026-10-14，板端探针）**：60 帧里蓝色轮廓 **0/60**（画面是灰地毯 + 一块
白方块，S≈9），红色轮廓 59/60 但中心每帧跳几十像素（不稳定）→ 共享计数恒为 0。

**处置**：红色用自己的 `red_found_count` / `found_red`，`grasp_cp` 里抓红块那条
分支（`'6'` / `'12'`）等 `found_red`。

> 注意：这次同时暴露了"**有红没蓝**时计数必为 0"。即使把蓝色水源放进视野，
> 只要红分支还在清零，照样抓不到 —— 所以这是必修项，不是优化。

### 5.2 水源认不出来时的排查顺序

`grasp.py` 认水源靠的是**固定 HSV 区间**：`H[95,130] S[90,255] V[80,255]`。

```sh
# 板上量一量相机到底看到什么（只订阅，不改任何东西）
run python3 /data/local/release/usr/share/semi_auto_match_24/test/probe_grasp_colors.py 60
```

它照抄 grasp.py 的两条流水线，打印：蓝/红各有几帧出轮廓、中心稳不稳、
按现在的共享计数 vs 各用各的计数分别会不会 `found_water`，并把当前帧与两张掩膜
落到 `/data/local/tmp/probe_*.png`。

| 观察 | 结论 |
|---|---|
| 蓝轮廓 0/N，且全图 S 均值很低 | **画面里没有蓝色目标** —— 把水源放进相机视野（相机装在机械臂上，朝下看夹爪附近） |
| 蓝轮廓有，但中心每帧跳 >4px | 目标在动 / 光斑不稳 —— 拿稳，或调阈值 |
| 有蓝有轮廓但 S 明显 <90 | 目标不是饱和蓝（白 / 浅蓝 / 过曝）—— 换目标，或放宽 `LowerBlue` 的 S 下限 |

### 5.3 抓取高度：一个高度给三种目标，并且能现场调

**单位是毫米** —— `swiftpro` 的 `position.z` 直接拼成 Gcode `G0 X.. Y.. Z..`，
`swiftpro_write_node.cpp` 的注释写明 `x,y,z(mm)`，Z=0 是机械臂底座平面。

原 `grasp()` 是按**按键大小写**选高度，和抓什么东西无关：

```python
if layer in ['10', '11', '12']:   # 大写 G/T/E
    z_up, z_down = 90, 30
else:                             # 小写 g/t/e
    z_up, z_down = 20, -55
```

**两支差了 85mm** —— 这才是"三个目标高度看着不一样"的真正原因。

板端日志实测（`swiftpro_write_node` 每次都会 `ROS_INFO` 打 Gcode，
所以实际发出去的 Z 是确定的，不用猜）：

| 你按的键 | 走的支 | 实际发的 Gcode | 结果 |
|---|---|---|---|
| 小写 `g` / `t`（水源、蔬菜） | `else` | `Z20` → `Z-55` | 太低，机械臂顶到自己 |
| 大写 `E`（红色方块） | `if` | `Z90` → `Z30` | 太高，吸盘够不到 |

**现在统一成一套，大小写只决定发哪条指令、不影响高度**：

| 项 | 值 | 说明 |
|---|---|---|
| `z_up` | 20 | 悬停高度 |
| `z_down` | **-45** | 原小写分支是 -55，实测太深，抬高 1cm |

**现场调（不用重新编译）**：

```sh
# 板上直接改，然后重启 grasp（或重跑菜单第 11 项）
vi /data/local/release/usr/share/semi_auto_match_24/config/grasp_heights.yaml
#   顶到自己/压住物体 -> z_down 调大；吸盘够不到 -> z_down 调小
```

单跑时也能临时覆盖：`run rosrun semi_auto_match_24 grasp.py _z_down:=-40`。
节点启动时会打印实际用的高度（`抓取高度(mm): z_up=.. z_down=..`），
抓取时还会打一条 `抓取目标=... layer=... -> z_up=.. z_down=..`，便于核对。

> **怎么自查实际发了什么 Z**：`swiftpro_write_node` 把每条 Gcode 都打进
> `<log>/<run>/swiftpro_write_node-9-stdout.log`，直接
> `grep -o 'G0 X.*Z[-0-9.]*'` 就能看到真实轨迹 —— 比看机械臂动作可靠得多。

### 5.4 "够不到"的位姿：机械臂不动，但泵照开、臂照抬

**症状**：抓某个目标时机械臂**不动**，却看到吸泵先开、然后直接抬到高位；
`swiftpro_rviz_node` 刷 `[ERROR] Inverse kinematic is wrong`。

**这不是两个问题，是同一个**：标定把像素映射成机械臂坐标，但机械臂
**够不到所有位置**。够不到时固件不接受那个位姿（机械臂原地不动），
而 `grasp()` 后面照样 `pub2.publish(1)` 开泵、照样抬到 `(230, 0, 160)` ——
现场看起来就是"没抓就跑了"。

那句 `Inverse kinematic is wrong` 来自 `swiftpro_rviz_node`，
**那个节点只把位姿画到 rviz、不参与实际运动**：

```cpp
if ( swiftpro_ik(position, angle) )   // 算不出来就只报一句
    all_joints_state(angle);
else
    ROS_ERROR("Inverse kinematic is wrong");
```

所以它是**同一个数学结论的旁证**，不是原因 —— 但它恰好是最早能看到的那条线索。

**可达区域（实测，640×480，标定用板上 thefile.txt，z=20 与 -45 都要可达）**：

```
行   0 ~ 110   整行都够不到        pos.x > ~370mm，超出臂展
行 112 ~ 300   只有中间一条横带可达，越往下越宽
行 320 ~ 400   整行都可达   ← 物体放这里最稳
行 410 ~ 480   底部中央出现缺口    列约 280~440，离底座太近折不过来
```

```sh
# 打印这张图
python3 test/reachable_area.py
```

### 5.5 红色方块检不到：HSV 的红色是**两端**的

**症状**：物块明明在可达区（`reachable_area.py --overlay` 里是绿的），就是抓不到；
抓水源/蔬菜却正常。

**根因**：红色在 OpenCV 色环上是**两端** —— H≈0 **和** H≈179。
原程序写的是：

```python
LowerRed = np.array([146, 80, 46])     # 只覆盖**高端**
UpperRed = np.array([180, 255, 255])
```

而板上这块方块实测 **H≈0~5**（直方图峰值在 H=1~3）。**一个像素都匹配不到**，
于是检测器只能去锁画面里的噪声，算出来的抓取坐标全是垃圾。

**还有第二个坑**（在真实帧上跑完整流水线才暴露）：

| 方案 | 结果 |
|---|---|
| 原样 `H[146,180]` + H 通道阈值 20 | 无轮廓 |
| 只把区间改成 `H[0,15]`，阈值仍是 20 | inRange 命中 **19238** 像素，**仍然无轮廓** |
| `H[0,15]` + 阈值降到 1 | 轮廓 22 个，最大面积 13788，中心 (328, 299) |
| **`H[0,15]` + 直接对掩膜二值化** | **轮廓 1 个**，面积 14998，中心 (329, 297) ← 采用 |
| `H[0,15] ∪ [160,180]` | 最大轮廓包围盒横跨半个画面 ← **把左侧鞋子/杂物也当成红色** |

原因：红色分支那步 `cv2.threshold(blurred, 20, ...)` 是对**取出来的 H 通道**取阈值，
`H≈0~5` 会被 20 直接抹平 —— 区间改对了也没用。改成直接对掩膜二值化，
顺带避开"对环绕的 H 做 blur"在红端产生的伪影。

**处置**：区间改成 `H[0, 15]`（**不要**把 `[160,180]` 并进来），
阈值走掩膜；两个区间可由 `config/grasp_colors.yaml` 覆盖。
改完中心落在 (329, 297) → 机械臂 (186.2, 21.2)、半径 187 → **可达**。

> 换方块颜色/材质后重新量：`test/probe_grasp_colors.py` 会打印高饱和像素的
> 色相直方图；`H` 落在哪就照那个改 `grasp_colors.yaml`。

唯一一处**必须改**的默认值：手眼标定文件的查找顺序。原程序写死 `$HOME/thefile.txt`，
而本板 `HOME=/`，实测标定文件在 `/data/local/release/usr/thefile.txt`。
现在按 `~calib_file` > `$SEMI_AUTO_MATCH_CALIB` > `$HOME/thefile.txt` >
`/data/local/release/usr/thefile.txt` 找第一个存在的，并把选中的路径打进日志
（标定文件选错不会报错，只会让机械臂抓偏）。

        self.index = int(attr.index)
        self.name = attr.name.decode("utf-8", "replace")
        self.shape = [int(attr.dims[d]) for d in range(attr.n_dims)]
        self.fmt = int(attr.fmt)
        self.type = int(attr.type)
        self.n_elems = int(attr.n_elems)
        self.size = int(attr.size)
        self.zp = int(attr.zp)
        self.scale = float(attr.scale)

    @property
    def is_nhwc(self):
        return self.fmt == RKNN_TENSOR_NHWC

    def __repr__(self):
        return "TensorInfo(index=%d, name=%r, shape=%s, fmt=%d, type=%d)" % (
            self.index, self.name, self.shape, self.fmt, self.type,
        )

_lib_cache = {}


def _preload_cxx_runtime(lib_path):
    """先 dlopen ``libc++_shared.so`` 再 dlopen ``librknnrt.so``。

    ``librknnrt.so`` 依赖 ``libc++_shared.so``，而 OpenHarmony release 里
    没有把它装到系统库路径；板上没有 ``ldd``，与其猜依赖不如把厂商放在
    运行时旁边的那个副本直接预加载（RTLD_GLOBAL 让后续 dlopen 能解析到）。
    """
    candidates = []
    env = os.environ.get("RKNN_CXX_LIB")
    if env:
        candidates.append(env)
    lib_dir = os.path.dirname(os.path.abspath(lib_path))
    candidates.append(os.path.join(lib_dir, "libc++_shared.so"))
    candidates.append("/data/local/release/rknn-demo/lib/libc++_shared.so")
    candidates.append("/usr/lib/libc++_shared.so")
    for path in candidates:
        if path and os.path.exists(path):
            try:
                ctypes.CDLL(path, mode=ctypes.RTLD_GLOBAL)
            except OSError:
                # 已经加载过、或与本次运行时不同代：交给下一步 dlopen 报错，
                # 不要在这里吞掉真正的原因。
                continue
            return path
    return None


def _load_library(lib_path):
    """按路径缓存地 dlopen librknnrt.so 并声明函数签名。"""
    if lib_path in _lib_cache:
        return _lib_cache[lib_path]

    if not os.path.exists(lib_path):
        raise OSError(
            "找不到 RKNN 运行时 %r；用 lib_path= 或环境变量 RKNN_RT_LIB 指定 "
            "librknnrt.so 的位置" % lib_path
        )

    _preload_cxx_runtime(lib_path)
    try:
        lib = ctypes.CDLL(lib_path, mode=ctypes.RTLD_GLOBAL)
    except OSError as exc:
        raise OSError(
            "dlopen(%s) 失败: %s —— 通常是缺 libc++_shared.so（设 RKNN_CXX_LIB 指向它）"
            % (lib_path, exc)
        )

    # argtypes 必须显式声明：rknn_context 是 uint64，靠默认的 int 会被截断。
    lib.rknn_init.argtypes = [
        ctypes.POINTER(ctypes.c_uint64), ctypes.c_void_p,
        ctypes.c_uint32, ctypes.c_uint32, ctypes.c_void_p,
    ]
    lib.rknn_init.restype = ctypes.c_int
    lib.rknn_query.argtypes = [ctypes.c_uint64, ctypes.c_int, ctypes.c_void_p, ctypes.c_uint32]
    lib.rknn_query.restype = ctypes.c_int
    lib.rknn_inputs_set.argtypes = [ctypes.c_uint64, ctypes.c_uint32, ctypes.POINTER(RknnInput)]
    lib.rknn_inputs_set.restype = ctypes.c_int
    lib.rknn_run.argtypes = [ctypes.c_uint64, ctypes.c_void_p]
    lib.rknn_run.restype = ctypes.c_int
    lib.rknn_outputs_get.argtypes = [
        ctypes.c_uint64, ctypes.c_uint32, ctypes.POINTER(RknnOutput), ctypes.c_void_p,
    ]
    lib.rknn_outputs_get.restype = ctypes.c_int
    lib.rknn_outputs_release.argtypes = [
        ctypes.c_uint64, ctypes.c_uint32, ctypes.POINTER(RknnOutput),
    ]
    lib.rknn_outputs_release.restype = ctypes.c_int
    lib.rknn_destroy.argtypes = [ctypes.c_uint64]
    lib.rknn_destroy.restype = ctypes.c_int

    # 1.4 之前的运行时没有 set_core_mask；有则声明，没有就跳过。
    if hasattr(lib, "rknn_set_core_mask"):
        lib.rknn_set_core_mask.argtypes = [ctypes.c_uint64, ctypes.c_int]
        lib.rknn_set_core_mask.restype = ctypes.c_int

    _lib_cache[lib_path] = lib
    return lib

_CORE_MASK_NAMES = {
    NPU_CORE_AUTO: "auto",
    NPU_CORE_0: "0",
    NPU_CORE_1: "1",
    NPU_CORE_2: "2",
    NPU_CORE_0_1: "0_1",
    NPU_CORE_0_1_2: "0_1_2",
    NPU_CORE_ALL: "all",
}


def core_mask_from_string(value):
    """把 ``"0_1_2"`` / ``"0,1,2"`` / int 统一成 rknn_core_mask 位掩码。

    ROS 参数只能是字符串或整数，这里同时容忍 ``"NPU_CORE_0_1_2"`` 这种
    从官方示例里抄来的写法。
    """
    if isinstance(value, int):
        return value
    text = str(value).strip().upper()
    if text.startswith("NPU_CORE_"):
        text = text[len("NPU_CORE_"):]
    text = text.replace(",", "_").replace("|", "_").replace("+", "_")
    if text in ("AUTO", ""):
        return NPU_CORE_AUTO
    if text == "ALL":
        return NPU_CORE_ALL
    mask = 0
    for part in text.split("_"):
        if part == "":
            continue
        if part not in ("0", "1", "2"):
            raise ValueError(
                "无法识别的 core_mask %r；可用: auto / 0 / 1 / 2 / 0_1 / 0_1_2 / all" % value
            )
        mask |= {0: NPU_CORE_0, 1: NPU_CORE_1, 2: NPU_CORE_2}[int(part)]
    return mask


class _Runtime(object):
    """两个门面共用的引擎：模型 → float32 输出。"""

    def __init__(self, lib_path=None, verbose=False):
        self.verbose = verbose
        self.lib_path = lib_path or os.environ.get("RKNN_RT_LIB") or DEFAULT_LIB
        self._lib = _load_library(self.lib_path)
        self._ctx = None
        self._model_path = None
        self.n_input = 0
        self.n_output = 0
        self.input_attrs = []
        self.output_attrs = []
        self.sdk_version = None
        self.driver_version = None
        self.core_mask = NPU_CORE_AUTO

    # -- 生命周期 ------------------------------------------------------
    def load_rknn(self, path, file_size=0):
        if not os.path.exists(path):
            raise FileNotFoundError("模型文件不存在: %s" % path)
        self._model_path = path
        return 0

    def init_runtime(self, target=None, device_id=None, perf_debug=False,
                     eval_mem=False, async_mode=False, core_mask=NPU_CORE_AUTO,
                     **kwargs):
        if self._model_path is None:
            raise RuntimeError("必须先 load_rknn(path) 再 init_runtime()")
        # target/device_id/async_mode 在单板本地推理里没有对应参数：留着是为了
        # 让从官方示例抄来的调用不用改。
        if async_mode:
            warnings.warn("本兼容层不支持 async_mode，按同步推理执行")
        del target, device_id, perf_debug, eval_mem, kwargs

        ctx = ctypes.c_uint64(0)
        # size=0 表示 model 传的是"模型文件路径"，不是内存里的模型数据。
        ret = self._lib.rknn_init(
            ctypes.byref(ctx), ctypes.c_char_p(self._model_path.encode()), 0, 0, None
        )
        if ret != 0:
            raise RuntimeError(
                "rknn_init(%s) 失败: %s —— 最常见的原因是模型与运行时不同代"
                % (self._model_path, error_name(ret))
            )
        self._ctx = ctx

        ver = RknnSdkVersion()
        if self._lib.rknn_query(ctx, RKNN_QUERY_SDK_VERSION,
                                ctypes.byref(ver), ctypes.sizeof(ver)) == 0:
            self.sdk_version = ver.api_version.decode("utf-8", "replace")
            self.driver_version = ver.drv_version.decode("utf-8", "replace")

        self.core_mask = core_mask_from_string(core_mask)
        if self.core_mask != NPU_CORE_AUTO:
            if not hasattr(self._lib, "rknn_set_core_mask"):
                warnings.warn("当前 librknnrt.so 没有 rknn_set_core_mask，core_mask 被忽略")
            else:
                ret = self._lib.rknn_set_core_mask(self._ctx, self.core_mask)
                if ret != 0:
                    warnings.warn("rknn_set_core_mask(%d) 失败: %s"
                                  % (self.core_mask, error_name(ret)))

        io_num = RknnInputOutputNum()
        ret = self._lib.rknn_query(ctx, RKNN_QUERY_IN_OUT_NUM,
                                   ctypes.byref(io_num), ctypes.sizeof(io_num))
        if ret != 0:
            raise RuntimeError("rknn_query(IN_OUT_NUM) 失败: %s" % error_name(ret))
        self.n_input, self.n_output = int(io_num.n_input), int(io_num.n_output)

        self.input_attrs = self._query_attrs(RKNN_QUERY_INPUT_ATTR, self.n_input)
        self.output_attrs = self._query_attrs(RKNN_QUERY_OUTPUT_ATTR, self.n_output)

        if self.verbose:
            print("rknn: api %s / driver %s / core_mask %s"
                  % (self.sdk_version, self.driver_version,
                     _CORE_MASK_NAMES.get(self.core_mask, self.core_mask)))
            for info in self.input_attrs:
                print("  in  %s" % info)
            for info in self.output_attrs:
                print("  out %s" % info)
        return 0

    def _query_attrs(self, cmd, count):
        attrs = []
        for i in range(count):
            attr = RknnTensorAttr()
            attr.index = i
            ret = self._lib.rknn_query(self._ctx, cmd, ctypes.byref(attr), ctypes.sizeof(attr))
            if ret != 0:
                raise RuntimeError("rknn_query(%d) 失败: %s" % (cmd, error_name(ret)))
            attrs.append(TensorInfo(attr))
        return attrs

    # -- 推理 ----------------------------------------------------------
    def inference(self, inputs, data_type=None, data_format="nhwc", inputs_pass_through=None):
        if self._ctx is None:
            raise RuntimeError("必须先 init_runtime()")
        if not isinstance(inputs, (list, tuple)):
            inputs = [inputs]
        if len(inputs) != max(self.n_input, 1):
            raise ValueError("模型需要 %d 个输入，收到 %d 个" % (self.n_input, len(inputs)))
        if data_format not in _DATA_FORMAT_TO_TENSOR:
            raise ValueError("data_format 只支持 nchw / nhwc，收到 %r" % (data_format,))
        fmt = _DATA_FORMAT_TO_TENSOR[data_format]

        if inputs_pass_through is None:
            pass_through = [0] * len(inputs)
        elif isinstance(inputs_pass_through, (list, tuple)):
            pass_through = [int(v) for v in inputs_pass_through]
        else:
            pass_through = [int(inputs_pass_through)] * len(inputs)

        rknn_inputs = (RknnInput * len(inputs))()
        # 这些数组必须活到 rknn_inputs_set 返回为止，否则 buf 指向已释放内存。
        held = []
        for i, arr in enumerate(inputs):
            buf = np.ascontiguousarray(arr)
            held.append(buf)
            if data_type is None:
                dtype_name = _NUMPY_TO_DATA_TYPE.get(buf.dtype.name)
                if dtype_name is None:
                    raise ValueError("不支持的输入 dtype %r；请显式给 data_type=" % (buf.dtype,))
            else:
                dtype_name = str(data_type).lower()
            if dtype_name not in _DATA_TYPE_TO_TENSOR:
                raise ValueError("不支持的 data_type %r" % (data_type,))
            rknn_inputs[i].index = i
            rknn_inputs[i].buf = buf.ctypes.data_as(ctypes.c_void_p)
            rknn_inputs[i].size = buf.nbytes
            # pass_through=0：前处理（mean/std、rgb2bgr）交给驱动按模型元数据做。
            rknn_inputs[i].pass_through = pass_through[i]
            rknn_inputs[i].type = _DATA_TYPE_TO_TENSOR[dtype_name]
            rknn_inputs[i].fmt = fmt

        ret = self._lib.rknn_inputs_set(self._ctx, len(inputs), rknn_inputs)
        if ret != 0:
            raise RuntimeError("rknn_inputs_set 失败: %s" % error_name(ret))

        ret = self._lib.rknn_run(self._ctx, None)
        if ret != 0:
            raise RuntimeError("rknn_run 失败: %s" % error_name(ret))

        outputs = (RknnOutput * self.n_output)()
        for i in range(self.n_output):
            outputs[i].want_float = 1     # 驱动负责反量化成 float32
            outputs[i].is_prealloc = 0    # 由驱动分配，用完必须 release
            outputs[i].index = i

        ret = self._lib.rknn_outputs_get(self._ctx, self.n_output, outputs, None)
        if ret != 0:
            raise RuntimeError("rknn_outputs_get 失败: %s" % error_name(ret))

        result = []
        try:
            for i in range(self.n_output):
                info = self.output_attrs[i]
                if not outputs[i].buf:
                    raise RuntimeError("输出 %d 的缓冲区为空" % i)
                flat = np.ctypeslib.as_array(
                    ctypes.cast(outputs[i].buf, ctypes.POINTER(ctypes.c_float)),
                    shape=(info.n_elems,),
                ).copy()    # 必须 copy：下面 release 之后这块内存就无效了
                arr = flat.reshape(info.shape)
                if info.is_nhwc:
                    # 统一成 NCHW：YOLO 系列的后处理都按 channel-first 写。
                    arr = np.transpose(arr, (0, 3, 1, 2))
                result.append(arr)
        finally:
            self._lib.rknn_outputs_release(self._ctx, self.n_output, outputs)
        return result

    def release(self):
        if self._ctx is not None:
            self._lib.rknn_destroy(self._ctx)
            self._ctx = None
        return 0

    """SDK 里预装的系统 ROS 包（rviz / map_server / tf ... 不在工作区里）。"""
    found = set()
    if not os.path.isdir(SDK_SHARE):
        return found
    for name in os.listdir(SDK_SHARE):
        if os.path.exists(os.path.join(SDK_SHARE, name, "package.xml")):

def skiplist():
    try:
        with open(BUILD_SH, "r") as handle:
            text = handle.read()
    except IOError:
        return set()
    block = re.search(r"#skiplist(.*?)#cmake-args", text, re.S)
    if not block:

    return set(re.findall(r'config_args\+=\("([^"]+)"\)', block.group(1)))


def node_exists(pkg, typ):
    for pattern in ("*/" + pkg + "/**/" + typ, "*/**/" + pkg + "/**/" + typ):
        if glob.glob(os.path.join(SRC_DIR, pattern), recursive=True):
            return True
    # SDK 预装包的可执行文件在 <sdk>/usr/lib/<pkg>/ 下（例如 rviz）
    if glob.glob(os.path.join(SDK_ROOT, "usr", "lib", pkg, typ)):
        return True
    return bool(glob.glob(os.path.join(SDK_SHARE, pkg, "**", typ), recursive=True))


def check_file(path, packages, skipped):
    root = ET.parse(path).getroot()
    name = os.path.basename(path)

    # 只有 <launch> 的直接子 <arg> 才是"声明"；<include> 里的 <arg> 是传参
    declared = {a.get("name") for a in root.findall("arg") if a.get("name")}
    passed = {a.get("name") for inc in root.iter("include")
              for a in inc.findall("arg") if a.get("name")}
    with open(path, "r") as handle:
        text = handle.read()

    used = set(ARG_REF.findall(text)) | set(ARG_EVAL.findall(text))
    undeclared = sorted(used - declared - passed)
    if undeclared:
        failures.append("%s: 用了没声明的 arg: %s" % (name, undeclared))
    else:
        print("  OK   %s: %d 个 arg 声明、%d 个 include 传参，引用全部有出处"
              % (name, len(declared), len(passed)))

    unused = sorted(declared - used)
    if unused:
        warnings.append("%s: 声明了但没用到的 arg（确认是刻意保留）: %s" % (name, unused))

    for pkg in sorted(set(FIND_REF.findall(text))):
        if pkg not in packages:
            failures.append("%s: $(find %s) 找不到这个包" % (name, pkg))
        elif pkg in skipped and pkg not in PREBUILT_ON_BOARD:
            failures.append("%s: $(find %s) 在 skiplist 里，板端不会有它" % (name, pkg))

    for node in root.iter("node"):
        pkg, typ = node.get("pkg"), node.get("type")
        if not pkg:
            continue
        if pkg not in packages:
            failures.append("%s: <node pkg=%r> 找不到这个包" % (name, pkg))
            continue
        if pkg in skipped and pkg not in PREBUILT_ON_BOARD:
            failures.append("%s: <node pkg=%r> 在 skiplist 里，板端不会有它" % (name, pkg))
        if typ and not node_exists(pkg, typ):
            warnings.append("%s: <node pkg=%r type=%r> 没找到同名可执行文件"
                            % (name, pkg, typ))


def main(argv):
    targets = argv or sorted(glob.glob(os.path.join(PACKAGE_DIR, "launch", "*.launch")))
    if not targets:
        print("没有要检查的 launch 文件")
        return 1

    packages = workspace_packages() | sdk_packages()
    skipped = skiplist()
    print("工作区包 %d 个；SDK 预装包 %d 个；skiplist %d 个；待查文件 %d 个"
          % (len(workspace_packages()), len(sdk_packages()), len(skipped), len(targets)))
    print("")
    for path in targets:
        check_file(path, packages, skipped)

    print("")
    for w in warnings:
        print("WARN %s" % w)
    if failures:
        for f in failures:
            print("FAIL %s" % f)
        return 1
    print("launch 引用检查通过")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

"""生成 semi_auto_match_24 自己的 rviz 布局（``rviz/semi_auto.rviz``）。

    python3 test/gen_rviz.py [fixed_frame]      # 默认 odom

为什么不能直接抄现成的
----------------------
* ``spark_gmapping_d435.rviz`` 是 **SLAM 布局** —— Fixed Frame = ``map``，
  还带 /map 上的 Map 显示。而 ``semi_auto.launch`` **不启动 SLAM / map_server**，
  ``map`` 帧根本不存在，rviz 会 ``Fixed Frame [map] does not exist`` 然后
  3D 视图什么都不画（上板实测就是"画面一片空白/发白"）。
* ``auto_match.rviz``（2026 版）同样是 ``map`` 中心 —— 但**它的 launch 会起
  SLAM 或 map_server**，所以那份没问题，抄过来就有问题。

本脚本的做法
------------
拿 ``spark_description/rviz/urdf.rviz`` 当底 —— 那是**官方的"看机器人"布局**，
Fixed Frame 就是本 launch 真实存在的 ``odom``，而且它恰好配对
``joint_state_publisher`` + ``robot_state_publisher`` 这一对节点；
再把 PointCloud2 面板换成雷达 + 两路图像：

    Grid / RobotModel / TF(关) / LaserScan(/scan)
    / Image(/camera/color/image_raw) / Image(/debug_image)      Fixed Frame: odom

LaserScan 与 Image 的字段块直接从 ``auto_match.rviz`` 里**整块搬**（同板同
rviz 版本，字段 schema 一定对），不手写。

本 launch 实测存在的帧：``odom``（spark_base_server）、``base_link`` /

import io
import os
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
PACKAGE_DIR = os.path.dirname(HERE)
SRC_DIR = os.path.dirname(os.path.dirname(PACKAGE_DIR))
URDF_RVIZ = os.path.join(SRC_DIR, "spark", "spark_description", "rviz", "urdf.rviz")
AUTO_RVIZ = os.path.join(SRC_DIR, "spark_app", "auto_match_25", "auto_match",
                         "rviz", "auto_match.rviz")

def read_lines(path):
    with io.open(path, "r", encoding="utf-8") as handle:
        return handle.read().split("\n")


def _is_block_end(line):
    """块内字段缩进 6 空格（嵌套键 8 空格）；遇到空行或缩进不足就是块结束。

    注意不能写成 ``line.startswith("  ")`` —— 6 空格的字段也"以两个空格开头"，
    那样每块都只剩第一行（踩过）。
    """
    if not line.strip():
        return False
    return not line.startswith("      ")


def extract_block(lines, class_marker):
    """按缩进切出一个顶层显示项。

    两种写法都要认：
    * ``    - Alpha: 1`` … ``      Class: rviz/X``   （块从上一行开始）
    * ``    - Class: rviz/X``                        （块就从这一行开始）
    """
    start = None
    inline = False
    for i, line in enumerate(lines):
        stripped = line.strip()
        if stripped == "Class: " + class_marker:
            start, inline = i, False
            break
        if stripped == "- Class: " + class_marker:
            start, inline = i, True
            break
    if start is None:
        raise SystemExit("找不到 %s 的显示项" % class_marker)

    if inline:
        begin = start
    else:
        begin = start
        while begin > 0 and not lines[begin].startswith("    - "):
            begin -= 1
        if not lines[begin].startswith("    - "):
            raise SystemExit("%s 的显示项找不到起始行" % class_marker)

    end = start + 1
    while end < len(lines) and not _is_block_end(lines[end]):
        end += 1
    return lines[begin:end]


def set_key(block, key, value):
    """把块里某个键的值改掉（保留缩进）。"""
    out = []
    for line in block:
        if line.strip().startswith(key + ":"):
            indent = line[:len(line) - len(line.lstrip())]
            out.append("%s%s: %s" % (indent, key, value))
        else:
            out.append(line)
    return out


# 视角与窗口：官方 urdf.rviz 存的是 Distance 6.64m / 窗口 1859x1056，
# 而板上的 Xvfb 是 1100x650 —— 直接照搬会把机器人挤到画面角落，

# 窗口尺寸对齐板上那块虚拟屏。
VIEW_FIXES = [
    ("      Target Frame: <Fixed Frame>", "      Target Frame: base_link"),
    ("      Distance: 6.64356", "      Distance: 2.50000"),
    ("      Pitch: 0.399797", "      Pitch: 0.350000"),
    ("  Width: 1859", "  Width: 1100"),
    ("  Height: 1056", "  Height: 650"),
    ("  X: 61", "  X: 0"),
    ("  Y: 24", "  Y: 0"),
]


def apply_view_fixes(lines):
    text = "\n".join(lines)
    applied = []
    for old, new in VIEW_FIXES:
        if old in text:
            text = text.replace(old, new, 1)
            applied.append(new.strip())
    return text.split("\n"), applied


def find_block_range(lines, class_marker):
    begin = None
    for i, line in enumerate(lines):
        if line.strip() == "Class: " + class_marker:
            begin = i
            break
    if begin is None:
        raise SystemExit("找不到 %s 的显示项" % class_marker)
    while begin > 0 and not lines[begin].startswith("    - "):
        begin -= 1
    end = begin + 1
    while end < len(lines) and not _is_block_end(lines[end]):
        end += 1
    return begin, end


def main():
    urdf = read_lines(URDF_RVIZ)
    auto = read_lines(AUTO_RVIZ)

    laser = set_key(set_key(extract_block(auto, "rviz/LaserScan"),

# -*- coding: utf-8 -*-
"""量一量 grasp.py 的水源（蓝）/红块检测到底有没有命中，以及计数器会不会被互相清零。

完全照抄 grasp.py 的两条 HSV 流水线与"最近轮廓"选法，只是把内部状态打印出来。
**只订阅话题，不改板端任何东西**，可以边跑比赛程序边量。

用法（板上）::

    run python3 test/probe_grasp_colors.py [帧数]

判据：
* 蓝轮廓 0/N 且全图 S 均值很低  -> 画面里没有蓝色目标，把水源放进相机视野
* 蓝轮廓有但中心每帧跳 >4px      -> 目标在动 / 光斑不稳
* 有蓝有轮廓但 S <90             -> 目标不是饱和蓝（白 / 浅蓝 / 过曝），要放宽阈值
"""

import math
import sys

import cv2
import numpy as np
import rospy
from cv_bridge import CvBridge
from sensor_msgs.msg import Image

TOPIC = "/camera/color/image_raw"
WINDOW = 60


def pipeline(cv_image, lower, upper, thresh_val):
    """照抄 grasp.py：HSV -> inRange -> 取 H 通道 -> blur -> threshold
    -> CLOSE(25x25) -> erode x4 -> dilate x4 -> findContours。"""
    hsv = cv2.cvtColor(cv_image, cv2.COLOR_BGR2HSV)
    mask = cv2.inRange(hsv, np.array(lower), np.array(upper))
    chan = cv2.bitwise_and(hsv, hsv, mask=mask)[:, :, 0]
    blurred = cv2.blur(chan, (9, 9))
    _, thresh = cv2.threshold(blurred, thresh_val, 255, cv2.THRESH_BINARY)
    kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (25, 25))
    closed = cv2.morphologyEx(thresh, cv2.MORPH_CLOSE, kernel)
    closed = cv2.erode(closed, None, iterations=4)
    closed = cv2.dilate(closed, None, iterations=4)
    contours = cv2.findContours(closed, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)[0]
    return contours, mask


def nearest_centre(contours):
    """照抄 grasp.py：取离 (320, 480) 最近的轮廓的中心。返回 (x, y) 或 None。"""
    if not contours:
        return None
    dis_min = 640
    best = None
    for c in contours:
        rect = cv2.minAreaRect(c)
        box = cv2.boxPoints(rect)
        box = np.intp(box)
        x_mid = (box[0][0] + box[2][0] + box[1][0] + box[3][0]) / 4
        y_mid = (box[0][1] + box[2][1] + box[1][1] + box[3][1]) / 4
        d = math.sqrt((x_mid - 320) ** 2 + (480 - y_mid) ** 2)
        if d < dis_min:
            dis_min = d
            best = (x_mid, y_mid)
    return best


class Probe(object):
    def __init__(self):
        self.frames = 0
        self.bridge = CvBridge()
        # 照抄 grasp.py 的两个计数器语义：**共用一个 water_found_count**
        self.shared_count = 0
        self.blue_only_count = 0
        self.red_only_count = 0
        self.found_water_shared = False
        self.found_water_blue_only = False
        self.found_red_only = False
        self.blue_prev = (0.0, 0.0)
        self.red_prev = (0.0, 0.0)
        self.saved = False
        self.blue_hits = 0
        self.red_hits = 0
        self.blue_stable = 0
        self.red_stable = 0
        self.blue_trace = []
        self.red_trace = []
        rospy.Subscriber(TOPIC, Image, self.cb, queue_size=1, buff_size=2 ** 24)

    def cb(self, msg):
        try:
            img = self.bridge.imgmsg_to_cv2(msg, "bgr8")
        except Exception as exc:
            print("图像转换失败: %s" % exc)
            return
        self.frames += 1

        blue, blue_mask = pipeline(img, [95, 90, 80], [130, 255, 255], 90)
        red, red_mask = pipeline(img, [146, 80, 46], [180, 255, 255], 20)
        bc = nearest_centre(blue)
        rc = nearest_centre(red)

        # ---- image_cb（蓝）----
        if blue:
            self.blue_hits += 1
            if bc:
                self.blue_trace.append((round(bc[0], 1), round(bc[1], 1)))
                if abs(bc[0] - self.blue_prev[0]) <= 4 and abs(bc[1] - self.blue_prev[1]) <= 4:
                    self.blue_stable += 1
                    self.blue_only_count += 1
                    self.shared_count += 1
                else:
                    self.blue_only_count = 0
                    self.shared_count = 0
                self.blue_prev = bc
        else:
            self.blue_only_count = 0
            self.shared_count = 0

        # ---- image_cb_red（红）—— 它也在动同一个 shared_count ----
        if red:
            self.red_hits += 1
            if rc:
                self.red_trace.append((round(rc[0], 1), round(rc[1], 1)))
                if abs(rc[0] - self.red_prev[0]) <= 4 and abs(rc[1] - self.red_prev[1]) <= 4:
                    self.red_stable += 1
                    self.red_only_count += 1
                    self.shared_count += 1
                else:
                    self.red_only_count = 0
                    self.shared_count = 0
                self.red_prev = rc
        else:
            self.shared_count = 0          # ★ 红没找到 -> 把蓝攒的计数也清了
            self.red_only_count = 0

        if self.blue_only_count >= 30:
            self.found_water_blue_only = True
        if self.red_only_count >= 10:
            self.found_red_only = True
        if self.shared_count >= 30:
            self.found_water_shared = True

        if not self.saved:
            self.saved = True
            cv2.imwrite("/data/local/tmp/probe_frame.png", img)
            cv2.imwrite("/data/local/tmp/probe_blue_mask.png", blue_mask)
            cv2.imwrite("/data/local/tmp/probe_red_mask.png", red_mask)

                  % (self.frames, len(blue), str(bc), self.blue_only_count,
                     len(red), str(rc), self.shared_count))

def read_label_table(path):

def reference_box_process(position, anchors, img_size):
    grid_h, grid_w = position.shape[2:4]
    col, row = np.meshgrid(np.arange(0, grid_w), np.arange(0, grid_h))
    col = col.reshape(1, 1, grid_h, grid_w)
    row = row.reshape(1, 1, grid_h, grid_w)
    grid = np.concatenate((col, row), axis=1)
    stride = np.array([img_size[1] // grid_h, img_size[0] // grid_w]).reshape(1, 2, 1, 1)

    col = col.repeat(len(anchors), axis=0)
    row = row.repeat(len(anchors), axis=0)
    anchor_arr = np.array(anchors)
    anchor_arr = anchor_arr.reshape(*anchor_arr.shape, 1, 1)

    box_xy = position[:, :2, :, :] * 2 - 0.5
    box_wh = pow(position[:, 2:4, :, :] * 2, 2) * anchor_arr
    box_xy += grid
    box_xy *= stride
    box = np.concatenate((box_xy, box_wh), axis=1)

    xyxy = np.copy(box)
    xyxy[:, 0, :, :] = box[:, 0, :, :] - box[:, 2, :, :] / 2
    xyxy[:, 1, :, :] = box[:, 1, :, :] - box[:, 3, :, :] / 2
    xyxy[:, 2, :, :] = box[:, 0, :, :] + box[:, 2, :, :] / 2
    xyxy[:, 3, :, :] = box[:, 1, :, :] + box[:, 3, :, :] / 2
    return xyxy


def reference_nms_boxes(boxes, scores, nms_thresh):
    x, y = boxes[:, 0], boxes[:, 1]
    w, h = boxes[:, 2] - boxes[:, 0], boxes[:, 3] - boxes[:, 1]
    areas = w * h
    order = scores.argsort()[::-1]
    keep = []
    while order.size > 0:
        i = order[0]
        keep.append(i)
        xx1 = np.maximum(x[i], x[order[1:]])
        yy1 = np.maximum(y[i], y[order[1:]])
        xx2 = np.minimum(x[i] + w[i], x[order[1:]] + w[order[1:]])
        yy2 = np.minimum(y[i] + h[i], y[order[1:]] + h[order[1:]])
        inter = (np.maximum(0.0, xx2 - xx1 + 1e-5) * np.maximum(0.0, yy2 - yy1 + 1e-5))
        ovr = inter / (areas[i] + areas[order[1:]] - inter)
        order = order[1:][np.where(ovr <= nms_thresh)[0]]
    return np.array(keep)


def reference_post_process_v5(outputs, anchors, img_size, conf_thres, nms_thresh):
    boxes, scores, classes_conf = [], [], []
    for idx, data in enumerate(outputs):
        data = data.reshape([len(anchors[idx]), -1] + list(data.shape[-2:]))
        boxes.append(reference_box_process(data[:, :4, :, :], anchors[idx], img_size))
        scores.append(data[:, 4:5, :, :])
        classes_conf.append(data[:, 5:, :, :])

    def flatten(arr):
        return arr.transpose(0, 2, 3, 1).reshape(-1, arr.shape[1])

    boxes = np.concatenate([flatten(v) for v in boxes])
    classes_conf = np.concatenate([flatten(v) for v in classes_conf])
    scores = np.concatenate([flatten(v) for v in scores])

    class_max = np.max(classes_conf, axis=-1)
    classes = np.argmax(classes_conf, axis=-1)
    pos = np.where(class_max * scores.reshape(-1) >= conf_thres)
    boxes, classes, scores = boxes[pos], classes[pos], (class_max * scores.reshape(-1))[pos]

    out_boxes, out_classes, out_scores = [], [], []
    for cls in set(classes):
        idx = np.where(classes == cls)
        keep = reference_nms_boxes(boxes[idx], scores[idx], nms_thresh)
        if len(keep):
            out_boxes.append(boxes[idx][keep])
            out_classes.append(classes[idx][keep])
            out_scores.append(scores[idx][keep])
    if not out_boxes:
        return None, None, None
    return (np.concatenate(out_boxes), np.concatenate(out_classes),
            np.concatenate(out_scores))


def normalize(rows):
    """把两组框排成可比顺序：按 (类别, x1, y1) 排。"""
    order = np.lexsort((rows[:, 2], rows[:, 1], rows[:, 0]))
    return rows[order]

def make_synthetic_heads(n_classes=10, conf_value=0.9):
    """按给定 anchor 尺寸在三个尺度上各放一个"目标"，返回 (1,C,H,W) 概率头。"""
    heads = []
    for level, (grid, anchor_idx) in enumerate([(80, 0), (40, 1), (20, 2)]):
        channels = 3 * (5 + n_classes)
        head = np.zeros((1, channels, grid, grid), dtype=np.float32)
        data = head.reshape(3, 5 + n_classes, grid, grid)
        row, col = grid // 2, grid // 3 + level
        data[anchor_idx, 0, row, col] = 0.6      # tx
        data[anchor_idx, 1, row, col] = 0.4      # ty
        data[anchor_idx, 2, row, col] = 0.3      # tw
        data[anchor_idx, 3, row, col] = 0.5      # th
        data[anchor_idx, 4, row, col] = conf_value
        data[anchor_idx, 5 + (level % n_classes), row, col] = 0.95
        # 一堆"有点分数但过不了阈值"的噪声锚点，检验先过滤后解码不漏不误
        data[anchor_idx, 4, 0, :] = 0.30
        data[anchor_idx, 5 + (level % n_classes), 0, :] = 0.4
        heads.append(head)
    return heads

def test_label_parsing():
    """标签文件的三种写法都要认，且带空格的类名不能被截断。"""
    # 板上 /tmp 只读，临时文件放到本目录下
    workdir = tempfile.mkdtemp(prefix="sam24_labels_", dir=os.path.dirname(
        os.path.abspath(__file__)))
    try:
        three = os.path.join(workdir, "three.txt")
        with open(three, "w") as fh:
            fh.write("0 1 person\n1 46 wine glass\n2 88 teddy bear\n")
        names = load_labels(three)
        id_map = load_id_map(three)
        check("标签/三列格式名字不被截断", names == ["person", "wine glass", "teddy bear"],
              "names=%s" % (names,))
        check("标签/三列格式抽出映射表", id_map == [1, 46, 88], "id_map=%s" % (id_map,))

        plain = os.path.join(workdir, "plain.txt")
        with open(plain, "w") as fh:
            fh.write("# 注释行\nperson\nzebra shark\n")
        check("标签/单列格式只有名字",
              load_labels(plain) == ["person", "zebra shark"],
              "names=%s" % (load_labels(plain),))
        check("标签/单列格式不做映射", load_id_map(plain) is None)
    finally:
        shutil.rmtree(workdir, ignore_errors=True)


def test_runtime_label_table():
    """生产路径（``load_id_map``）读本包标签表，结果必须与第一层的数据断言一致。"""
    id_map = load_id_map(LABELS_PATH)
    labels = load_labels(LABELS_PATH)
    check("运行时/标签表解析出 80 类映射", id_map is not None and len(id_map) == 80,
          "len=%s" % (None if id_map is None else len(id_map),))
    if id_map is None or len(id_map) != 80:
        return
    check("运行时/西兰花经解析后对外 id 仍是 56", id_map[50] == 56 and labels[50] == "broccoli",
          "id_map[50]=%s name=%s" % (id_map[50], labels[50]))
